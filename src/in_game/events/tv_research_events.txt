# Towards Victory — Research Mechanism Events
# Covers: A interest accumulation, target selection, D start confirmation, D unlock notification.

namespace = tv_research

# ── tv_research.1 — Interest Collection (Sub-process A) ─────────────────────────
# Fires monthly via random_events when the research cycle is in phase 1 and A is incomplete.
# Player encourages inquiry (+1 interest) or dismisses (no effect).
tv_research.1 = {
	type = country_event
	title = tv_research.1.t
	desc = tv_research.1.d
	outcome = neutral

	trigger = {
		custom_tooltip = {
			text = TV_HAS_VARIABLE_SET_TT
			has_variable = tv_academy_io_member
		}
		custom_tooltip = {
			text = TV_HAS_VARIABLE_SET_TT
			has_variable = tv_research_phase
		}
		var:tv_research_phase = 1
		NOT = { var:tv_rm_subprocess_a >= 2 }
	}

	option = {
		name = tv_research.1.a	# "Encourage the inquiry"
		change_variable = { name = tv_research_interest add = 1 }
		if = {
			limit = { NOT = { var:tv_rm_subprocess_a >= 1 } }
			set_variable = { name = tv_rm_subprocess_a value = 1 }
		}
		if = {
			limit = { tv_research_check_a_trigger = yes }
			tv_research_complete_a_effect = yes
		}
	}
	option = {
		name = tv_research.1.b	# "Dismiss it"
	}
}

# ── tv_research.10 — Target Selection Proposal ───────────────────────────────────
# Fired by tv_request_research_target generic action.
# Proposes the advance stored in tv_research_proposed_advance.
# Player authorizes (starts cycle) or vetoes (12-month cooldown).
tv_research.10 = {
	type = country_event
	title = tv_research.10.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 1 }
				desc = tv_research.10.d_printing_press
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 2 }
				desc = tv_research.10.d_iron_working
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 3 }
				desc = tv_research.10.d_anatomy
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 4 }
				desc = tv_research.10.d_gunpowder
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 5 }
				desc = tv_research.10.d_blast_furnace
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 6 }
				desc = tv_research.10.d_scientific_experimentation
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 7 }
				desc = tv_research.10.d_scientific_revolution
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 8 }
				desc = tv_research.10.d_manufactories
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 9 }
				desc = tv_research.10.d_industrialization
			}
			triggered_desc = {
				trigger = { var:tv_research_proposed_advance = 10 }
				desc = tv_research.10.d_vaccination
			}
		}
	}
	outcome = neutral

	trigger = {
		custom_tooltip = {
			text = TV_HAS_VARIABLE_SET_TT
			has_variable = tv_research_proposed_advance
		}
		custom_tooltip = {
			text = TV_HAS_VARIABLE_NOT_SET_TT
			NOT = { has_variable = tv_research_target }
		}
	}

	option = {
		name = tv_research.10.a	# "Authorize the investigation"
		tv_research_confirm_target_effect = yes
	}
	option = {
		name = tv_research.10.b	# "Veto (12-month cooldown)"
		remove_variable = tv_research_proposed_advance
		set_variable = { name = tv_research_selection_cd value = 12 }
	}
}
