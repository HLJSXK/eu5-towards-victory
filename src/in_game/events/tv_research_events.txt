# Covers: A interest accumulation, target selection, D-ready notification, D start

namespace = tv_research

# ── tv_research.1 — Interest Collection (Sub-process A) ─────────────────────────
# Fires monthly via random_events when the research cycle is in phase 1 and A is incomplete.
# Player funds inquiry (+1 interest, dynamic gold cost) or dismisses (no effect).
tv_research.1 = {
	type = country_event
	title = tv_research.1.t
	desc = tv_research.1.d
	outcome = neutral

	trigger = {
		custom_tooltip = {
			text = TV_HAS_ACADEMY_IO_MEMBER_TT
			has_variable = tv_academy_io_member
		}
		custom_tooltip = {
			text = TV_HAS_RESEARCH_PHASE_TT
			has_variable = tv_research_phase
		}
		var:tv_research_phase = 1
		NOT = { var:tv_rm_subprocess_a >= 2 }
	}

	option = {
		name = tv_research.1.a	# "Fund the inquiry"
		change_gold_effect = { scale = -2 }
		change_variable = { name = tv_research_interest add = 1 }
		if = {
			limit = { NOT = { var:tv_rm_subprocess_a >= 1 } }
			set_variable = { name = tv_rm_subprocess_a value = 1 }
		}
		if = {
			limit = { tv_research_check_a_trigger = yes }
			tv_research_complete_a_effect = yes
		}
	}
	option = {
		name = tv_research.1.b	# "Dismiss it"
	}
}
