# ── Domestic Exhibition ───────────────────────────────────────────────────────
# Fired by the "Host Domestic Exhibition" action (costs 1000 ducats).
# Counts every work of art owned by this country and grants 0.1 International
# Influence per piece.
tv_arts_exhibition.1 = {
	type = country_event
	title = tv_arts_exhibition.1.t
	desc = tv_arts_exhibition.1.d
	outcome = good

	option = {
		name = tv_arts_exhibition.1.a
		custom_tooltip = tv_arts_exhibition.1.tt
		every_work_of_art_in_country = {
			root = {
				every_international_organizations_member_of = {
					limit = { international_organization_type = international_organization_type:tv_arts_exhibition }
					change_variable = { name = tv_arts_intl_influence add = 0.1 }
				}
			}
		}
	}
}

# ── Return Special: Artwork Brought Home ─────────────────────────────────────
# 2% chance on natural tour end (requires host artwork). The artwork is moved
# to the home country's capital.
# scope:finished_artist = returning artist. scope:exhibition_target = host country.
tv_arts_exhibition.120 = {
	type = country_event
	title = tv_arts_exhibition.120.t
	desc = tv_arts_exhibition.120.d
	outcome = good

	option = {
		name = tv_arts_exhibition.120.a
		scope:exhibition_target = {
			random_work_of_art_in_country = { save_scope_as = acquired_art }
		}
		scope:acquired_art = {
			move_art = root.capital
			set_art_owner = root
		}
	}
}
