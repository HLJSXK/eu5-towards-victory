# ── Touring Event: Artist Hired ───────────────────────────────────────────────
# Fired on the home country. Grants +1.0 International Influence.
# The artist remains permanently in the foreign court (tracking variables already
# cleared by the time this event is processed).
# scope:touring_artist = the hired artist character.
tv_arts_exhibition.104 = {
	type = country_event
	title = tv_arts_exhibition.104.t
	desc = tv_arts_exhibition.104.d
	outcome = good

	option = {
		name = tv_arts_exhibition.104.a
		tv_change_arts_intl_influence_effect = { value = 1.0 }
		hidden_effect = {
			scope:touring_artist = {
				var:tv_exhibition_target_country ?= {
					remove_variable = tv_has_visiting_artist
				}
			}
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
						has_variable = tv_dispatched_artists_count
					}
					var:tv_dispatched_artists_count >= 1
				}
				change_variable = { name = tv_dispatched_artists_count add = -1 }
			}
			tv_update_tour_capacity_effect = yes
			remove_list_variable = {
				name = tv_dispatched_artists_list
				target = scope:touring_artist
			}
			scope:touring_artist = {
				remove_variable = tv_on_exhibition
				remove_variable = tv_exhibition_home_country
				remove_variable = tv_exhibition_target_country
				remove_variable = tv_exhibition_location
				remove_variable = tv_exhibition_elapsed
				remove_variable = tv_exhibition_duration
				remove_variable = tv_tour_event_1_fired
				remove_variable = tv_tour_event_2_fired
				remove_variable = tv_tour_event_3_fired
				remove_variable = tv_tour_event_4_fired
				remove_variable = tv_tour_event_5_fired
			}
		}
	}
}

# ── Monthly Special: Artist Exchange ──────────────────────────────────────────
# 1/300 monthly chance (requires host artist). Both touring artist and a local
# artist permanently gain +0.05 artist_skill.
# scope:touring_artist = the touring artist. scope:local_artist = host artist.
tv_arts_exhibition.204 = {
	type = country_event
	title = tv_arts_exhibition.204.t
	desc = tv_arts_exhibition.204.d
	outcome = good

	immediate = {
		scope:touring_artist = {
			var:tv_exhibition_target_country ?= {
				random_character = {
					limit = {
						is_alive = yes
						is_artist = yes
						this != scope:touring_artist
					}
					save_scope_as = local_artist
				}
			}
		}
	}

	option = {
		name = tv_arts_exhibition.204.a
		scope:touring_artist = { add_artist_skill = 0.05 }
		scope:local_artist = { add_artist_skill = 0.05 }
	}
}

# ── Monthly Special: Art Theft + Artist Death ────────────────────────────────
# 1/300 monthly chance (requires host artwork). Artwork is destroyed, the
# touring artist dies, and the host country gains −20 permanent opinion.
# scope:touring_home fires this event. scope:theft_target = host country.
tv_arts_exhibition.205 = {
	type = country_event
	title = tv_arts_exhibition.205.t
	desc = tv_arts_exhibition.205.d
	outcome = bad

	immediate = {
		scope:touring_artist = {
			var:tv_exhibition_target_country ?= {
				save_scope_as = theft_target
				random_work_of_art_in_country = { save_scope_as = stolen_art }
			}
		}
	}

	option = {
		name = tv_arts_exhibition.205.a
		scope:theft_target = {
			add_opinion = { target = root  modifier = tv_art_theft_opinion }
		}
		destroy_art = scope:stolen_art
		hidden_effect = {
			scope:touring_artist = {
				var:tv_exhibition_target_country ?= { remove_variable = tv_has_visiting_artist }
			}
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
						has_variable = tv_dispatched_artists_count
					}
					var:tv_dispatched_artists_count >= 1
				}
				change_variable = { name = tv_dispatched_artists_count add = -1 }
			}
			tv_update_tour_capacity_effect = yes
			remove_list_variable = {
				name = tv_dispatched_artists_list
				target = scope:touring_artist
			}
			scope:touring_artist = {
				remove_variable = tv_on_exhibition
				remove_variable = tv_exhibition_home_country
				remove_variable = tv_exhibition_target_country
				remove_variable = tv_exhibition_location
				remove_variable = tv_exhibition_elapsed
				remove_variable = tv_exhibition_duration
				remove_variable = tv_tour_event_1_fired
				remove_variable = tv_tour_event_2_fired
				remove_variable = tv_tour_event_3_fired
				remove_variable = tv_tour_event_4_fired
				remove_variable = tv_tour_event_5_fired
			}
		}
		scope:touring_artist = { kill_character = this }
	}
}
