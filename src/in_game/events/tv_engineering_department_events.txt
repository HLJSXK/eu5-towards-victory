tv_engineering_department.300 = {
	type = country_event
	title = tv_engineering_department.300.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { var:tv_wonder_survey_competence_average >= 75 }
				desc = tv_engineering_department.300.d_high
			}
			triggered_desc = {
				trigger = { var:tv_wonder_survey_competence_average >= 50 }
				desc = tv_engineering_department.300.d_mid
			}
			desc = tv_engineering_department.300.d_low
		}
	}
	outcome = good

	immediate = {
		if = {
			limit = { tv_wonder_survey_site_selected_trigger = yes }
			var:tv_wonder_survey_site ?= { save_scope_as = tv_wonder_event_location }
		}
		set_variable = { name = tv_wonder_survey_competence_average value = var:tv_wonder_scale_competence }
		change_variable = { name = tv_wonder_survey_competence_average add = var:tv_wonder_logistics_competence }
		change_variable = { name = tv_wonder_survey_competence_average add = var:tv_wonder_organization_competence }
		change_variable = { name = tv_wonder_survey_competence_average divide = 3 }
	}

	option = {
		name = tv_engineering_department.300.a
	}
}
