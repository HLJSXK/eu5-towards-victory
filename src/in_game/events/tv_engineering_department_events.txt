namespace = tv_engineering_department

# Fired on the leader_country when the appointed Great Engineer dies.
# The option clears tv_great_engineer_char to formally dismiss them.

	type = country_event
	title = tv_engineering_department.10.t
	desc = tv_engineering_department.10.d
	outcome = bad

	option = {
		name = tv_engineering_department.10.a
		tv_engineering_on_great_engineer_removed_effect = yes
		remove_variable = tv_great_engineer_char
	}
}
