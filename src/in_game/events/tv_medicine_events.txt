# Towards Victory — Medicine mechanic events
# Namespace: tv_medicine
# Events .1/.2/.3/.4 cover the full treatment lifecycle.

namespace = tv_medicine

# ── tv_medicine.1 — Critical Medical Finding ──────────────────────────────────
# Fired monthly via random_events when treatment is active (~1% chance).
# Player chooses a treatment approach; outcomes may advance progress or kill the patient.
tv_medicine.1 = {
	type = country_event
	title = tv_medicine.1.t
	desc = tv_medicine.1.d
	outcome = neutral

	trigger = {
		custom_tooltip = {
			text = TV_HAS_MED_TARGET_CHAR_TT
			has_variable = tv_med_target_char
		}
		var:tv_med_target_char ?= { is_alive = yes }
	}

	# ── Conservative: 95% +5 progress, 5% patient dies ────────────────────────
	option = {
		name = tv_medicine.1.a
		random_list = {
			95 = {
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_MED_PROGRESS_TT
							has_variable = tv_med_progress
						}
					}
					change_variable = { name = tv_med_progress  add = 5 }
				}
			}
			5 = {
				var:tv_med_target_char ?= { kill_character = this }
				tv_med_cleanup_effect = yes
			}
		}
	}

	# ── Funded/Risky: 80% +10 progress, 20% patient dies ─────────────────────
	option = {
		name = tv_medicine.1.b
		random_list = {
			80 = {
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_MED_PROGRESS_TT
							has_variable = tv_med_progress
						}
					}
					change_variable = { name = tv_med_progress  add = 10 }
				}
			}
			20 = {
				var:tv_med_target_char ?= { kill_character = this }
				tv_med_cleanup_effect = yes
			}
		}
	}

	# ── Continue research — no intervention ───────────────────────────────────
	option = {
		name = tv_medicine.1.c
	}
}

# ── tv_medicine.2 — Minor Medical Accident ────────────────────────────────────
# 2% monthly chance. Immediately sets back treatment progress by 10.
tv_medicine.2 = {
	type = country_event
	title = tv_medicine.2.t
	desc = tv_medicine.2.d
	outcome = negative

	trigger = {
		custom_tooltip = {
			text = TV_HAS_MED_TARGET_CHAR_TT
			has_variable = tv_med_target_char
		}
	}

	immediate = {
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_MED_PROGRESS_TT
					has_variable = tv_med_progress
				}
			}
			change_variable = { name = tv_med_progress  add = -10 }
			if = {
				limit = { var:tv_med_progress < 0 }
				set_variable = { name = tv_med_progress  value = 0 }
			}
		}
	}

	option = {
		name = tv_medicine.2.a
	}
}

# ── tv_medicine.22 — A Character Seeks Meditation ───────────────────────────
tv_medicine.22 = {
	type = country_event
	title = tv_medicine.22.t
	desc = tv_medicine.22.d
	outcome = neutral

	trigger = {
		has_variable = tv_academy_io_member
		NOT = { has_variable = tv_meditation_target_char }
		has_variable = tv_academy_leader_char
		var:tv_academy_leader_char ?= { is_alive = yes }
		tv_chief_scientist_available_for_meditation_trigger = yes
		any_character = {
			is_alive = yes
			tv_has_meditation_negative_trait_trigger = yes
		}
	}

	immediate = {
		random_character = {
			limit = {
				is_alive = yes
				tv_has_meditation_negative_trait_trigger = yes
			}
			save_scope_as = tv_meditation_target
		}
		set_variable = { name = tv_meditation_target_char  value = scope:tv_meditation_target }
		set_variable = { name = tv_meditation_progress  value = 0 }
		if = {
			limit = { scope:tv_meditation_target = { is_ruler = yes } }
			add_country_modifier = { modifier = tv_meditating_ruler_country_modifier years = -1 mode = add_and_extend }
		}
		scope:tv_meditation_target = {
			add_character_modifier = { modifier = tv_meditating_character_modifier years = -1 mode = add_and_extend }
		}
	}

	option = {
		name = tv_medicine.22.a
	}
}
