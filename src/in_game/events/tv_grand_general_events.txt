# Towards Victory — Grand General (大将军) training outcome events
# Fired by tv_training_resolve_effect (tv_govhouse_effects.txt) after a 1-year training period.
# Root/scope: country that owns the Grand General.
# The Grand General character is accessed via var:tv_conquest_general_char on the root country.

namespace = tv_grand_general

# ── Outcome 1: Mil skill gain (weight 49) ────────────────────────────────────
# Fired when mil < 100. The gain is capped at 100.
tv_grand_general.1 = {
	type = country_event
	title = tv_grand_general.1.t
	desc = tv_grand_general.1.d
	outcome = neutral

	option = {
		name = tv_grand_general.1.a
		var:tv_conquest_general_char ?= {
			add_mil = { value = 10 max = 100 }
		}
	}
}

# ── Outcome 2: 世间无双 — ruler +3 mil (mil = 100 on skill-gain path) ────────
tv_grand_general.2 = {
	type = country_event
	title = tv_grand_general.2.t
	desc = tv_grand_general.2.d
	outcome = positive

	option = {
		name = tv_grand_general.2.a
		ruler_or_regent ?= {
			add_mil = { value = 3 max = 100 }
		}
	}
}

# ── Outcome 3: Trait upgrade (weight 49) ────────────────────────────────────
# Player chooses offense / defense / tactics path. Each option upgrades that path's trait level.
# If the chosen path is already at L3, the event should not offer that option (use trigger).
# Routing to events 4/5 (百炼成钢/世间无双) is handled before this event fires.
tv_grand_general.3 = {
	type = country_event
	title = tv_grand_general.3.t
	desc = tv_grand_general.3.d
	outcome = neutral

	# Option A: Offense path
	option = {
		name = tv_grand_general.3.a
		trigger = {
			var:tv_conquest_general_char ?= {
				NOT = { has_trait = tv_gg_offense_3 }
			}
		}
		var:tv_conquest_general_char ?= {
			if = {
				limit = { has_trait = tv_gg_offense_2 }
				if = {
					limit = { has_trait = tv_gg_offense_1 }
					remove_trait = trait:tv_gg_offense_1
				}
				remove_trait = trait:tv_gg_offense_2
				add_trait = trait:tv_gg_offense_3
			}
			else_if = {
				limit = { has_trait = tv_gg_offense_1 }
				remove_trait = trait:tv_gg_offense_1
				add_trait = trait:tv_gg_offense_2
			}
			else = {
				add_trait = trait:tv_gg_offense_1
			}
		}
	}

	# Option B: Defense path
	option = {
		name = tv_grand_general.3.b
		trigger = {
			var:tv_conquest_general_char ?= {
				NOT = { has_trait = tv_gg_defense_3 }
			}
		}
		var:tv_conquest_general_char ?= {
			if = {
				limit = { has_trait = tv_gg_defense_2 }
				if = {
					limit = { has_trait = tv_gg_defense_1 }
					remove_trait = trait:tv_gg_defense_1
				}
				remove_trait = trait:tv_gg_defense_2
				add_trait = trait:tv_gg_defense_3
			}
			else_if = {
				limit = { has_trait = tv_gg_defense_1 }
				remove_trait = trait:tv_gg_defense_1
				add_trait = trait:tv_gg_defense_2
			}
			else = {
				add_trait = trait:tv_gg_defense_1
			}
		}
	}

	# Option C: Tactics path
	option = {
		name = tv_grand_general.3.c
		trigger = {
			var:tv_conquest_general_char ?= {
				NOT = { has_trait = tv_gg_tactics_3 }
			}
		}
		var:tv_conquest_general_char ?= {
			if = {
				limit = { has_trait = tv_gg_tactics_2 }
				if = {
					limit = { has_trait = tv_gg_tactics_1 }
					remove_trait = trait:tv_gg_tactics_1
				}
				remove_trait = trait:tv_gg_tactics_2
				add_trait = trait:tv_gg_tactics_3
			}
			else_if = {
				limit = { has_trait = tv_gg_tactics_1 }
				remove_trait = trait:tv_gg_tactics_1
				add_trait = trait:tv_gg_tactics_2
			}
			else = {
				add_trait = trait:tv_gg_tactics_1
			}
		}
	}
}
