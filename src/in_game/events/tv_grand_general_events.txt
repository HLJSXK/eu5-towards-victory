# Towards Victory — Grand General (大将军) training outcome events
# Fired by tv_training_resolve_effect (tv_govhouse_effects.txt) after a 1-year training period.
# Root/scope: country that owns the Grand General.
# The Grand General character is accessed via var:tv_conquest_general_char on the root country.

namespace = tv_grand_general

# ── Outcome 1: Mil skill gain (weight 49) ────────────────────────────────────
# If mil < 100 → +10 mil. If mil = 100 → 世间无双 (fire event 2 instead).
tv_grand_general.1 = {
	type = country_event
	title = tv_grand_general.1.t
	desc = tv_grand_general.1.d
	outcome = neutral

	immediate = {
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
					has_variable = tv_conquest_general_char
				}
				var:tv_conquest_general_char ?= { mil >= 100 }
			}
			# Redirect: mil already maxed → 世间无双
			trigger_event_non_silently = { id = tv_grand_general.2 }
			# Hide this event since we fired the replacement
			hidden_effect = { set_variable = { name = tv_gg_skip_event value = 1 } }
		}
	}

	option = {
		name = tv_grand_general.1.a
		# TODO: var:tv_conquest_general_char = { add_character_skill = { skill = mil value = 10 } }
		# Verify add_character_skill syntax against reference_game_files before implementing.
		custom_tooltip = { text = tv_gg_mil_gain_todo_tt }
	}
}

# ── Outcome 3: Trait upgrade (weight 49) ────────────────────────────────────
# Player chooses offense / defense / tactics path. Each option upgrades that path's trait level.
# If the chosen path is already at L3, the event should not offer that option (use trigger).
# Routing to events 4/5 (百炼成钢/世间无双) is handled by the option effects.
tv_grand_general.3 = {
	type = country_event
	title = tv_grand_general.3.t
	desc = tv_grand_general.3.d
	outcome = neutral

	# Option A: Offense path
	option = {
		name = tv_grand_general.3.a
		trigger = {
			# Only show if offense path is not yet at L3
			OR = {
				NOT = {
					var:tv_conquest_general_char ?= { has_trait = tv_gg_offense_1 }
					var:tv_conquest_general_char ?= { has_trait = tv_gg_offense_2 }
					var:tv_conquest_general_char ?= { has_trait = tv_gg_offense_3 }
				}
				var:tv_conquest_general_char ?= { has_trait = tv_gg_offense_1 }
				var:tv_conquest_general_char ?= { has_trait = tv_gg_offense_2 }
			}
		}
		# TODO: upgrade offense trait:
		#   if tv_gg_offense_2 → remove + add tv_gg_offense_3
		#   elif tv_gg_offense_1 → remove + add tv_gg_offense_2
		#   else → add tv_gg_offense_1
		# Verify add_trait / remove_trait syntax on characters.
		custom_tooltip = { text = tv_gg_offense_upgrade_todo_tt }
	}

	# Option B: Defense path
	option = {
		name = tv_grand_general.3.b
		trigger = {
			OR = {
				NOT = {
					var:tv_conquest_general_char ?= { has_trait = tv_gg_defense_1 }
					var:tv_conquest_general_char ?= { has_trait = tv_gg_defense_2 }
					var:tv_conquest_general_char ?= { has_trait = tv_gg_defense_3 }
				}
				var:tv_conquest_general_char ?= { has_trait = tv_gg_defense_1 }
				var:tv_conquest_general_char ?= { has_trait = tv_gg_defense_2 }
			}
		}
		custom_tooltip = { text = tv_gg_defense_upgrade_todo_tt }
	}

	# Option C: Tactics path
	option = {
		name = tv_grand_general.3.c
		trigger = {
			OR = {
				NOT = {
					var:tv_conquest_general_char ?= { has_trait = tv_gg_tactics_1 }
					var:tv_conquest_general_char ?= { has_trait = tv_gg_tactics_2 }
					var:tv_conquest_general_char ?= { has_trait = tv_gg_tactics_3 }
				}
				var:tv_conquest_general_char ?= { has_trait = tv_gg_tactics_1 }
				var:tv_conquest_general_char ?= { has_trait = tv_gg_tactics_2 }
			}
		}
		custom_tooltip = { text = tv_gg_tactics_upgrade_todo_tt }
	}
}
