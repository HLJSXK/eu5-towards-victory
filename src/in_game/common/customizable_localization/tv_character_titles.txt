# Towards Victory — character title prefixes for IO leaders
# Evaluated by the engine as part of character_title_prefix (additive with vanilla entries).
# Triggers on characters who hold tv_chief_artist_role_modifier or tv_chief_scientist_role_modifier.

character_title_prefix = {
	type = character

	text = {
		localization_key = character_title_tv_chief_artist
		trigger = { has_character_modifier = tv_chief_artist_role_modifier }
	}

	text = {
		localization_key = character_title_tv_chief_scientist
		trigger = { has_character_modifier = tv_chief_scientist_role_modifier }
	}
}
