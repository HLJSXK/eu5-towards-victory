pa_tv_alliance_delegation_subsidies = {
	type = international_organization
	special_status = tv_alliance_member_status
	importance = 1.25
	potential = {
		international_organization_type = international_organization_type:tv_diplomatic_alliance
		NOT = { has_cooldown = pa_tv_alliance_delegation_subsidies }
	}
	allow = {
		is_leader_of_international_organization = scope:recipient
		scope:recipient.parliament_issue ?= {
			this != parliament_issue:hre_law_debate
		}
		gold >= 100
	}
	on_accept = {
		hidden_effect = {
			add_cooldown = { type = pa_tv_alliance_delegation_subsidies years = 5 }
		}
		leader_country ?= {
			add_gold = -100
		}
		every_international_organization_member = {
			limit = {
				has_special_status_in_international_organization = {
					type = scope:target
					international_organization = root
				}
			}
			add_country_modifier = {
				modifier = tv_alliance_delegation_subsidy_modifier
				years = 3
				mode = add_and_extend
			}
		}
	}
	ai_will_do = {
		add = 5
	}
	chance = 10
}
