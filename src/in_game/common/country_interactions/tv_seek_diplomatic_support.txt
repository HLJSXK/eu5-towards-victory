tv_seek_diplomatic_support = {
	type = diplomacy
	category = CATEGORY_FRIENDLY_ACTIONS

	cooldown = {
		type = tv_seek_diplomatic_support
		years = 1
	}

	ai_tick = monthly
	ai_tick_frequency = 60

	potential = {
		NOT = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= this
			}
		}
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		ai_interaction_source_list = {
			scope:actor = {
				every_friendly_or_high_opinion_country = {
					add_to_list = source
				}
			}
		}
		visible = {
			scope:actor = {
				NOT = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
						leader_country ?= scope:actor
					}
				}
			}
		}
		enabled = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_diplomatic_supporter_of }
			}
			custom_tooltip = {
				text = TV_NOT_IN_DIPLOMATIC_ALLIANCE_TT
				NOT = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
					}
				}
			}
			is_subject = no
		}
		show_why_not_enabled = yes
	}

	effect = {
		# Mark recipient as a diplomatic supporter of actor
		scope:recipient = {
			set_variable = { name = tv_diplomatic_supporter_of value = scope:actor }
		}
		# Increment actor's support count
		scope:actor = {
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_VARIABLE_NOT_SET_TT
						NOT = { has_variable = tv_diplomatic_support_count }
					}
				}
				set_variable = { name = tv_diplomatic_support_count value = 0 }
			}
			change_variable = { name = tv_diplomatic_support_count add = 1 }
			if = {
				limit = {
					var:tv_diplomatic_support_count >= 2
					NOT = {
						any_international_organizations_member_of = {
							international_organization_type = international_organization_type:tv_diplomatic_alliance
							leader_country ?= scope:actor
						}
					}
				}
				save_scope_as = tv_founder
				create_international_organization = {
					type = international_organization_type:tv_diplomatic_alliance
					add_country_to_international_organization = prev
					set_leader_country = prev
					save_scope_as = tv_alliance_io
					every_country = {
						limit = {
							var:tv_diplomatic_supporter_of ?= scope:tv_founder
						}
						scope:tv_alliance_io = {
							add_country_to_international_organization = prev
						}
					}
					set_variable = { name = tv_alliance_cohesion value = 0 }
					set_variable = { name = tv_alliance_tier value = 0 }
				}
				set_variable = { name = tv_alliance_tier value = 0 }
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_VARIABLE_NOT_SET_TT
							NOT = { has_variable = tv_diplomatic_victory_points }
						}
					}
					set_variable = { name = tv_diplomatic_victory_points value = 1 }
				}
				else_if = {
					limit = { var:tv_diplomatic_victory_points < 1 }
					set_variable = { name = tv_diplomatic_victory_points value = 1 }
				}
			}
		}
	}

	ai_will_do = {
		if = {
			limit = {
				scope:actor = {
					OR = {
						custom_tooltip = {
							text = TV_HAS_VARIABLE_NOT_SET_TT
							NOT = { has_variable = tv_diplomatic_support_count }
						}
						var:tv_diplomatic_support_count < 15
					}
				}
			}
			add = 5
		}
	}

	diplo_chance = {
		base = -200
		opinion = 0.5
		same_culture = 50
	}

	accept = {
		# Same religion bonus (+50)
		if = {
			limit = {
				scope:recipient = {
					religion = scope:actor.religion
				}
			}
			add = {
				desc = "TV_SAME_RELIGION"
				value = 50
			}
		}
		# Diplomatic reputation bonus: +5 per point
		add = {
			desc = "TV_DIPLOMATIC_REPUTATION"
			value = scope:actor.modifier:diplomatic_reputation
			multiply = 5
		}
	}
}
