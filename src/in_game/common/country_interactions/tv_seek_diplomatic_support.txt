tv_seek_diplomatic_support = {
	type = diplomacy
	category = CATEGORY_FRIENDLY_ACTIONS

	cooldown = {
		type = tv_seek_diplomatic_support
		years = 1
	}

	ai_tick = monthly
	ai_tick_frequency = 60

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		ai_interaction_source_list = {
			scope:actor = {
				every_friendly_or_high_opinion_country = {
					add_to_list = source
				}
			}
		}
		visible = {
			# Cannot already be a supporter of any country
			NOT = { has_variable = tv_diplomatic_supporter_of }
			is_subject = no
		}
		enabled = {
		}
	}

	effect = {
		# Mark recipient as a diplomatic supporter of actor
		scope:recipient = {
			set_variable = { name = tv_diplomatic_supporter_of value = scope:actor }
		}
		# Increment actor's support count
		scope:actor = {
			if = {
				limit = { NOT = { has_variable = tv_diplomatic_support_count } }
				set_variable = { name = tv_diplomatic_support_count value = 0 }
			}
			change_variable = { name = tv_diplomatic_support_count add = 1 }
		}
		# If our Diplomatic Alliance already exists, add supporter to it
		scope:actor = {
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_diplomatic_alliance
					leader_country = scope:actor
				}
				add_country_to_international_organization = scope:recipient
			}
		}
	}

	ai_will_do = {
		if = {
			limit = {
				scope:actor = {
					OR = {
						NOT = { has_variable = tv_diplomatic_support_count }
						var:tv_diplomatic_support_count < 15
					}
				}
			}
			add = 5
		}
	}

	diplo_chance = {
		base = -200
		opinion = 0.5
		same_culture = 50
	}

	accept = {
		# Same religion bonus (+50)
		if = {
			limit = {
				scope:recipient = {
					religion = scope:actor.religion
				}
			}
			add = {
				desc = "TV_SAME_RELIGION"
				value = 50
			}
		}
		# Diplomatic reputation bonus: +5 per point
		add = {
			desc = "TV_DIPLOMATIC_REPUTATION"
			value = scope:actor.modifier:diplomatic_reputation
			multiply = 5
		}
	}
}
