tv_invite_to_diplomatic_alliance = {
	type = diplomacy
	category = CATEGORY_FRIENDLY_ACTIONS
	ai_tick = monthly
	ai_tick_frequency = 60

	potential = {
		scope:actor = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:actor
			}
		}
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		ai_interaction_source_list = {
			scope:actor = {
				every_friendly_or_high_opinion_country = {
					add_to_list = source
				}
			}
		}
		visible = {
		}
		enabled = {
			is_subject = no
			custom_tooltip = {
				text = TV_NOT_IN_DIPLOMATIC_ALLIANCE_TT
				NOT = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
					}
				}
			}
		}
		show_why_not_enabled = yes
	}

	effect = {
		if = {
			limit = { exists = scope:recipient }
			scope:actor = {
				every_international_organizations_member_of = {
					limit = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
						leader_country ?= scope:actor
					}
					add_country_to_international_organization = scope:recipient
				}
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_VARIABLE_NOT_SET_TT
							NOT = { has_variable = tv_diplomatic_alliance_member_count }
						}
					}
					set_variable = { name = tv_diplomatic_alliance_member_count value = 1 }
				}
				change_variable = { name = tv_diplomatic_alliance_member_count add = 1 }
			}
		}
	}

	ai_will_do = {
		add = 5
	}

	diplo_chance = {
		base = -100
		opinion = 0.5
		same_culture = 50
	}

	accept = {
		if = {
			limit = {
				scope:actor = { has_variable = tv_diplomatic_alliance_member_count }
			}
			add = {
				desc = "TV_ALLIANCE_SIZE"
				value = scope:actor.var:tv_diplomatic_alliance_member_count
				multiply = -10
			}
		}
		if = {
			limit = { exists = scope:recipient }
			add = {
				desc = "TV_RECIPIENT_OWNED_LOCATIONS"
				value = scope:recipient.num_locations
				multiply = -1
			}
			add = {
				desc = "TV_PREFERS_OWN_ALLIANCE"
				value = scope:recipient.modifier:diplomatic_reputation
				multiply = -5
			}
			if = {
				limit = {
					scope:recipient = {
						religion = scope:actor.religion
					}
				}
				add = {
					desc = "TV_SAME_RELIGION"
					value = 50
				}
			}
		}
		add = {
			desc = "TV_DIPLOMATIC_REPUTATION"
			value = scope:actor.modifier:diplomatic_reputation
			multiply = 5
		}
	}
}
