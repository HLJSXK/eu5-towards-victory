# Towards Victory — Locked Advance Gates (Phase 5)
#
# Connects Academy of Sciences research unlock variables to EU5 advance allow blocks.
# Initial state: unlock variables are unset → has_variable returns false → advance locked.
# After research cycle: set_variable fires → has_variable = true → advance researchable.
#
# REPLACE used for advances that already have an allow block (institution-based);
#   original institution condition preserved in OR so vanilla path still works.
#   Verification — Step 3, Reference: reference_mods/3649381461/in_game/common/advances/
#     mod_improved_trade_and_buildings_advances.txt:1
#     Quote: "REPLACE:terraces_advance = { age = ... }"
#
# INJECT used for advances with no existing allow block.
#   Verification — Step 3, Reference: reference_mods/3599735023/in_game/common/advances/
#     iv2_advances.txt:9
#     Quote: "INJECT:university_advance = { unlock_building = iv2_building_9_0 }"

# ── REPLACE: advances with existing institution-based allow blocks ────────────

# Age 1 — iron_working (ref: 0_age_of_traditions.txt:344)
# Original allow: has_embraced_institution = institution:feudalism
REPLACE:iron_working = {
	icon = efficient_mining
	age = age_1_traditions
	requires = metallurgy
	research_cost = 2.0
	starting_technology_level = 1
	allow = {
		OR = {
			has_variable = tv_advance_iron_working_unlocked
			has_embraced_institution = institution:feudalism
		}
	}
	can_extract_coal = yes
	can_extract_iron = yes
	unlock_building = bog_iron_smelter
}
