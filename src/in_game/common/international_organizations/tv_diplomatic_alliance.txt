tv_diplomatic_alliance = {
	has_target = no
	unique = no
	expel_members_who_are_targets_of_other_members = no
	has_buildings = no
	show_leave_message = yes
	use_laws_as_join_reason = no

	has_leader_country = yes
	leader_title_key = "TV_ALLIANCE_ADVOCATOR"
	use_regnal_number = no
	leader_type = character
	leader = {
		leader_country ?= {
			var:tv_diplomatic_alliance_leader_char ?= {
				add_to_list = leaders
			}
		}
	}
	leader_change_trigger_type = none
	leader_change_method = none
	disband_if_no_leader = no

	has_parliament = yes
	parliament_type = tv_alliance_assembly
	resolution_widget = parliament
	special_statuses_implemented = { tv_alliance_advocator tv_alliance_member_status }

	laws = {
		tv_court_protocol = tv_adhoc_envoys_policy
		tv_information_network = tv_royal_couriers_policy
		tv_commerce_consensus = tv_non_piracy_policy
		tv_faith_harmony = tv_cuius_regio_policy
		tv_colonization_sphere = tv_free_colonization_policy
	}

	variables = {
		# Tier is first so DataModelFirst in the GUI decoration picks it up as the large piechart
		tv_alliance_tier = {
			format = "TV_TIER_DISPLAY"
			min = 0
			max = 15
		}
		tv_alliance_cohesion = {
			format = "TV_COHESION_DISPLAY"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				add = {
					desc = "TV_COHESION_BASE_GROWTH"
					value = 0.1
				}
				if = {
					limit = { international_organization_has_leader = yes }
					add = {
						desc = "TV_COHESION_DIPLOMATIC_REP"
						value = leader_country.modifier:diplomatic_reputation
						multiply = 0.01
					}
				}
				if = {
					limit = {
						NOT = {
							any_international_organization_member = {
								at_war = yes
							}
						}
					}
					add = {
						desc = "TV_COHESION_INTERNAL_PEACE"
						value = 0.05
					}
				}
			}
		}
	}

	can_initiate_policy_votes = {
		is_leader_of_international_organization = scope:recipient
	}

	can_vote_in_parliament = {
		"country_highest_rated_special_status_power(scope:recipient)" > 0
	}

	show_on_diplomatic_map = yes

	can_join_trigger = {
		is_subject = no
		is_rebel_country = no
	}

	on_creation = {
		set_parliament_type = parliament_type:tv_alliance_assembly
		tv_diplomatic_alliance_stamp_creation_order_effect = yes
		set_variable = { name = tv_alliance_cohesion value = 0 }
		set_variable = { name = tv_alliance_tier value = 0 }
		if = {
			limit = { exists = scope:actor }
			set_leader_country = scope:actor
		}
	}

	on_joined = {
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				international_organization_add_special_status = {
					type = special_status:tv_alliance_member_status
					country = root
				}
			}
		}
		set_variable = { name = tv_diplomatic_alliance_member value = 1 }
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_VARIABLE_NOT_SET_TT
					NOT = { has_variable = tv_diplomatic_victory_points }
				}
			}
			set_variable = { name = tv_diplomatic_victory_points value = 0 }
		}
		set_variable = { name = tv_diplomatic_victory_enabled value = 1 }
		tv_update_all_progress_pct_effect = yes
		if = {
			limit = {
				NOT = { has_variable = tv_diplomatic_alliance_self_establishing }
				NOT = { is_leader_of_international_organization = prev }
			}
			tv_check_diplomatic_milestones_effect = yes
		}
	}

	on_left = {
		if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				international_organization_remove_special_status = {
					type = special_status:tv_alliance_advocator
					country = root
				}
				international_organization_remove_special_status = {
					type = special_status:tv_alliance_member_status
					country = root
				}
			}
		}
		remove_variable = tv_diplomatic_alliance_member
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

}
