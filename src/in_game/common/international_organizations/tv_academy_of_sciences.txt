# Towards Victory — Academy of Sciences International Organization type definition
# Non-unique IO; each country pursuing Scientific Victory creates their own instance
# through the Scientific IO establishment flow.
# Country membership flag: tv_academy_io_member (set to 1 when joining).
# Research subprocess state and target advance are country-scoped variables.
# C and D progress tracked as IO variables (vanilla variable system).
#
# Leader: character (the "Chief Scientist"). Selected manually by the leader_country via
# the Appoint / Remove / Change Leader generic actions (tv_io_leader_actions.txt).
# The leader country variable tv_academy_leader_char holds the appointed character.
# D-progress uses the appointed leader's effective ADM skill.
#
# Verification — Step 3, Reference: hre.txt:19-44
#   Quote: "has_leader_country = yes  leader_title_key = "HRE_LEADER"  leader_type = character
#           leader = { leader_country ?= { ... add_to_list = leaders } }"
# NOTE: The vanilla IO "choose new leader" effect is GLOBALLY BANNED on TV IOs.
#   It triggers the leader-election process, violating the no-elections design.
#   leader_country is locked to the founding country via on_creation and never reassigned.

tv_academy_of_sciences = {
	has_target = no
	unique = no
	has_leader_country = yes
	leader_title_key = "TV_ACADEMY_IO_LEADER"
	leader_type = character
	leader_change_trigger_type = none
	leader_change_method = none
	disband_if_no_leader = no
	show_on_diplomatic_map = no

	# Keep the founder-only Academy focused on Laws, Chief Scientist, and Medicine.
	# The shared IO parliament tab is tied to HasParliament in vanilla common.gui.
	has_parliament = no

	leader = {
		leader_country ?= {
			var:tv_academy_leader_char ?= {
				add_to_list = leaders
			}
		}
	}

	# ── IO Variables: C and D research progress, Philosophy inspiration ───────
	# These are auto-accumulated via monthly_change; completion is detected by
	# tv_academy_of_sciences_monthly_pulse.
	# Calibration C: average_country_literacy * 0.05 /month → 100 in ~40 months at literacy=50.
	# Calibration D: effective ADM * 0.05 /month.
	# D only advances and completes when the Chief Scientist has at least 20 effective ADM.
	# At 50 effective ADM, D gains 2.5/month and finishes in roughly 40 months.
	# D halts when no leader appointed (international_organization_has_leader = no).
	variables = {
		tv_research_c_progress = {
			format = "TV_C_PROGRESS_FORMAT"
			change_format = "TV_C_PROGRESS_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				add = {
					desc = "TV_C_LITERACY_CONTRIBUTION"
					value = 0
					if = {
						limit = {
							leader_country ?= {
								custom_tooltip = {
									text = TV_HAS_RESEARCH_PHASE_TT
									has_variable = tv_research_phase
								}
								var:tv_research_phase ?= 1
								NOT = { var:tv_rm_subprocess_c ?= { this >= 2 } }
							}
						}
						value = leader_country.average_country_literacy
						multiply = 0.05
					}
				}
			}
		}
		tv_research_d_progress = {
			format = "TV_D_PROGRESS_FORMAT"
			change_format = "TV_D_PROGRESS_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				add = {
					desc = "TV_D_RESEARCH_CONTRIBUTION"
					value = 0
					if = {
						limit = {
							international_organization_has_leader = yes
							leader_country ?= {
								custom_tooltip = {
									text = TV_HAS_RESEARCH_PHASE_TT
									has_variable = tv_research_phase
								}
								var:tv_research_phase ?= 2
								custom_tooltip = {
									text = TV_HAS_ACADEMY_LEADER_CHAR_TT
									has_variable = tv_academy_leader_char
								}
								tv_chief_scientist_effective_adm_at_least_20_trigger = yes
							}
						}
						value = leader_country.var:tv_academy_leader_char.adm
						multiply = 0.05
					}
				}
				add = {
					desc = "TV_D_RESEARCH_TRAIT_BONUS"
					value = 0
					if = {
						limit = {
							international_organization_has_leader = yes
							leader_country ?= {
								var:tv_research_phase ?= 2
								tv_chief_scientist_effective_adm_at_least_20_trigger = yes
								var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_3 }
							}
						}
						value = 60
					}
					else_if = {
						limit = {
							international_organization_has_leader = yes
							leader_country ?= {
								var:tv_research_phase ?= 2
								tv_chief_scientist_effective_adm_at_least_20_trigger = yes
								var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_2 }
							}
						}
						value = 40
					}
					else_if = {
						limit = {
							international_organization_has_leader = yes
							leader_country ?= {
								var:tv_research_phase ?= 2
								tv_chief_scientist_effective_adm_at_least_20_trigger = yes
								var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_1 }
							}
						}
						value = 20
					}
					multiply = 0.05
				}
			}
		}
		tv_academy_philosophy_inspiration = {
			format = "TV_PHILOSOPHY_INSPIRATION_FORMAT"
			change_format = "TV_PHILOSOPHY_INSPIRATION_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				add = {
					desc = "TV_PHILOSOPHY_IDLE_CHIEF_SCIENTIST"
					value = 0
					if = {
						limit = {
							international_organization_has_leader = yes
							leader_country ?= {
								has_variable = tv_academy_leader_char
								var:tv_academy_leader_char ?= { is_alive = yes }
								tv_chief_scientist_available_for_philosophy_trigger = yes
							}
						}
						value = leader_country.var:tv_academy_leader_char.adm
						if = {
							limit = {
								leader_country ?= {
									var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_3 }
								}
							}
							add = 60
						}
						else_if = {
							limit = {
								leader_country ?= {
									var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_2 }
								}
							}
							add = 40
						}
						else_if = {
							limit = {
								leader_country ?= {
									var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_1 }
								}
							}
							add = 20
						}
						multiply = 0.2
					}
				}
			}
		}
	}

	modifier = {
	}

	can_initiate_policy_votes = {
		is_leader_of_international_organization = scope:recipient
		tv_chief_scientist_available_for_law_change_trigger = yes
	}

	on_creation = {
		if = {
			limit = { exists = scope:actor }
			set_leader_country = scope:actor
		}
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_join_trigger = {
		always = no
	}

	can_leave_trigger = {
		always = no
	}

	auto_leave_trigger = {
		always = no
	}

	auto_disband_trigger = {
		total_members < 1
	}

	join_defensive_wars_always = {
		always = no
	}

	join_offensive_wars_always = {
		always = no
	}
}
