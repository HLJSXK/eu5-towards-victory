# Towards Victory — Governor's House (总督府) International Organization type definition
# Non-unique IO; each country pursuing Conquest Victory creates their own instance
# at Conquest Victory Milestone 1 (200 locations). Country membership flag: tv_govhouse_member.
# Authority is an IO type variable (tv_govhouse_authority, 0-100), decays 0.05%/month,
# grows by Grand General's mil skill contribution.
#
# Leader: Grand General character. Selected manually by leader_country via
# tv_appoint_conquest_general_leader / tv_remove_conquest_general_leader / tv_change_conquest_general_leader
# (generated from data/io_leaders.yaml). The country variable tv_conquest_general_char holds the character.
#
# Tab 1: Governor status per owned region
# Resolutions tab: Grand General management actions
# Custom tab: War Tent tactics and war plans
# Members tab: IO members (default display)

tv_govhouse = {
	has_target = no
	has_leader_country = yes
	leader_title_key = "TV_GOVHOUSE_GRAND_GENERAL"
	use_regnal_number = no
	leader_type = character
	leader = {
		leader_country ?= {
			var:tv_conquest_general_char ?= {
				if = {
					limit = {
						is_female = no
						is_artist = no
					}
					add_to_list = leaders
				}
			}
		}
	}
	leader_change_trigger_type = none
	leader_change_method = none
	disband_if_no_leader = no
	show_on_diplomatic_map = no

	# Governor's House laws use the vanilla IO Laws tab with policy votes.
	# Keep the separate IO parliament tab hidden, matching the Academy pattern.
	has_parliament = no

	variables = {
		tv_govhouse_authority = {
			format = "TV_GOVHOUSE_AUTHORITY_FORMAT"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				subtract = {
					desc = "TV_GOVHOUSE_AUTHORITY_DECAY"
					value = var:tv_govhouse_authority
					# TODO: tune decay rate (currently 0.05% of current value per month)
					multiply = 0.0005
				}
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country = {
							custom_tooltip = {
								text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
								has_variable = tv_conquest_general_char
							}
							var:tv_conquest_general_char ?= { is_alive = yes }
						}
					}
					add = {
						desc = "TV_GOVHOUSE_GENERAL_CONTRIBUTION"
						value = leader_country.var:tv_conquest_general_char.mil
						# TODO: tune contribution multiplier
						multiply = 0.05
					}
				}
			}
		}
		tv_govhouse_power_consolidation = {
			format = "TV_GOVHOUSE_POWER_BALANCE_FORMAT"
			change_format = "PERCENTAGE_CHANGE_FORMAT"
			min = 0
			max = 1
			start = 0
			monthly_change = {
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country ?= {
							has_ruler = yes
							custom_tooltip = {
								text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
								has_variable = tv_conquest_general_char
							}
							var:tv_conquest_general_char ?= { is_alive = yes }
						}
					}
					add = {
						desc = "TV_GOVHOUSE_POWER_CONSOLIDATION_IN_OFFICE"
						value = 1
						subtract = var:tv_govhouse_power_consolidation
						min = 0
						max = 0.004
					}
				}
				if = {
					limit = {
						NOT = {
							AND = {
								international_organization_has_leader = yes
								leader_country ?= {
									has_ruler = yes
									custom_tooltip = {
										text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
										has_variable = tv_conquest_general_char
									}
									var:tv_conquest_general_char ?= { is_alive = yes }
								}
							}
						}
					}
					subtract = {
						desc = "TV_GOVHOUSE_POWER_CONSOLIDATION_NO_GENERAL"
						value = var:tv_govhouse_power_consolidation
					}
				}
			}
		}
		tv_govhouse_power_equilibrium = {
			format = "TV_GOVHOUSE_POWER_BALANCE_FORMAT"
			change_format = "PERCENTAGE_CHANGE_FORMAT"
			min = 0
			max = 1
			start = 0
			monthly_change = {
				subtract = {
					value = var:tv_govhouse_power_equilibrium
				}
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country ?= {
							has_ruler = yes
							custom_tooltip = {
								text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
								has_variable = tv_conquest_general_char
							}
							var:tv_conquest_general_char ?= { is_alive = yes }
						}
					}
					subtract = {
						desc = "TV_GOVHOUSE_POWER_EQUILIBRIUM_BASE"
						value = 0.20
					}
					add = {
						desc = "TV_GOVHOUSE_POWER_EQUILIBRIUM_RELATIVE_ABILITIES"
						value = leader_country.var:tv_conquest_general_char.total_abilities
						divide = {
							value = leader_country.ruler_or_regent.total_abilities
							add = leader_country.var:tv_conquest_general_char.total_abilities
							min = 1
						}
						max = 1
						min = 0
					}
					add = {
						desc = "TV_GOVHOUSE_POWER_EQUILIBRIUM_CONSOLIDATION"
						value = var:tv_govhouse_power_consolidation
						min = 0
						max = {
							value = 1
							add = 0.10
							subtract = {
								value = leader_country.var:tv_conquest_general_char.total_abilities
								divide = {
									value = leader_country.ruler_or_regent.total_abilities
									add = leader_country.var:tv_conquest_general_char.total_abilities
									min = 1
								}
								max = 1
								min = 0
							}
							min = 0
						}
					}
					if = {
						limit = { leader_country ?= { has_variable = tv_gg_loyalty_value } }
						subtract = {
							desc = "TV_GOVHOUSE_POWER_EQUILIBRIUM_GENERAL_LOYALTY"
							value = leader_country.var:tv_gg_loyalty_value
							min = 0
							max = 1
						}
					}
				}
			}
		}
		tv_govhouse_power_balance = {
			format = "TV_GOVHOUSE_POWER_BALANCE_FORMAT"
			change_format = "PERCENTAGE_CHANGE_FORMAT"
			min = 0
			max = 1
			start = 0
			monthly_change = {
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country ?= {
							has_ruler = yes
							custom_tooltip = {
								text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
								has_variable = tv_conquest_general_char
							}
							var:tv_conquest_general_char ?= { is_alive = yes }
						}
						var:tv_govhouse_power_balance < var:tv_govhouse_power_equilibrium
					}
					add = {
						desc = "TV_GOVHOUSE_POWER_BALANCE_BELOW_EQUILIBRIUM"
						value = var:tv_govhouse_power_equilibrium
						subtract = var:tv_govhouse_power_balance
						min = 0
						max = 0.006
					}
				}
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country ?= {
							has_ruler = yes
							custom_tooltip = {
								text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
								has_variable = tv_conquest_general_char
							}
							var:tv_conquest_general_char ?= { is_alive = yes }
						}
						var:tv_govhouse_power_balance > var:tv_govhouse_power_equilibrium
					}
					subtract = {
						desc = "TV_GOVHOUSE_POWER_BALANCE_ABOVE_EQUILIBRIUM"
						value = var:tv_govhouse_power_balance
						subtract = var:tv_govhouse_power_equilibrium
						min = 0
						max = 0.006
					}
				}
				if = {
					limit = {
						NOT = {
							AND = {
								international_organization_has_leader = yes
								leader_country ?= {
									has_ruler = yes
									custom_tooltip = {
										text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
										has_variable = tv_conquest_general_char
									}
									var:tv_conquest_general_char ?= { is_alive = yes }
								}
							}
						}
					}
					subtract = {
						desc = "TV_GOVHOUSE_POWER_BALANCE_NO_GENERAL"
						value = var:tv_govhouse_power_balance
						min = 0
						max = 0.006
					}
				}
			}
		}
	}

	modifier = {
	}

	can_initiate_policy_votes = {
		is_leader_of_international_organization = scope:recipient
	}

	monthly_effect = {
		hidden_effect = {
			leader_country ?= {
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_GOVHOUSE_MEMBER_TT
							has_variable = tv_govhouse_member
						}
					}
					tv_apply_governor_bonuses_effect = yes
				}
			}
		}
	}

	on_creation = {
		if = {
			limit = { exists = scope:actor }
			set_leader_country = scope:actor
		}
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_join_trigger = {
		always = no
	}

	can_leave_trigger = {
		always = no
	}

	auto_leave_trigger = {
		always = no
	}

	auto_disband_trigger = {
		total_members < 1
	}

	join_defensive_wars_always = {
		always = no
	}

	join_offensive_wars_always = {
		always = no
	}
}
