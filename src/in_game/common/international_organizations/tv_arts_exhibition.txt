# Towards Victory — Arts Exhibition International Organization type definition
# Non-unique IO; each country pursuing Cultural Victory creates their own instance
# at Cultural Victory Milestone 1. Country membership flag: tv_arts_exhibition_member.
# International Influence is an IO type variable (tv_arts_intl_influence), decays 0.1% of current value/month.
#
# Leader: character (artist). Selected manually by the leader_country via the
# Appoint / Remove / Change Leader generic actions (tv_io_leader_actions.txt).
# The leader country variable tv_arts_exhibition_leader_char holds the appointed character.

tv_arts_exhibition = {
	has_target = no
	has_leader_country = yes
	leader_title_key = "TV_ARTS_EXHIBITION_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			var:tv_arts_exhibition_leader_char ?= {
				add_to_list = leaders
			}
		}
	}
	leader_change_trigger_type = none
	leader_change_method = none
	disband_if_no_leader = no
	show_on_diplomatic_map = no

	variables = {
		tv_arts_intl_influence = {
			format = "TV_ARTS_INTL_INFLUENCE_FORMAT"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				subtract = {
					desc = "TV_ARTS_INTL_INFLUENCE_DECAY"
					value = var:tv_arts_intl_influence
					multiply = 0.001
				}
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country = {
							custom_tooltip = {
								text = TV_HAS_ARTS_EXHIBITION_LEADER_CHAR_TT
								has_variable = tv_arts_exhibition_leader_char
							}
							var:tv_arts_exhibition_leader_char ?= { is_alive = yes }
						}
					}
					add = {
						desc = "TV_ARTS_LEADER_CONTRIBUTION"
						value = leader_country.var:tv_arts_exhibition_leader_char.artist_skill
						multiply = 0.1
					}
				}
			}
		}
		tv_local_exhibition_enthusiasm = {
			format = "TV_LOCAL_EXHIBITION_ENTHUSIASM_FORMAT"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 200
			start = 0
			monthly_change = {
				if = {
					limit = {
						leader_country ?= {
							custom_tooltip = {
								text = TV_HAS_LOCAL_EXHIBITION_ACTIVE_TT
								has_variable = tv_local_exhibition_active
							}
						}
					}
					subtract = {
						desc = "TV_LOCAL_EXHIBITION_ENTHUSIASM_DECAY"
						value = 8
					}
				}
			}
		}
	}

	modifier = {
	}

	on_creation = {
		if = {
			limit = { exists = scope:actor }
			set_leader_country = scope:actor
			scope:actor = {
				set_variable = { name = tv_dispatched_artists_count value = 0 }
			}
		}
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_join_trigger = {
		always = no
	}

	can_leave_trigger = {
		always = no
	}

	auto_leave_trigger = {
		always = no
	}

	auto_disband_trigger = {
		total_members < 1
	}

	join_defensive_wars_always = {
		always = no
	}

	join_offensive_wars_always = {
		always = no
	}
}
