# Towards Victory — Arts Exhibition International Organization type definition
# Countries join via script when they achieve Cultural Victory Milestone 1.
# Membership tracked per-country (tv_arts_exhibition_member).
# International Influence is an IO type variable (tv_arts_intl_influence), auto-decays -0.05/month.

tv_arts_exhibition = {
	has_target = no
	has_leader_country = yes
	leader_title_key = "TV_ARTS_EXHIBITION_LEADER"
	leader_type = country
	leader_change_trigger_type = none
	leader_change_method = none
	disband_if_no_leader = no
	show_on_diplomatic_map = no

	variables = {
		tv_arts_intl_influence = {
			format = "TV_ARTS_INTL_INFLUENCE_FORMAT"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 0
			monthly_change = {
				subtract = {
					desc = "TV_ARTS_INTL_INFLUENCE_DECAY"
					value = 0.05
				}
				if = {
					limit = {
						international_organization_has_leader = yes
						leader_country = { has_variable = tv_chief_artist_monthly_bonus }
					}
					add = {
						desc = "TV_CHIEF_ARTIST_CONTRIBUTION"
						value = leader_country.var:tv_chief_artist_monthly_bonus
					}
				}
			}
		}
	}

	modifier = {
	}

	on_creation = {
		if = {
			limit = { exists = scope:actor }
			set_leader_country = scope:actor
		}
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_join_trigger = {
		always = no
	}

	can_leave_trigger = {
		always = no
	}

	auto_leave_trigger = {
		always = no
	}

	auto_disband_trigger = {
		total_members < 1
	}

	join_defensive_wars_always = {
		always = no
	}

	join_offensive_wars_always = {
		always = no
	}
}
