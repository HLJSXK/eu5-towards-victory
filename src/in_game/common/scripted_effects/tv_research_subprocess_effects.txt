# Towards Victory — Research Mechanism Subprocess Effects
# All effects are country-scoped unless noted.
# IO variables (tv_research_c_progress, tv_research_d_progress) are reset by navigating
# to the Academy IO via every_international_organizations_member_of.
#
# Verification — Step 3, Reference: jurchen_confederation_events.txt:47-54
#   Quote: "every_international_organizations_member_of = { limit = {
#           international_organization_type = international_organization_type:jurchen_confederation }
#           leader_country ?= { save_scope_as = target_jc_leader } }"

# ── Internal helper: reset IO-scoped progress variables ───────────────────────
tv_research_reset_io_progress_effect = {
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_academy_of_sciences
		}
		set_variable = { name = tv_research_c_progress value = 0 }
		set_variable = { name = tv_research_d_progress value = 0 }
	}
}

# ── Academic Interest helper (country scope) ─────────────────────────────────
tv_change_research_interest_effect = {
	custom_description = {
		text = tv_change_research_interest_text
		value = $value$
		change_variable = { name = tv_research_interest add = $value$ }
		clamp_variable = { name = tv_research_interest min = 0 max = 5 }
	}
}

# ── Initialize research cycle ──────────────────────────────────────────────────
# Called when the player confirms a research target. Starts phase 1 (A/B/C running).
tv_research_init_cycle_effect = {
	set_variable = { name = tv_research_phase value = 1 }
	set_variable = { name = tv_research_interest value = 0 }
	set_variable = { name = tv_research_b_done value = 0 }
	set_variable = { name = tv_rm_subprocess_a value = 0 }
	set_variable = { name = tv_rm_subprocess_b value = 0 }
	set_variable = { name = tv_rm_subprocess_c value = 0 }
	set_variable = { name = tv_rm_subprocess_d_ready value = 0 }
	tv_research_reset_io_progress_effect = yes
}

# ── Reset full research cycle (after completion or abandon) ───────────────────
# Clears all progress variables. tv_research_target and tv_research_phase are removed.
# tv_research_selection_cd is NOT touched here; set it separately if needed.
tv_research_reset_cycle_effect = {
	set_variable = { name = tv_research_phase value = 0 }
	remove_variable = tv_research_target
	remove_variable = tv_research_proposed_advance
	remove_variable = tv_research_interest
	remove_variable = tv_research_b_done
	set_variable = { name = tv_rm_subprocess_a value = 0 }
	set_variable = { name = tv_rm_subprocess_b value = 0 }
	set_variable = { name = tv_rm_subprocess_c value = 0 }
	set_variable = { name = tv_rm_subprocess_d_ready value = 0 }
	tv_research_reset_io_progress_effect = yes
	hidden_effect = {
		remove_country_modifier = tv_concentrated_research_penalty
	}
}

# ── Abandon research cycle ─────────────────────────────────────────────────────
# Resets cycle and applies 12-month cooldown on target selection.
tv_research_abandon_effect = {
	tv_research_reset_cycle_effect = yes
	set_variable = { name = tv_research_selection_cd value = 12 }
}

# ── Sub-process A complete ────────────────────────────────────────────────────
tv_research_complete_a_effect = {
	set_variable = { name = tv_rm_subprocess_a value = 2 }
	tv_research_check_d_ready_effect = yes
}

# ── Sub-process B complete ────────────────────────────────────────────────────
tv_research_complete_b_effect = {
	set_variable = { name = tv_research_b_done value = 1 }
	set_variable = { name = tv_rm_subprocess_b value = 2 }
	tv_research_check_d_ready_effect = yes
}

# ── Sub-process C complete (called from IO monthly_effect) ────────────────────
tv_research_complete_c_effect = {
	set_variable = { name = tv_rm_subprocess_c value = 2 }
	tv_research_check_d_ready_effect = yes
}

# ── Check if D is now ready ────────────────────────────────────────────────────
# Called after each subprocess completes to potentially enable the D button.
tv_research_check_d_ready_effect = {
	if = {
		limit = {
			tv_research_can_start_d_trigger = yes
			NOT = { var:tv_rm_subprocess_d_ready ?= { this >= 1 } }
		}
		set_variable = { name = tv_rm_subprocess_d_ready value = 1 }
		trigger_event_non_silently = { id = tv_research.19 }
	}
}

# ── Start sub-process D (Concentrated Research) ───────────────────────────────
# Called from the confirmation event when player clicks the D button.
tv_research_start_d_effect = {
	set_variable = { name = tv_research_phase value = 2 }
	add_country_modifier = {
		modifier = tv_concentrated_research_penalty
		years = -1
		mode = replace
	}
}

# ── Concentrated Research complete ────────────────────────────────────────────
# Called from IO monthly_effect when tv_research_d_progress >= 100.
# Fires the notification event. The event option applies the selected advance unlock
# so the reward is visible to the player, then resets the cycle for the next round.
tv_research_concentrated_complete_effect = {
	# Stop D monthly completion checks while keeping tv_research_target for the event.
	set_variable = { name = tv_research_phase value = 0 }
	hidden_effect = {
		remove_country_modifier = tv_concentrated_research_penalty
	}
	trigger_event_non_silently = { id = tv_research.30 }
}

# ── Confirm selected target and start cycle ────────────────────────────────────
# Called from the target selection event when the player authorizes.
tv_research_confirm_target_effect = {
	set_variable = { name = tv_research_target value = var:tv_research_proposed_advance }
	remove_variable = tv_research_proposed_advance
	tv_research_init_cycle_effect = yes
}
