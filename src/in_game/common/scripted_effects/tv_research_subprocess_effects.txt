# Towards Victory — Research Mechanism Subprocess Effects
# All effects are country-scoped unless noted.
# IO variables (tv_research_c_progress, tv_research_d_progress) are reset by navigating
# to the Academy IO via every_international_organizations_member_of.
#
# Verification — Step 3, Reference: jurchen_confederation_events.txt:47-54
#   Quote: "every_international_organizations_member_of = { limit = {
#           international_organization_type = international_organization_type:jurchen_confederation }
#           leader_country ?= { save_scope_as = target_jc_leader } }"

# ── Internal helper: reset IO-scoped progress variables ───────────────────────
tv_research_reset_io_progress_effect = {
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_academy_of_sciences
		}
		set_variable = { name = tv_research_c_progress value = 0 }
		set_variable = { name = tv_research_d_progress value = 0 }
	}
}

# ── Academic Interest helper (country scope) ─────────────────────────────────
tv_change_research_interest_effect = {
	custom_description = {
		text = tv_change_research_interest_text
		value = $value$
		change_variable = { name = tv_research_interest add = $value$ }
		clamp_variable = { name = tv_research_interest min = 0 max = 5 }
	}
}

# ── Concentrated Research progress helper (country scope) ───────────────────
tv_change_research_d_progress_effect = {
	custom_description = {
		text = tv_change_research_d_progress_text
		value = $value$
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_academy_of_sciences
			}
			change_variable = { name = tv_research_d_progress add = $value$ }
			clamp_variable = { name = tv_research_d_progress min = 0 max = 100 }
			if = {
				limit = {
					var:tv_research_d_progress >= 100
					leader_country ?= {
						var:tv_research_phase ?= 2
						tv_chief_scientist_effective_adm_at_least_20_trigger = yes
					}
				}
				leader_country ?= { tv_research_concentrated_complete_effect = yes }
			}
		}
	}
}

# ── Concentrated Research health accident helper (country scope) ─────────────
tv_research_scientist_health_penalty_effect = {
	var:tv_academy_leader_char ?= {
		if = {
			limit = { has_trait = maimed }
			kill_character = this
		}
		else = {
			add_trait = trait:maimed
		}
	}
}

# ── Chief Scientist effective ADM cache (country scope) ─────────────────────
# Effective ADM is used by Academy work while the character is Chief Scientist.
# Great Scientist traits add +20 / +40 / +60 to the appointed character's ADM.
tv_update_chief_scientist_effective_adm_effect = {
	set_variable = { name = tv_scientist_effective_adm value = 0 }
	if = {
		limit = {
			has_variable = tv_academy_leader_char
			var:tv_academy_leader_char ?= { is_alive = yes }
		}
		set_variable = { name = tv_scientist_effective_adm value = var:tv_academy_leader_char.adm }
		if = {
			limit = { var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_3 } }
			change_variable = { name = tv_scientist_effective_adm add = 60 }
		}
		else_if = {
			limit = { var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_2 } }
			change_variable = { name = tv_scientist_effective_adm add = 40 }
		}
		else_if = {
			limit = { var:tv_academy_leader_char ?= { has_trait = tv_great_scientist_1 } }
			change_variable = { name = tv_scientist_effective_adm add = 20 }
		}
	}
}

# ── Great Scientist trait progression (country scope) ───────────────────────
# Completing concentrated research teaches the current Chief Scientist. Great
# Scientist levels are mutually exclusive; this also cleans up any stale lower
# level if a save/debug path ever leaves multiple levels on the same character.
tv_upgrade_great_scientist_trait_effect = {
	var:tv_academy_leader_char ?= {
		if = {
			limit = { has_trait = tv_great_scientist_3 }
			if = {
				limit = { has_trait = tv_great_scientist_2 }
				remove_trait = trait:tv_great_scientist_2
			}
			if = {
				limit = { has_trait = tv_great_scientist_1 }
				remove_trait = trait:tv_great_scientist_1
			}
		}
		else_if = {
			limit = { has_trait = tv_great_scientist_2 }
			if = {
				limit = { has_trait = tv_great_scientist_1 }
				remove_trait = trait:tv_great_scientist_1
			}
			remove_trait = trait:tv_great_scientist_2
			add_trait = trait:tv_great_scientist_3
		}
		else_if = {
			limit = { has_trait = tv_great_scientist_1 }
			remove_trait = trait:tv_great_scientist_1
			add_trait = trait:tv_great_scientist_2
		}
		else = {
			add_trait = trait:tv_great_scientist_1
		}
	}
}

# ── Initialize research cycle ──────────────────────────────────────────────────
# Called when the player confirms a research target. Starts phase 1 (A/B/C running).
tv_research_init_cycle_effect = {
	set_variable = { name = tv_research_phase value = 1 }
	set_variable = { name = tv_research_interest value = 0 }
	set_variable = { name = tv_research_b_done value = 0 }
	set_variable = { name = tv_rm_subprocess_a value = 0 }
	set_variable = { name = tv_rm_subprocess_b value = 0 }
	set_variable = { name = tv_rm_subprocess_c value = 0 }
	set_variable = { name = tv_rm_subprocess_d_ready value = 0 }
	tv_research_reset_io_progress_effect = yes
}

# ── Reset full research cycle (after completion or abandon) ───────────────────
# Clears all progress variables. tv_research_target and tv_research_phase are removed.
tv_research_reset_cycle_effect = {
	set_variable = { name = tv_research_phase value = 0 }
	remove_variable = tv_research_target
	remove_variable = tv_research_proposed_advance
	remove_variable = tv_research_interest
	remove_variable = tv_research_b_done
	set_variable = { name = tv_rm_subprocess_a value = 0 }
	set_variable = { name = tv_rm_subprocess_b value = 0 }
	set_variable = { name = tv_rm_subprocess_c value = 0 }
	set_variable = { name = tv_rm_subprocess_d_ready value = 0 }
	tv_research_reset_io_progress_effect = yes
	hidden_effect = {
		remove_country_modifier = tv_concentrated_research_penalty
	}
}

# ── Abandon research cycle ─────────────────────────────────────────────────────
# Resets cycle and costs 10 Prestige. No cooldown is applied.
tv_research_abandon_effect = {
	add_prestige = -10
	tv_research_reset_cycle_effect = yes
}

	tv_research_check_d_ready_effect = yes
}

# ── Check if D is now ready ────────────────────────────────────────────────────
# Called after each subprocess completes to potentially enable the D button.
tv_research_check_d_ready_effect = {
	if = {
		limit = {
			tv_research_can_start_d_trigger = yes
			NOT = { var:tv_rm_subprocess_d_ready ?= { this >= 1 } }
		}
		set_variable = { name = tv_rm_subprocess_d_ready value = 1 }
		trigger_event_non_silently = { id = tv_research.19 days = 1 }
	}
}

# ── Start sub-process D (Concentrated Research) ───────────────────────────────
# Called from the confirmation event when player clicks the D button.
tv_research_start_d_effect = {
	set_variable = { name = tv_research_phase value = 2 }
	add_country_modifier = {
		modifier = tv_concentrated_research_penalty
		years = -1
		mode = replace
	}
}

# ── Pause sub-process D (Concentrated Research) ──────────────────────────────
# Keeps accumulated D progress and target selection, but frees the Chief Scientist.
tv_research_pause_d_effect = {
	if = {
		limit = { tv_research_d_in_progress_trigger = yes }
		set_variable = { name = tv_research_phase value = 1 }
		hidden_effect = {
			remove_country_modifier = tv_concentrated_research_penalty
		}
	}
}

# ── Concentrated Research complete ────────────────────────────────────────────
# Called when tv_research_d_progress reaches 100.
# Fires the notification event. The event option applies the selected advance unlock
# and Great Scientist trait reward visibly, then resets the cycle for the next round.
tv_research_concentrated_complete_effect = {
	# Stop D monthly completion checks while keeping tv_research_target for the event.
	set_variable = { name = tv_research_phase value = 0 }
	hidden_effect = {
		remove_country_modifier = tv_concentrated_research_penalty
	}
	trigger_event_non_silently = { id = tv_research.30 days = 1 }
}

# ── Potential Projects display refresh ─────────────────────────────────────────
# Mirrors trigger-only eligibility state into GUI-readable variables. GUI
# expressions cannot evaluate scripted triggers (docs/knowledge/risk_cards/gui.md
# rule 2), so this refresh only runs from lifecycle hooks (join effect + monthly
# pulse) — never from a GUI/tooltip read path.
#
# The set of ages open to the Concentrated Research pool is always a contiguous
# prefix {1..K}: tv_research_proposal_includes_age_N_frontier_trigger is true
# whenever current_age_or_later(age N), which holds for every age <= the
# country's actual current age; the fallback-only clause for age N requires
# ages 1..N-1 to already be fully researched, which in turn satisfies their own
# fallback clauses too. So one integer ceiling captures the whole open set.
tv_research_refresh_open_age_ceiling_effect = {
	set_variable = { name = tv_research_open_age_ceiling value = 1 }
	if = {
		limit = { tv_research_proposal_includes_age_2_frontier_trigger = yes }
		set_variable = { name = tv_research_open_age_ceiling value = 2 }
	}
	if = {
		limit = { tv_research_proposal_includes_age_3_frontier_trigger = yes }
		set_variable = { name = tv_research_open_age_ceiling value = 3 }
	}
	if = {
		limit = { tv_research_proposal_includes_age_4_frontier_trigger = yes }
		set_variable = { name = tv_research_open_age_ceiling value = 4 }
	}
	if = {
		limit = { tv_research_proposal_includes_age_5_frontier_trigger = yes }
		set_variable = { name = tv_research_open_age_ceiling value = 5 }
	}
}
