# ── Capacity helper (country scope) ──────────────────────────────────────────
# Removes all touring penalty modifiers then re-applies the one matching the
# current tv_dispatched_artists_count value.  Call after any change to the counter.
tv_update_tour_capacity_effect = {
	remove_country_modifier = tv_touring_penalty_1
	remove_country_modifier = tv_touring_penalty_2
	remove_country_modifier = tv_touring_penalty_3
	remove_country_modifier = tv_touring_penalty_4
	remove_country_modifier = tv_touring_penalty_5
	remove_country_modifier = tv_touring_penalty_6
	remove_country_modifier = tv_touring_penalty_7
	remove_country_modifier = tv_touring_penalty_8
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 8
		}
		add_country_modifier = { modifier = tv_touring_penalty_8 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 7
		}
		add_country_modifier = { modifier = tv_touring_penalty_7 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 6
		}
		add_country_modifier = { modifier = tv_touring_penalty_6 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 5
		}
		add_country_modifier = { modifier = tv_touring_penalty_5 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 4
		}
		add_country_modifier = { modifier = tv_touring_penalty_4 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 3
		}
		add_country_modifier = { modifier = tv_touring_penalty_3 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 2
		}
		add_country_modifier = { modifier = tv_touring_penalty_2 days = -1 }
	}
	else_if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
				has_variable = tv_dispatched_artists_count
			}
			var:tv_dispatched_artists_count >= 1
		}
		add_country_modifier = { modifier = tv_touring_penalty_1 days = -1 }
	}
}

# ── Return / death cleanup (character scope) ──────────────────────────────────
# Called when a touring artist reaches their exhibition duration limit.
# Clears the target country marker and home country counter BEFORE removing
# character variables, then fires the return event and moves the artist home.
tv_end_artist_exhibition_effect = {
	save_scope_as = finished_artist
	# Preserve target country scope for return event option effects (before variables are cleared)
	var:tv_exhibition_target_country ?= {
		save_scope_as = exhibition_target
	}
	# Clear target-country marker before the variable is removed
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_EXHIBITION_TARGET_COUNTRY_TT
				has_variable = tv_exhibition_target_country
			}
		}
		var:tv_exhibition_target_country ?= {
			remove_variable = tv_has_visiting_artist
		}
	}
	var:tv_exhibition_home_country ?= {
		# 90% plain return; 2% each × 5 special return outcomes
		random_list = {
			90 = { trigger_event_non_silently = { id = tv_arts_exhibition.5 } }
			2 = {
				trigger = { exists = scope:exhibition_target  scope:exhibition_target = { any_work_of_art_in_country = { always = yes } } }
				trigger_event_non_silently = { id = tv_arts_exhibition.120 }
			}
			2 = { trigger_event_non_silently = { id = tv_arts_exhibition.121 } }
			2 = { trigger_event_non_silently = { id = tv_arts_exhibition.122 } }
			2 = { trigger_event_non_silently = { id = tv_arts_exhibition.123 } }
			2 = { trigger_event_non_silently = { id = tv_arts_exhibition.124 } }
		}
		scope:finished_artist = {
			move_country = prev
		}
		# Decrement counter and update capacity penalty
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
					has_variable = tv_dispatched_artists_count
				}
				var:tv_dispatched_artists_count >= 1
			}
			change_variable = { name = tv_dispatched_artists_count add = -1 }
		}
		tv_update_tour_capacity_effect = yes
		remove_list_variable = {
			name = tv_dispatched_artists_list
			target = scope:finished_artist
		}
	}
	remove_variable = tv_on_exhibition
	remove_variable = tv_exhibition_home_country
	remove_variable = tv_exhibition_target_country
	remove_variable = tv_exhibition_location
	remove_variable = tv_exhibition_elapsed
	remove_variable = tv_exhibition_duration
	remove_variable = tv_tour_event_1_fired
	remove_variable = tv_tour_event_2_fired
	remove_variable = tv_tour_event_3_fired
	remove_variable = tv_tour_event_4_fired
	remove_variable = tv_tour_event_5_fired
}

	}
	else_if = {
		limit = { var:tv_local_exhibition_duration >= 18 }

	}
}

# Maintains the active exhibition location list and count. The legacy
# tv_local_exhibition_location / secondary variables are folded in for saves
# made before the list model existed.
tv_local_exhibition_ensure_location_list_effect = {
	save_scope_as = local_exhibition_owner
	if = {
		limit = { NOT = { has_variable_list = tv_local_exhibition_locations } }
		clear_variable_list = tv_local_exhibition_locations
	}
	if = {
		limit = {
			NOT = { has_variable = tv_local_exhibition_primary_location }
			has_variable = tv_local_exhibition_location
		}
		var:tv_local_exhibition_location ?= {
			save_scope_as = local_exhibition_primary_location
		}
		if = {
			limit = { exists = scope:local_exhibition_primary_location }
			set_variable = { name = tv_local_exhibition_primary_location value = scope:local_exhibition_primary_location }
		}
	}
	var:tv_local_exhibition_primary_location ?= {
		save_scope_as = local_exhibition_primary_location
	}
	if = {
		limit = {
			exists = scope:local_exhibition_primary_location
			OR = {
				NOT = { has_variable_list = tv_local_exhibition_locations }
				NOT = {
					is_target_in_variable_list = {
						name = tv_local_exhibition_locations
						target = scope:local_exhibition_primary_location
					}
				}
			}
		}
		add_to_variable_list = {
			name = tv_local_exhibition_locations
			target = scope:local_exhibition_primary_location
		}
	}
	var:tv_local_exhibition_location ?= {
		save_scope_as = local_exhibition_legacy_location
	}
	if = {
		limit = {
			exists = scope:local_exhibition_legacy_location
			OR = {
				NOT = { has_variable_list = tv_local_exhibition_locations }
				NOT = {
					is_target_in_variable_list = {
						name = tv_local_exhibition_locations
						target = scope:local_exhibition_legacy_location
					}
				}
			}
		}
		add_to_variable_list = {
			name = tv_local_exhibition_locations
			target = scope:local_exhibition_legacy_location
		}
	}
	var:tv_local_exhibition_secondary_location ?= {
		save_scope_as = local_exhibition_legacy_secondary_location
	}
	if = {
		limit = {
			exists = scope:local_exhibition_legacy_secondary_location
			OR = {
				NOT = { has_variable_list = tv_local_exhibition_locations }
				NOT = {
					is_target_in_variable_list = {
						name = tv_local_exhibition_locations
						target = scope:local_exhibition_legacy_secondary_location
					}
				}
			}
		}
		add_to_variable_list = {
			name = tv_local_exhibition_locations
			target = scope:local_exhibition_legacy_secondary_location
		}
	}
	set_variable = { name = tv_local_exhibition_location_count value = 0 }
	every_in_list = {
		variable = tv_local_exhibition_locations
		scope:local_exhibition_owner = {
			change_variable = { name = tv_local_exhibition_location_count add = 1 }
		}
	}
}

tv_start_local_exhibition_effect = {
	if = {
		limit = { exists = scope:target }
		set_variable = { name = tv_local_exhibition_active value = 1 }
		set_variable = { name = tv_local_exhibition_location value = scope:target }
		set_variable = { name = tv_local_exhibition_primary_location value = scope:target }
		set_variable = { name = tv_local_exhibition_location_count value = 1 }
		set_variable = { name = tv_local_exhibition_duration value = 0 }
		set_variable = { name = tv_local_exhibition_rating value = 1 }
		set_variable = { name = tv_local_exhibition_artworks_committed value = 5 }
		set_variable = { name = tv_local_exhibition_add_art_threshold value = 5 }
		clear_variable_list = tv_local_exhibition_locations
		add_to_variable_list = { name = tv_local_exhibition_locations target = scope:target }
		remove_variable = tv_local_exhibition_secondary_location
		remove_variable = tv_local_exhibition_royal_visit_done
		remove_variable = tv_local_exhibition_expand_done
		remove_variable = tv_local_exhibition_foreign_invite_done
		remove_variable = tv_local_exhibition_treasure_done
		remove_variable = tv_local_exhibition_guest_artist
		remove_variable = tv_local_exhibition_guest_country
		remove_variable = tv_local_exhibition_last_duration
		remove_variable = tv_local_exhibition_last_rating
		remove_variable = tv_local_exhibition_last_reward
		remove_variable = tv_local_exhibition_last_location
		remove_variable = tv_local_exhibition_last_location_count
		clear_variable_list = tv_local_exhibition_last_locations
		scope:target = {
			construct_building = {
				building_type = building_type:tv_local_arts_exhibition
				cost_multiplier = 0
				cost_multiplier_reason = "game_concept_event"
				instant = yes
			}
		}
		tv_set_local_exhibition_enthusiasm_effect = { value = 80 }
		trigger_event_non_silently = { id = tv_arts_exhibition.1 }
	}
}

tv_add_local_exhibition_artworks_effect = {
	if = {
		limit = { exists = scope:target }
		tv_local_exhibition_ensure_location_list_effect = yes
		change_variable = { name = tv_local_exhibition_artworks_committed add = var:tv_local_exhibition_add_art_threshold }
		if = {
			limit = { var:tv_local_exhibition_add_art_threshold < 10 }
			set_variable = { name = tv_local_exhibition_add_art_threshold value = 10 }
		}
		else = {
			set_variable = { name = tv_local_exhibition_add_art_threshold value = 20 }
		}
		scope:target = {
			construct_building = {
				building_type = building_type:tv_local_arts_exhibition
				cost_multiplier = 0
				cost_multiplier_reason = "game_concept_event"
				instant = yes
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = tv_local_exhibition_locations }
					NOT = {
						is_target_in_variable_list = {
							name = tv_local_exhibition_locations
							target = scope:target
						}
					}
				}
			}
			add_to_variable_list = { name = tv_local_exhibition_locations target = scope:target }
			change_variable = { name = tv_local_exhibition_location_count add = 1 }
		}
		tv_change_local_exhibition_enthusiasm_effect = { value = 10 }
	}
}

tv_cleanup_local_exhibition_buildings_effect = {
	save_scope_as = local_exhibition_owner
	tv_local_exhibition_ensure_location_list_effect = yes
	every_in_list = {
		variable = tv_local_exhibition_locations
		if = {
			limit = { has_building = building_type:tv_local_arts_exhibition }
			destroy_building_forcefully = "building(building_type:tv_local_arts_exhibition|scope:local_exhibition_owner)"
		}
	}
}

tv_snapshot_local_exhibition_locations_effect = {
	save_scope_as = local_exhibition_owner

			add_to_variable_list = {
				name = tv_local_exhibition_last_locations
				target = scope:local_exhibition_snapshot_location
			}
		}
	}
}

tv_return_local_exhibition_guest_artist_effect = {
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_LOCAL_EXHIBITION_GUEST_ARTIST_TT
				has_variable = tv_local_exhibition_guest_artist
			}
			custom_tooltip = {
				text = TV_HAS_LOCAL_EXHIBITION_GUEST_COUNTRY_TT
				has_variable = tv_local_exhibition_guest_country
			}
		}
		var:tv_local_exhibition_guest_artist ?= {
			save_scope_as = local_exhibition_guest_artist
		}
		var:tv_local_exhibition_guest_country ?= {
			save_scope_as = local_exhibition_guest_country
		}
		if = {
			limit = {
				exists = scope:local_exhibition_guest_artist
				scope:local_exhibition_guest_artist = { is_alive = yes }
				exists = scope:local_exhibition_guest_country
			}
			if = {
				limit = {
					var:tv_arts_exhibition_leader_char ?= {
						this = scope:local_exhibition_guest_artist
					}
				}

					remove_character_modifier = tv_chief_artist_role_modifier

				remove_variable = tv_arts_exhibition_leader_char

			scope:local_exhibition_guest_artist = {
				move_country = scope:local_exhibition_guest_country
				remove_variable = tv_local_exhibition_guest_home
				remove_variable = tv_local_exhibition_guest_host
			}
			scope:local_exhibition_guest_country = {
				trigger_event_non_silently = { id = tv_arts_exhibition.363 }
			}
		}
	}
}

tv_end_local_exhibition_effect = {
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_LOCAL_EXHIBITION_ACTIVE_TT
				has_variable = tv_local_exhibition_active
			}
		}
		tv_update_local_exhibition_rating_effect = yes
		tv_snapshot_local_exhibition_locations_effect = yes
		set_variable = { name = tv_local_exhibition_last_duration value = var:tv_local_exhibition_duration }
		set_variable = { name = tv_local_exhibition_last_rating value = var:tv_local_exhibition_rating }
		set_variable = { name = tv_local_exhibition_last_reward value = var:tv_local_exhibition_duration }
		change_variable = { name = tv_local_exhibition_last_reward multiply = 0.5 }
		set_variable = { name = tv_local_exhibition_last_location_count value = var:tv_local_exhibition_location_count }
		if = {
			limit = { var:tv_local_exhibition_last_location_count < 1 }
			set_variable = { name = tv_local_exhibition_last_location_count value = 1 }
		}
		var:tv_local_exhibition_location ?= {
			save_scope_as = local_exhibition_reward_location
		}
		if = {
			limit = { exists = scope:local_exhibition_reward_location }
			set_variable = { name = tv_local_exhibition_last_location value = scope:local_exhibition_reward_location }
		}
		if = {
			limit = { var:tv_local_exhibition_rating >= 4 }
			change_variable = { name = tv_local_exhibition_last_reward add = 6 }
		}
		else_if = {
			limit = { var:tv_local_exhibition_rating >= 3 }
			change_variable = { name = tv_local_exhibition_last_reward add = 3 }
		}
		else_if = {
			limit = { var:tv_local_exhibition_rating >= 2 }
			change_variable = { name = tv_local_exhibition_last_reward add = 0 }
		}
		else = {
			change_variable = { name = tv_local_exhibition_last_reward add = -3 }
		}
		change_variable = { name = tv_local_exhibition_last_reward multiply = var:tv_local_exhibition_last_location_count }
		tv_cleanup_local_exhibition_buildings_effect = yes
		tv_return_local_exhibition_guest_artist_effect = yes
		set_variable = { name = tv_local_exhibition_cd value = 12 }
		tv_set_local_exhibition_enthusiasm_effect = { value = 0 }
		if = {
			limit = { var:tv_local_exhibition_last_rating >= 4 }
			trigger_event_non_silently = { id = tv_arts_exhibition.313 }
		}
		else_if = {
			limit = { var:tv_local_exhibition_last_rating >= 3 }
			trigger_event_non_silently = { id = tv_arts_exhibition.312 }
		}
		else_if = {
			limit = { var:tv_local_exhibition_last_rating >= 2 }
			trigger_event_non_silently = { id = tv_arts_exhibition.311 }
		}
		else = {
			trigger_event_non_silently = { id = tv_arts_exhibition.310 }
		}
		remove_variable = tv_local_exhibition_active
		remove_variable = tv_local_exhibition_location
		remove_variable = tv_local_exhibition_primary_location
		remove_variable = tv_local_exhibition_secondary_location
		remove_variable = tv_local_exhibition_location_count
		remove_variable = tv_local_exhibition_duration
		remove_variable = tv_local_exhibition_rating
		remove_variable = tv_local_exhibition_artworks_committed
		remove_variable = tv_local_exhibition_add_art_threshold
		remove_variable = tv_local_exhibition_royal_visit_done
		remove_variable = tv_local_exhibition_expand_done
		remove_variable = tv_local_exhibition_foreign_invite_done
		remove_variable = tv_local_exhibition_treasure_done
		remove_variable = tv_local_exhibition_guest_artist
		remove_variable = tv_local_exhibition_guest_country
		clear_variable_list = tv_local_exhibition_locations
	}
}

			limit = { var:tv_arts_exchange_months_remaining <= 0 }

		}
