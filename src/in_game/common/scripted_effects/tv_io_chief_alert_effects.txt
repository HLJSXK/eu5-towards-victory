tv_sync_missing_io_chief_alert_bridge_effect = {
	if = {
		limit = { has_variable = tv_pending_io_chief_alert_io }
		set_variable = { name = tv_io_chief_alert_open_requested value = scope:tv_pending_io_chief_alert_io }
		remove_variable = tv_pending_io_chief_alert_io
	}
}

		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_arts_exhibition
				leader_country ?= scope:tv_io_chief_alert_country
