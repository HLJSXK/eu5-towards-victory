tv_handle_io_chief_alert_callback_effect = {
	remove_variable = tv_io_chief_alert_open_requested

	if = {
		limit = { var:cmf_callback = flag:tv_missing_arts_exhibition_chief_alert }
		set_variable = { name = tv_io_chief_alert_open_requested value = 1 }
	}
	else_if = {
		limit = { var:cmf_callback = flag:tv_missing_diplomatic_alliance_chief_alert }
		set_variable = { name = tv_io_chief_alert_open_requested value = 2 }
	}
	else_if = {
		limit = { var:cmf_callback = flag:tv_missing_academy_chief_alert }
		set_variable = { name = tv_io_chief_alert_open_requested value = 3 }
	}
	else_if = {
		limit = { var:cmf_callback = flag:tv_missing_engineering_department_chief_alert }
		set_variable = { name = tv_io_chief_alert_open_requested value = 4 }
	}
	else_if = {
		limit = { var:cmf_callback = flag:tv_missing_trade_league_chief_alert }
		set_variable = { name = tv_io_chief_alert_open_requested value = 5 }
	}
	else_if = {
		limit = { var:cmf_callback = flag:tv_missing_govhouse_chief_alert }
		set_variable = { name = tv_io_chief_alert_open_requested value = 6 }
	}
}
