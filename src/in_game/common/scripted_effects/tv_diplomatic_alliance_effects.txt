tv_create_diplomatic_alliance_effect = {
	save_scope_as = tv_diplomatic_alliance_founder
	if = {
		limit = {
			is_subject = no
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_diplomatic_alliance
				}
			}
		}
		custom_tooltip = { text = tv_create_diplomatic_alliance_tt }
		hidden_effect = {
			set_variable = { name = tv_diplomatic_alliance_self_establishing value = 1 }
			create_international_organization = {
				type = international_organization_type:tv_diplomatic_alliance
				creator = prev
				hidden_effect = {
					add_country_to_international_organization = prev
					set_leader_country = prev
					international_organization_add_special_status = {
						type = special_status:tv_alliance_advocator
						country = prev
					}
				}
				save_scope_as = tv_diplomatic_alliance_new_io
			}
			remove_variable = tv_diplomatic_alliance_self_establishing
			set_variable = { name = tv_alliance_tier value = 0 }
			if = {
				limit = { exists = scope:tv_diplomatic_alliance_new_io }
				scope:tv_diplomatic_alliance_new_io = {
					tv_diplomatic_alliance_sync_advocator_status_effect = yes
					tv_diplomatic_alliance_update_member_count_effect = yes
				}
			}
			clear_saved_scope = tv_diplomatic_alliance_new_io
		}
	}
	if = {
		limit = {
			is_subject = no
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:tv_diplomatic_alliance_founder
			}
		}
		tv_diplomatic_alliance_sync_subject_chain_to_owned_alliance_effect = yes
	}
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_diplomatic_victory_points }
			}
		}
		set_variable = { name = tv_diplomatic_victory_points value = 0 }
	}
}

tv_diplomatic_alliance_sync_advocator_status_effect = {
	if = {
		limit = { exists = leader_country }
		leader_country = { save_scope_as = tv_diplomatic_alliance_advocator_country }
		international_organization_remove_special_status = {
			type = special_status:tv_alliance_member_status
			country = scope:tv_diplomatic_alliance_advocator_country
		}
		international_organization_add_special_status = {
			type = special_status:tv_alliance_advocator
			country = scope:tv_diplomatic_alliance_advocator_country
		}
	}
	clear_saved_scope = tv_diplomatic_alliance_advocator_country
}

tv_diplomatic_alliance_update_member_count_effect = {
	leader_country ?= {
		set_variable = { name = tv_diplomatic_alliance_member_count value = prev.total_members }
	}
}

tv_diplomatic_alliance_sync_subject_chain_to_owned_alliance_effect = {
	save_scope_as = tv_diplomatic_alliance_subject_chain_owner
	if = {
		limit = { is_subject = no }
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:tv_diplomatic_alliance_subject_chain_owner
			}
			tv_diplomatic_alliance_sync_advocator_status_effect = yes
			save_scope_as = tv_diplomatic_alliance_subject_chain_io
			scope:tv_diplomatic_alliance_subject_chain_owner = {
				every_subject_or_below = {
					save_scope_as = tv_diplomatic_alliance_subject_chain_country
					scope:tv_diplomatic_alliance_subject_chain_io = {
						if = {
							limit = {
								scope:tv_diplomatic_alliance_subject_chain_country = {
									NOT = { is_member_of_international_organization = scope:tv_diplomatic_alliance_subject_chain_io }
								}
							}
							add_country_to_international_organization = scope:tv_diplomatic_alliance_subject_chain_country
						}
						international_organization_remove_special_status = {
							type = special_status:tv_alliance_advocator
							country = scope:tv_diplomatic_alliance_subject_chain_country
						}
						international_organization_add_special_status = {
							type = special_status:tv_alliance_member_status
							country = scope:tv_diplomatic_alliance_subject_chain_country
						}
					}
				}
			}
			tv_diplomatic_alliance_update_member_count_effect = yes
		}
	}
	clear_saved_scope = tv_diplomatic_alliance_subject_chain_io
	clear_saved_scope = tv_diplomatic_alliance_subject_chain_country
}

tv_diplomatic_alliance_destroy_owned_alliance_effect = {
	save_scope_as = tv_diplomatic_alliance_destroy_owner
	clear_variable_list = tv_diplomatic_alliance_destroy_ios
	remove_variable = tv_diplomatic_alliance_destroy_found
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_diplomatic_alliance
			leader_country ?= scope:tv_diplomatic_alliance_destroy_owner
		}
		save_scope_as = tv_diplomatic_alliance_destroy_io
		scope:tv_diplomatic_alliance_destroy_owner = {
			set_variable = { name = tv_diplomatic_alliance_destroy_found value = 1 }
			set_variable = { name = tv_diplomatic_alliance_rebuild_on_freedom value = 1 }
			add_to_variable_list = { name = tv_diplomatic_alliance_destroy_ios target = scope:tv_diplomatic_alliance_destroy_io }
		}
	}

	if = {
		limit = { has_variable = tv_diplomatic_alliance_destroy_found }
		if = {
			limit = { has_variable = tv_diplomatic_alliance_leader_char }
			var:tv_diplomatic_alliance_leader_char ?= {
				remove_variable = tv_io_chief_role
			}
		}
		remove_variable = tv_diplomatic_alliance_leader_char
		remove_variable = tv_diplomatic_alliance_member_count
		remove_variable = tv_alliance_strongest_member
		remove_variable = tv_alliance_tier
		remove_variable = tv_diplomatic_alliance_member

		every_in_list = {
			variable = tv_diplomatic_alliance_destroy_ios
			save_scope_as = tv_diplomatic_alliance_destroy_io
			scope:tv_diplomatic_alliance_destroy_owner = {
				destroy_international_organization = { target = scope:tv_diplomatic_alliance_destroy_io }
			}
		}
	}
	remove_variable = tv_diplomatic_alliance_destroy_found
	clear_variable_list = tv_diplomatic_alliance_destroy_ios
	clear_saved_scope = tv_diplomatic_alliance_destroy_io
}

tv_diplomatic_alliance_stamp_creation_order_effect = {
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_alliance_creation_order }
			}
		}
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_VARIABLE_NOT_SET_TT

				}
			}
			set_global_variable = { name = tv_alliance_creation_order_counter value = 0 }

		set_local_variable = { name = tv_alliance_next_creation_order value = global_var:tv_alliance_creation_order_counter }

		set_variable = { name = tv_alliance_creation_order value = local_var:tv_alliance_next_creation_order }
		set_global_variable = { name = tv_alliance_creation_order_counter value = local_var:tv_alliance_next_creation_order }
		remove_local_variable = tv_alliance_next_creation_order
	}
}

tv_diplomatic_alliance_reinitialize_membership_effect = {
	save_scope_as = tv_diplomatic_alliance_reinit_country
	if = {
		limit = { is_subject = yes }
		tv_diplomatic_alliance_destroy_owned_alliance_effect = yes
	}

	clear_saved_scope = tv_diplomatic_alliance_keep_io
	clear_saved_scope = tv_diplomatic_alliance_cleanup_io
	clear_saved_scope = tv_diplomatic_alliance_disband_io
	clear_variable_list = tv_diplomatic_alliance_reinit_disband_ios
	clear_variable_list = tv_diplomatic_alliance_reinit_leave_ios

	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_diplomatic_alliance
		}
		tv_diplomatic_alliance_stamp_creation_order_effect = yes
	}

	# Prefer the earliest Diplomatic Alliance led by the current country.
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_diplomatic_alliance
			leader_country ?= scope:tv_diplomatic_alliance_reinit_country
		}
		if = {
			limit = { NOT = { exists = scope:tv_diplomatic_alliance_keep_io } }
			save_scope_as = tv_diplomatic_alliance_keep_io
		}
		else_if = {
			limit = {
				exists = scope:tv_diplomatic_alliance_keep_io
				var:tv_alliance_creation_order < scope:tv_diplomatic_alliance_keep_io.var:tv_alliance_creation_order
			}
			save_scope_as = tv_diplomatic_alliance_keep_io
		}
	}

	# If none are self-led, keep the earliest alliance we are already in.
	if = {
		limit = { NOT = { exists = scope:tv_diplomatic_alliance_keep_io } }
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
			}
			if = {
				limit = { NOT = { exists = scope:tv_diplomatic_alliance_keep_io } }
				save_scope_as = tv_diplomatic_alliance_keep_io
			}
			else_if = {
				limit = {
					exists = scope:tv_diplomatic_alliance_keep_io
					var:tv_alliance_creation_order < scope:tv_diplomatic_alliance_keep_io.var:tv_alliance_creation_order
				}
				save_scope_as = tv_diplomatic_alliance_keep_io
			}
		}
	}

	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_diplomatic_alliance
			exists = scope:tv_diplomatic_alliance_keep_io
			NOT = { this = scope:tv_diplomatic_alliance_keep_io }
		}
		save_scope_as = tv_diplomatic_alliance_cleanup_io
		if = {
			limit = { leader_country ?= scope:tv_diplomatic_alliance_reinit_country }
			scope:tv_diplomatic_alliance_reinit_country = {
				add_to_variable_list = { name = tv_diplomatic_alliance_reinit_disband_ios target = scope:tv_diplomatic_alliance_cleanup_io }
			}
		}
		else = {
			scope:tv_diplomatic_alliance_reinit_country = {
				add_to_variable_list = { name = tv_diplomatic_alliance_reinit_leave_ios target = scope:tv_diplomatic_alliance_cleanup_io }
			}
		}
	}

	clear_saved_scope = tv_diplomatic_alliance_cleanup_io
	clear_saved_scope = tv_diplomatic_alliance_disband_io

	every_in_list = {
		variable = tv_diplomatic_alliance_reinit_leave_ios
		remove_country_from_international_organization = scope:tv_diplomatic_alliance_reinit_country
	}
	every_in_list = {
		variable = tv_diplomatic_alliance_reinit_disband_ios
		save_scope_as = tv_diplomatic_alliance_disband_io
		scope:tv_diplomatic_alliance_reinit_country = {
			destroy_international_organization = { target = scope:tv_diplomatic_alliance_disband_io }
		}
	}
	clear_variable_list = tv_diplomatic_alliance_reinit_disband_ios
	clear_variable_list = tv_diplomatic_alliance_reinit_leave_ios
	clear_saved_scope = tv_diplomatic_alliance_disband_io

	if = {
		limit = { exists = scope:tv_diplomatic_alliance_keep_io }
		set_variable = { name = tv_diplomatic_alliance_member value = 1 }
		scope:tv_diplomatic_alliance_keep_io = {
			tv_diplomatic_alliance_update_member_count_effect = yes
		}
	}
	clear_saved_scope = tv_diplomatic_alliance_keep_io
}

tv_diplomatic_alliance_update_strongest_member_effect = {
	hidden_effect = {
		if = {
			limit = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				international_organization_has_leader = yes
			}
			clear_saved_scope = tv_alliance_strongest_member_candidate
			ordered_international_organization_member = {
				order_by = {
					add = {
						value = military_strength
					}
				}
				max = 1
				check_range_bounds = no
				save_scope_as = tv_alliance_strongest_member_candidate
			}
			leader_country ?= {
				if = {
					limit = { exists = scope:tv_alliance_strongest_member_candidate }
					set_variable = { name = tv_alliance_strongest_member value = scope:tv_alliance_strongest_member_candidate }
				}
				else = {
					remove_variable = tv_alliance_strongest_member
				}
			}
			clear_saved_scope = tv_alliance_strongest_member_candidate
		}
	}
}

tv_diplomatic_alliance_monthly_victory_points_effect = {
	hidden_effect = {
		if = {
			limit = { international_organization_type = international_organization_type:tv_diplomatic_alliance }
			leader_country ?= {
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_VARIABLE_NOT_SET_TT
							NOT = { has_variable = tv_diplomatic_victory_points }
						}
					}
					set_variable = { name = tv_diplomatic_victory_points value = 0 }
				}
			}
			if = {
				limit = {
					var:tv_alliance_tier > 0
					international_organization_has_leader = yes
				}
				set_local_variable = { name = tv_alliance_monthly_dvp value = var:tv_alliance_tier }
				leader_country ?= {
					change_variable = { name = tv_diplomatic_victory_points add = local_var:tv_alliance_monthly_dvp }
				}
				set_local_variable = { name = tv_alliance_member_monthly_dvp value = var:tv_alliance_tier }
				change_local_variable = { name = tv_alliance_member_monthly_dvp divide = 2 }
				every_international_organization_member = {
					limit = { NOT = { is_leader_of_international_organization = prev } }
					if = {
						limit = { NOT = { has_variable = tv_diplomatic_victory_points } }
						set_variable = { name = tv_diplomatic_victory_points value = 0 }
					}
					change_variable = { name = tv_diplomatic_victory_points add = local_var:tv_alliance_member_monthly_dvp }
				}
				remove_local_variable = tv_alliance_monthly_dvp
				remove_local_variable = tv_alliance_member_monthly_dvp
			}
		}
	}
}
