# Towards Victory — Governor's House (总督府) scripted effects

# Scope: Governor's House IO. Grants the leader country the Governor's House
# "Supreme Overlord" status.
tv_govhouse_sync_supreme_overlord_status_effect = {
	if = {
		limit = { exists = leader_country }
		leader_country = { save_scope_as = tv_govhouse_supreme_overlord_country }
		international_organization_remove_special_status = {
			type = special_status:tv_govhouse_subject_status
			country = scope:tv_govhouse_supreme_overlord_country
		}
		international_organization_add_special_status = {
			type = special_status:tv_govhouse_supreme_overlord_status
			country = scope:tv_govhouse_supreme_overlord_country
		}
	}
	clear_saved_scope = tv_govhouse_supreme_overlord_country
}

# Creates the Governor's House IO for scope country (must be called as owner country).
# Sets the tv_govhouse_member owner flag, creates the IO, and adds the subject chain.
tv_govhouse_create_effect = {
	save_scope_as = tv_govhouse_founder
	if = {
		limit = {
			is_subject = no
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_govhouse_founder
				}
			}
		}
		clear_saved_scope = tv_govhouse_new_io
		set_variable = { name = tv_govhouse_member value = 1 }
		create_international_organization = {
			type = international_organization_type:tv_govhouse
			add_country_to_international_organization = prev
			set_leader_country = prev
			save_scope_as = tv_govhouse_new_io
		}
		if = {
			limit = { exists = scope:tv_govhouse_new_io }
			scope:tv_govhouse_new_io = {
				tv_govhouse_sync_supreme_overlord_status_effect = yes
			}
		}
		clear_saved_scope = tv_govhouse_new_io
	}
	if = {
		limit = {
			is_subject = no
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_govhouse
				leader_country ?= scope:tv_govhouse_founder
			}
		}
		set_variable = { name = tv_govhouse_member value = 1 }
		tv_govhouse_refresh_region_list_effect = yes
	}
	tv_govhouse_sync_subject_chain_to_owned_house_effect = yes
}

# Adds every subject below the current independent owner to its own Governor's House.
# Scope: Governor's House owner country.
tv_govhouse_sync_subject_chain_to_owned_house_effect = {
	save_scope_as = tv_govhouse_subject_chain_owner
	if = {
		limit = { is_subject = no }
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_govhouse
				leader_country ?= scope:tv_govhouse_subject_chain_owner
			}
			tv_govhouse_sync_supreme_overlord_status_effect = yes
			save_scope_as = tv_govhouse_subject_chain_io
			scope:tv_govhouse_subject_chain_owner = {
				every_subject_or_below = {
					save_scope_as = tv_govhouse_subject_chain_country
					scope:tv_govhouse_subject_chain_io = {
						if = {
							limit = {
								scope:tv_govhouse_subject_chain_country = {
									NOT = { is_member_of_international_organization = scope:tv_govhouse_subject_chain_io }
								}
							}
							add_country_to_international_organization = scope:tv_govhouse_subject_chain_country
						}
						international_organization_remove_special_status = {
							type = special_status:tv_govhouse_supreme_overlord_status
							country = scope:tv_govhouse_subject_chain_country
						}
						international_organization_add_special_status = {
							type = special_status:tv_govhouse_subject_status
							country = scope:tv_govhouse_subject_chain_country
						}
					}
				}
			}
		}
	}
	clear_saved_scope = tv_govhouse_subject_chain_io
	clear_saved_scope = tv_govhouse_subject_chain_country
}

# Removes scope:tv_govhouse_subject_root and all subjects below it from the current
# owner's Governor's House. Scope: former top overlord / Governor's House owner.
tv_govhouse_remove_subject_chain_from_house_effect = {
	save_scope_as = tv_govhouse_subject_chain_owner
	if = {
		limit = { exists = scope:tv_govhouse_subject_root }
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_govhouse
				leader_country ?= scope:tv_govhouse_subject_chain_owner
			}
			save_scope_as = tv_govhouse_subject_chain_io
			if = {
				limit = {
					scope:tv_govhouse_subject_root = {
						is_member_of_international_organization = scope:tv_govhouse_subject_chain_io
					}
				}
				international_organization_remove_special_status = {
					type = special_status:tv_govhouse_supreme_overlord_status
					country = scope:tv_govhouse_subject_root
				}
				international_organization_remove_special_status = {
					type = special_status:tv_govhouse_subject_status
					country = scope:tv_govhouse_subject_root
				}
				remove_country_from_international_organization = scope:tv_govhouse_subject_root
			}
			scope:tv_govhouse_subject_root = {
				every_subject_or_below = {
					limit = { is_member_of_international_organization = scope:tv_govhouse_subject_chain_io }
					save_scope_as = tv_govhouse_subject_chain_country
					scope:tv_govhouse_subject_chain_io = {
						international_organization_remove_special_status = {
							type = special_status:tv_govhouse_supreme_overlord_status
							country = scope:tv_govhouse_subject_chain_country
						}
						international_organization_remove_special_status = {
							type = special_status:tv_govhouse_subject_status
							country = scope:tv_govhouse_subject_chain_country
						}
						remove_country_from_international_organization = scope:tv_govhouse_subject_chain_country
					}
				}
			}
		}
	}
	clear_saved_scope = tv_govhouse_subject_chain_io
	clear_saved_scope = tv_govhouse_subject_chain_country
}

# Rechecks Governor's House creation after a former subject becomes independent.
# Scope: freed country.
tv_govhouse_check_rebuild_after_freedom_effect = {
	save_scope_as = tv_govhouse_freed_country
	tv_update_conquest_score_effect = yes
	if = {
		limit = {
			tv_conquest_milestone_1 = yes
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_govhouse_freed_country
				}
			}
		}
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_CONQUEST_MILESTONE_TT
					has_variable = tv_conquest_milestone
				}
			}
			if = {
				limit = { var:tv_conquest_milestone >= 1 }
				if = {
					limit = { is_ai = no }
					trigger_event_non_silently = { id = tv_govhouse.4 }
				}
				else = {
					tv_govhouse_create_effect = yes
				}
			}
			else = {
				tv_check_conquest_milestones_effect = yes
			}
		}
		else = {
			tv_check_conquest_milestones_effect = yes
		}
	}
}

tv_govhouse_sync_grand_general_loyalty_effect = {
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
				has_variable = tv_conquest_general_char
			}
			var:tv_conquest_general_char ?= {
				is_alive = yes
				has_variable = tv_gg_loyalty
			}
		}
		var:tv_conquest_general_char ?= {
			save_temporary_scope_as = tv_govhouse_loyalty_general
		}

	}

tv_govhouse_initialize_grand_general_loyalty_effect = {
	save_scope_as = tv_govhouse_loyalty_owner
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
				has_variable = tv_conquest_general_char
			}
			var:tv_conquest_general_char ?= {
				is_alive = yes
				NOT = { has_variable = tv_gg_loyalty }
			}
		}
		var:tv_conquest_general_char ?= {
			if = {
				limit = { has_estate = estate_type:nobles_estate }
				set_variable = { name = tv_gg_loyalty value = 0 }
			}
			else_if = {
				limit = {
					OR = {
						has_estate = estate_type:burghers_estate
						has_estate = estate_type:clergy_estate
					}
				}
				set_variable = { name = tv_gg_loyalty value = 0.10 }
			}
			else_if = {
				limit = { has_estate = estate_type:crown_estate }
				set_variable = { name = tv_gg_loyalty value = 0.30 }
			}
			else = {
				set_variable = { name = tv_gg_loyalty value = 0.20 }
			}
			if = {
				limit = {
					scope:tv_govhouse_loyalty_owner = {
						any_international_organizations_member_of = {
							international_organization_type = international_organization_type:tv_govhouse
							leader_country ?= scope:tv_govhouse_loyalty_owner
							international_organization_has_policy = policy:tv_govhouse_general_marching_orders_policy
						}
					}
				}
				change_variable = { name = tv_gg_loyalty add = 0.10 }
			}
			else_if = {
				limit = {
					scope:tv_govhouse_loyalty_owner = {
						any_international_organizations_member_of = {
							international_organization_type = international_organization_type:tv_govhouse
							leader_country ?= scope:tv_govhouse_loyalty_owner
							international_organization_has_policy = policy:tv_govhouse_general_supreme_command_policy
						}
					}
				}
				change_variable = { name = tv_gg_loyalty add = -0.10 }
			}
			clamp_variable = { name = tv_gg_loyalty min = 0 max = 1 }
		}
	}
	tv_govhouse_sync_grand_general_loyalty_effect = yes
}

# Starts Grand General training. Marks the Grand General as in training,
# starts the month counter, and applies the temporary court cost modifier.
# Called by the tv_train_grand_general generic action.
# Scope: actor country.
tv_training_start_effect = {
	var:tv_conquest_general_char ?= {
		set_variable = { name = tv_gg_training_active value = 1 }
		set_variable = { name = tv_gg_training_months value = 0 }
	}
	set_variable = { name = tv_gg_training_progress value = 0 }
	add_country_modifier = {
		modifier = tv_grand_general_training_modifier
		years = 1
		mode = replace
	}
}

# Clears Grand General training state from the current stored Grand General.
# Scope: owner country.
tv_training_clear_effect = {
	var:tv_conquest_general_char ?= {
		remove_variable = tv_gg_training_active
		remove_variable = tv_gg_training_months
	}
	remove_variable = tv_gg_training_progress
	remove_country_modifier = tv_grand_general_training_modifier
}

# Fires the weighted training outcome for the Grand General.
# Called once tv_gg_training_active has accumulated 12 monthly ticks.
# Scope: actor country. 49 / 49 / 1 / 1 split.
# Routing is resolved here so each outcome event is directly meaningful.
tv_training_resolve_effect = {
	if = {
		limit = {
			var:tv_conquest_general_char ?= {
				is_alive = yes
				has_variable = tv_gg_training_active
			}
		}
		random_list = {
			49 = {
				if = {
					limit = { var:tv_conquest_general_char ?= { mil >= 100 } }
					trigger_event_non_silently = { id = tv_grand_general.2 days = 1 }
				}
				else = {
					trigger_event_non_silently = { id = tv_grand_general.1 days = 1 }
				}
			}
			49 = {
				if = {
					limit = {
						var:tv_conquest_general_char ?= {
							OR = {
								has_trait = tv_gg_offense_3
								has_trait = tv_gg_defense_3
								has_trait = tv_gg_tactics_3
							}
						}
					}
					if = {
						limit = { var:tv_conquest_general_char ?= { mil >= 100 } }
						trigger_event_non_silently = { id = tv_grand_general.5 days = 1 }
					}
					else = {
						trigger_event_non_silently = { id = tv_grand_general.4 days = 1 }
					}
				}
				else = {
					trigger_event_non_silently = { id = tv_grand_general.3 days = 1 }
				}
			}
			1 = {
				if = {
					limit = { var:tv_conquest_general_char ?= { has_trait = tv_gg_injured } }
					trigger_event_non_silently = { id = tv_grand_general.7 days = 1 }
				}
				else = {
					trigger_event_non_silently = { id = tv_grand_general.6 days = 1 }
				}
			}
			1 = {
				trigger_event_non_silently = { id = tv_grand_general.8 days = 1 }
			}
		}
	}
	tv_training_clear_effect = yes
}

# ── Governor system (总督 per region) ───────────────────────────────────────
# Data model:
#   tv_govhouse_region_list: country variable list of region scopes, rebuilt monthly
#                            from the country's currently owned locations.
#   tv_govhouse_region_display_list:
#                            country variable list of representative owned location scopes,
#                            one per owned region, used by the Governor's House GUI.
#   tv_govhouse_region_display_region_list:
#                            country variable list of region scopes, internal de-dupe helper
#                            for building tv_govhouse_region_display_list.
#   tv_governor_list:        country variable list of governor character scopes.
# Each representative display location may carry:
#   tv_govhouse_display_governor_char      (character scope value)
#   tv_govhouse_display_task_type          (1 prosperity, 2 integration, 3 culture, 4 religion, 5 recruitment, 6 order)
#   tv_govhouse_display_task_value         (copied from governor)
#   tv_govhouse_display_weight_restore_prosperity
#   tv_govhouse_display_weight_integrate_region
#   tv_govhouse_display_weight_convert_culture
#   tv_govhouse_display_weight_convert_religion
#   tv_govhouse_display_weight_strengthen_recruitment
#   tv_govhouse_display_weight_strengthen_order
#   tv_govhouse_display_admin_bonus_type   (1 control, 2 development, 3 production)
#   tv_govhouse_display_admin_bonus_value  (copied from governor)
#   tv_govhouse_display_dip_bonus_type     (1 construction, 2 pop promotion, 3 pop growth)
#   tv_govhouse_display_dip_bonus_value    (copied from governor)
#   tv_govhouse_display_mil_bonus_type     (1 unrest, 2 levy size, 3 fort defense)
#   tv_govhouse_display_mil_bonus_value    (copied from governor)
# Each governor character carries:
#   tv_is_governor = 1            (presence flag)
#   tv_governed_region            (region scope value)
#   tv_governed_region_capital    (representative owned location, for legacy display compatibility)
#   tv_gov_current_task           (1 prosperity, 2 integration, 3 culture, 4 religion, 5 recruitment, 6 order)
#   tv_gov_current_task_value     (active task effect size)
#   tv_gov_admin_bonus_type/value (chosen by Local Governor Administrative Function, if enacted)
#   tv_gov_dip_bonus_type/value   (chosen by Local Governor Diplomatic Function, if enacted)
#   tv_gov_mil_bonus_type/value   (chosen by Local Governor Military Function, if enacted)
# Location modifiers applied: one current task, plus any enacted ADM/DIP/MIL governor function modifiers
# Scaling: size reads pre-computed pct variable directly (no inline multiply needed)

# Rebuilds the player-region backing list from currently owned locations.
# Scope: country.
tv_govhouse_refresh_region_list_effect = {
	save_scope_as = tv_governing_country
	clear_variable_list = tv_govhouse_region_list
	clear_variable_list = tv_govhouse_region_display_list
	clear_variable_list = tv_govhouse_region_display_region_list
	every_owned_location = {
		remove_variable = tv_govhouse_display_governor_char
		remove_variable = tv_govhouse_display_task_type
		remove_variable = tv_govhouse_display_task_value
		remove_variable = tv_govhouse_display_weight_restore_prosperity
		remove_variable = tv_govhouse_display_weight_integrate_region
		remove_variable = tv_govhouse_display_weight_convert_culture
		remove_variable = tv_govhouse_display_weight_convert_religion
		remove_variable = tv_govhouse_display_weight_strengthen_recruitment
		remove_variable = tv_govhouse_display_weight_strengthen_order
		remove_variable = tv_govhouse_display_admin_bonus_type
		remove_variable = tv_govhouse_display_admin_bonus_value
		remove_variable = tv_govhouse_display_dip_bonus_type
		remove_variable = tv_govhouse_display_dip_bonus_value
		remove_variable = tv_govhouse_display_mil_bonus_type
		remove_variable = tv_govhouse_display_mil_bonus_value
		region = {
			save_scope_as = tv_owned_region
		}
		scope:tv_governing_country = {
			if = {
				limit = {
					exists = scope:tv_owned_region
					OR = {
						NOT = { has_variable_list = tv_govhouse_region_list }
						NOT = {
							is_target_in_variable_list = {
								name = tv_govhouse_region_list
								target = scope:tv_owned_region
							}
						}
					}
				}
				add_to_variable_list = {
					name = tv_govhouse_region_list
					target = scope:tv_owned_region
				}
			}
		}
	}
	every_in_list = { variable = tv_governor_list
		limit = {
			is_alive = yes
			var:tv_governed_region ?= {
				any_location_in_region = {
					owner ?= scope:tv_governing_country
				}
			}
		}
		save_scope_as = tv_display_governor
		var:tv_governed_region ?= {
			save_scope_as = tv_display_region
			random_location_in_region = {
				limit = { owner ?= scope:tv_governing_country }
				save_scope_as = tv_display_location
			}
		}
		if = {
			limit = { exists = scope:tv_display_location }
			set_variable = { name = tv_governed_region_capital value = scope:tv_display_location }
			scope:tv_display_location = {
				set_variable = { name = tv_govhouse_display_governor_char value = scope:tv_display_governor }
				set_variable = { name = tv_govhouse_display_task_type value = scope:tv_display_governor.var:tv_gov_current_task }
				set_variable = { name = tv_govhouse_display_task_value value = scope:tv_display_governor.var:tv_gov_current_task_value }
				if = {
					limit = {
						scope:tv_display_governor = { var:tv_gov_current_task ?= 6 }
					}
					change_variable = { name = tv_govhouse_display_task_value multiply = 1000 }
				}
				set_variable = { name = tv_govhouse_display_weight_restore_prosperity value = scope:tv_display_governor.var:tv_gov_weight_restore_prosperity }
				set_variable = { name = tv_govhouse_display_weight_integrate_region value = scope:tv_display_governor.var:tv_gov_weight_integrate_region }
				set_variable = { name = tv_govhouse_display_weight_convert_culture value = scope:tv_display_governor.var:tv_gov_weight_convert_culture }
				set_variable = { name = tv_govhouse_display_weight_convert_religion value = scope:tv_display_governor.var:tv_gov_weight_convert_religion }
				set_variable = { name = tv_govhouse_display_weight_strengthen_recruitment value = scope:tv_display_governor.var:tv_gov_weight_strengthen_recruitment }
				set_variable = { name = tv_govhouse_display_weight_strengthen_order value = scope:tv_display_governor.var:tv_gov_weight_strengthen_order }
				if = {
					limit = {
						scope:tv_display_governor = {
							has_variable = tv_gov_admin_bonus_type
							has_variable = tv_gov_admin_bonus_value
						}
					}
					set_variable = { name = tv_govhouse_display_admin_bonus_type value = scope:tv_display_governor.var:tv_gov_admin_bonus_type }
					set_variable = { name = tv_govhouse_display_admin_bonus_value value = scope:tv_display_governor.var:tv_gov_admin_bonus_value }
				}
				if = {
					limit = {
						scope:tv_display_governor = {
							has_variable = tv_gov_dip_bonus_type
							has_variable = tv_gov_dip_bonus_value
						}
					}
					set_variable = { name = tv_govhouse_display_dip_bonus_type value = scope:tv_display_governor.var:tv_gov_dip_bonus_type }
					set_variable = { name = tv_govhouse_display_dip_bonus_value value = scope:tv_display_governor.var:tv_gov_dip_bonus_value }
				}
				if = {
					limit = {
						scope:tv_display_governor = {
							has_variable = tv_gov_mil_bonus_type
							has_variable = tv_gov_mil_bonus_value
						}
					}
					set_variable = { name = tv_govhouse_display_mil_bonus_type value = scope:tv_display_governor.var:tv_gov_mil_bonus_type }
					set_variable = { name = tv_govhouse_display_mil_bonus_value value = scope:tv_display_governor.var:tv_gov_mil_bonus_value }
				}
			}
			scope:tv_governing_country = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable_list = tv_govhouse_region_display_region_list }
							NOT = {
								is_target_in_variable_list = {
									name = tv_govhouse_region_display_region_list
									target = scope:tv_display_region
								}
							}
						}
					}
					add_to_variable_list = { name = tv_govhouse_region_display_region_list target = scope:tv_display_region }
					add_to_variable_list = { name = tv_govhouse_region_display_list target = scope:tv_display_location }
				}
			}
		}
	}
	every_owned_location = {
		save_scope_as = tv_display_location
		region = {
			save_scope_as = tv_display_region
		}
		scope:tv_governing_country = {
			if = {
				limit = {
					exists = scope:tv_display_region
					OR = {
						NOT = { has_variable_list = tv_govhouse_region_display_region_list }
						NOT = {
							is_target_in_variable_list = {
								name = tv_govhouse_region_display_region_list
								target = scope:tv_display_region
							}
						}
					}
				}
				add_to_variable_list = { name = tv_govhouse_region_display_region_list target = scope:tv_display_region }
				add_to_variable_list = { name = tv_govhouse_region_display_list target = scope:tv_display_location }

		}

# Recomputes the autonomous task that one local Governor considers most valuable.
# Scope: governor character. Caller must provide scope:tv_governing_country.

	save_scope_as = tv_task_governor
	set_variable = { name = tv_gov_task_restore_prosperity_value value = { value = this.adm multiply = 0.0005 } }
	set_variable = { name = tv_gov_task_integrate_region_value value = { value = this.adm multiply = 0.001 } }
	set_variable = { name = tv_gov_task_convert_culture_value value = { value = this.dip multiply = 0.0001 } }
	set_variable = { name = tv_gov_task_convert_religion_value value = { value = this.dip multiply = 0.0001 } }
	set_variable = { name = tv_gov_task_strengthen_recruitment_value value = { value = this.mil multiply = 0.005 } }
	set_variable = { name = tv_gov_task_strengthen_order_value value = { value = this.mil multiply = 0.00005 } }
	if = {
		limit = {
			scope:tv_governing_country = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_governing_country
					international_organization_has_policy = policy:tv_govhouse_general_garrison_orders_policy
				}
			}
		}
		change_variable = { name = tv_gov_task_restore_prosperity_value multiply = 1.2 }
		change_variable = { name = tv_gov_task_integrate_region_value multiply = 1.2 }
		change_variable = { name = tv_gov_task_convert_culture_value multiply = 1.2 }
		change_variable = { name = tv_gov_task_convert_religion_value multiply = 1.2 }
		change_variable = { name = tv_gov_task_strengthen_recruitment_value multiply = 1.2 }
		change_variable = { name = tv_gov_task_strengthen_order_value multiply = 1.2 }
	}

	set_variable = { name = tv_gov_weight_restore_prosperity value = 0 }
	set_variable = { name = tv_gov_weight_integrate_region value = 0 }
	set_variable = { name = tv_gov_weight_convert_culture value = 0 }
	set_variable = { name = tv_gov_weight_convert_religion value = 0 }
	set_variable = { name = tv_gov_weight_strengthen_recruitment value = 0 }
	set_variable = { name = tv_gov_weight_strengthen_order value = 0 }

				change_variable = { name = tv_gov_task_location_count add = 1 }
				change_variable = { name = tv_gov_task_prosperity_sum add = scope:tv_task_location.prosperity }
			}
			if = {
				limit = {
					OR = {
						integration_level = conquered
						integration_level = colonized
					}
				}
				scope:tv_task_governor = {
					change_variable = { name = tv_gov_task_unintegrated_count add = 1 }
				}
			}
			else_if = {
				limit = {
					NOT = { is_core_of = scope:tv_governing_country }
				}

					change_variable = { name = tv_gov_task_integrated_noncore_count add = 1 }
				}
			}
			if = {
				limit = {
					NOT = { dominant_religion = scope:tv_governing_country.religion }
				}
				scope:tv_task_governor = {
					change_variable = { name = tv_gov_task_religion_need_count add = 1 }
				}
			}
		}
	}

	if = {
		limit = { var:tv_gov_task_location_count ?= { this > 0 } }
		change_variable = { name = tv_gov_task_prosperity_sum divide = var:tv_gov_task_location_count }
		set_variable = { name = tv_gov_weight_restore_prosperity value = var:tv_gov_task_prosperity_sum }
		change_variable = { name = tv_gov_weight_restore_prosperity multiply = -1 }
		if = {
			limit = { var:tv_gov_weight_restore_prosperity ?= { this < 0 } }
			set_variable = { name = tv_gov_weight_restore_prosperity value = 0 }
		}

		set_variable = { name = tv_gov_weight_integrate_region value = var:tv_gov_task_unintegrated_count }
		change_variable = { name = tv_gov_weight_integrate_region divide = var:tv_gov_task_location_count }

		if = {
			limit = { var:tv_gov_task_unintegrated_count ?= 0 }
			set_variable = { name = tv_gov_weight_convert_culture value = var:tv_gov_task_integrated_noncore_count }
			change_variable = { name = tv_gov_weight_convert_culture divide = var:tv_gov_task_location_count }
			set_variable = { name = tv_gov_weight_convert_religion value = var:tv_gov_task_religion_need_count }
			change_variable = { name = tv_gov_weight_convert_religion divide = var:tv_gov_task_location_count }
			set_variable = { name = tv_gov_weight_strengthen_order value = 0.2 }
		}
	}

	if = {
		limit = { scope:tv_governing_country = { at_war = yes } }
		set_variable = { name = tv_gov_weight_strengthen_recruitment value = 0.5 }
	}

	if = {
		limit = {
			scope:tv_governing_country = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_governing_country

		}
		change_variable = { name = tv_gov_weight_restore_prosperity add = 0.30 }
	}
	if = {
		limit = {
			scope:tv_governing_country = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_governing_country
					international_organization_has_policy = policy:tv_govhouse_duty_integrate_policy
				}
			}

	}

		limit = {
			scope:tv_governing_country = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_governing_country

	if = {
		limit = {

					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_governing_country

				}
			}
		}
		change_variable = { name = tv_gov_weight_strengthen_order add = 0.30 }
	}

	if = {
		limit = {
			has_variable = tv_gov_weight_restore_prosperity
			has_variable = tv_gov_weight_integrate_region
			has_variable = tv_gov_weight_convert_culture
			has_variable = tv_gov_weight_convert_religion

			has_variable = tv_gov_task_convert_culture_value
			has_variable = tv_gov_task_convert_religion_value

			set_variable = { name = tv_gov_weight_best value = var:tv_gov_weight_convert_culture }
			set_variable = { name = tv_gov_current_task_value value = var:tv_gov_task_convert_culture_value }
		}
		if = {
			limit = { var:tv_gov_weight_convert_religion > var:tv_gov_weight_best }

			set_variable = { name = tv_gov_current_task_value value = var:tv_gov_task_convert_religion_value }

			limit = { var:tv_gov_weight_strengthen_order > var:tv_gov_weight_best }
			set_variable = { name = tv_gov_current_task value = 6 }
			set_variable = { name = tv_gov_weight_best value = var:tv_gov_weight_strengthen_order }
			set_variable = { name = tv_gov_current_task_value value = var:tv_gov_task_strengthen_order_value }
		}
	}
	else = {
		set_variable = { name = tv_gov_current_task value = 1 }
		set_variable = { name = tv_gov_weight_best value = 0 }
		set_variable = { name = tv_gov_current_task_value value = { value = this.adm multiply = 0.0005 } }
	}

	remove_variable = tv_gov_task_location_count
	remove_variable = tv_gov_task_unintegrated_count
	remove_variable = tv_gov_task_integrated_noncore_count
	remove_variable = tv_gov_task_religion_need_count
	remove_variable = tv_gov_task_prosperity_sum
	remove_variable = tv_gov_weight_best
}

# Refreshes all governor modifiers; cleans up dead governors.
# Scope: country. Called monthly from tv_govhouse_governor_bonuses_monthly_pulse.
tv_apply_governor_bonuses_effect = {
	save_scope_as = tv_governing_country
	tv_govhouse_refresh_region_list_effect = yes
	every_in_list = { variable = tv_governor_list
		if = {
			limit = { is_alive = no }
			# Dead governor — strip their region's modifiers and remove them from the list
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_dead_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_dead_governor }
			}
		}
		else_if = {
			limit = {
				NOT = { has_variable = tv_governed_region }
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_orphaned_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_orphaned_governor }
			}
		}
		else_if = {
			limit = {
				var:tv_governed_region ?= {
					NOT = {
						any_location_in_region = {
							owner ?= scope:tv_governing_country
						}
					}
				}
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_lost_region_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_lost_region_governor }
			}
		}
		else = {
			save_scope_as = target
			scope:tv_governing_country = {
				tv_governor_apply_modifiers_effect = yes
			}
		}
	}
	tv_govhouse_refresh_region_list_effect = yes
}

		limit = {
			exists = scope:target
			exists = scope:target_1
			scope:target = {
				is_female = no
				is_artist = no
			}
		}
		# Displace any existing governor for this region first
		every_in_list = { variable = tv_governor_list
			limit = {
				OR = {
					var:tv_governed_region ?= scope:target_1
					var:tv_governed_area ?= {
						region = scope:target_1
					}
				}
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_displaced_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_displaced_governor }
			}
		}
		# Set governor variables on the character
		scope:target = {
			set_variable = { name = tv_is_governor value = 1 }
			set_variable = { name = tv_governor_country value = scope:tv_governing_country }
			set_variable = { name = tv_governed_region value = scope:target_1 }
		}
		# Store a representative location from the region so the GUI can display the region name.
		scope:target_1 = {
			random_location_in_region = {
				limit = { owner ?= scope:tv_governing_country }
				save_scope_as = tv_temp_region_loc
			}
		}
		if = {
			limit = { exists = scope:tv_temp_region_loc }
			scope:target = {
				set_variable = { name = tv_governed_region_capital value = scope:tv_temp_region_loc }
			}
		}
		# Register in the country governor list
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = tv_governor_list }
					NOT = {
						is_target_in_variable_list = {
							name = tv_governor_list
							target = scope:target
						}
					}
				}
			}
			add_to_variable_list = { name = tv_governor_list target = scope:target }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = tv_govhouse_region_list }
					NOT = {
						is_target_in_variable_list = {
							name = tv_govhouse_region_list
							target = scope:target_1
						}
					}
				}
			}
			add_to_variable_list = { name = tv_govhouse_region_list target = scope:target_1 }
		}
		# Apply location modifiers and initialise display variables
		tv_governor_apply_modifiers_effect = yes
		tv_govhouse_refresh_region_list_effect = yes
	}
}

# Fills empty Local Governor seats with the highest total-ability eligible
# characters, stopping when all owned regions have governors or no candidate
# remains. Scope: country.
tv_governor_auto_assign_effect = {
	save_scope_as = tv_governing_country
	tv_apply_governor_bonuses_effect = yes
	set_variable = { name = tv_govhouse_auto_assign_continue value = 1 }
	set_variable = { name = tv_govhouse_auto_assign_safety value = 0 }
	while = {
		limit = {
			has_variable = tv_govhouse_auto_assign_continue
			var:tv_govhouse_auto_assign_continue ?= { this > 0 }
			var:tv_govhouse_auto_assign_safety ?= { this < 200 }
			any_in_list = { variable = tv_govhouse_region_list
				save_temporary_scope_as = tv_govhouse_auto_candidate_region
				any_location_in_region = {
					owner ?= scope:tv_governing_country
				}
				NOT = {
					scope:tv_governing_country = {
						any_in_list = { variable = tv_governor_list
							OR = {
								var:tv_governed_region ?= scope:tv_govhouse_auto_candidate_region
								var:tv_governed_area ?= {
									region = scope:tv_govhouse_auto_candidate_region
								}
							}
						}
					}
				}
			}
			any_character = {
				is_alive = yes
				is_adult = yes
				is_female = no
				is_artist = no
				is_ruler = no
				is_heir = no
				is_consort = no
				is_loyal = yes
				NOT = { has_variable = tv_is_governor }
			}
		}
		set_variable = { name = tv_govhouse_auto_assign_continue value = 0 }
		change_variable = { name = tv_govhouse_auto_assign_safety add = 1 }
		random_in_list = { variable = tv_govhouse_region_list
			limit = {
				save_temporary_scope_as = tv_govhouse_auto_candidate_region
				any_location_in_region = {
					owner ?= scope:tv_governing_country
				}
				NOT = {
					scope:tv_governing_country = {
						any_in_list = { variable = tv_governor_list
							OR = {
								var:tv_governed_region ?= scope:tv_govhouse_auto_candidate_region
								var:tv_governed_area ?= {
									region = scope:tv_govhouse_auto_candidate_region
								}
							}
						}
					}
				}
			}
			save_scope_as = target_1
		}
		ordered_character = {
			limit = {
				is_alive = yes
				is_adult = yes
				is_female = no
				is_artist = no
				is_ruler = no
				is_heir = no
				is_consort = no
				is_loyal = yes
				NOT = { has_variable = tv_is_governor }
			}
			order_by = total_abilities
			max = 1
			check_range_bounds = no
			save_scope_as = target
		}
		if = {
			limit = {
				exists = scope:target
				exists = scope:target_1
				scope:target = {
					is_alive = yes
					is_adult = yes
					is_female = no
					is_artist = no
					is_ruler = no
					is_heir = no
					is_consort = no
					is_loyal = yes
					NOT = { has_variable = tv_is_governor }
				}
				scope:target_1 = {
					any_location_in_region = {
						owner ?= scope:tv_governing_country
					}
				}
			}
			tv_governor_appoint_effect = yes
			set_variable = { name = tv_govhouse_auto_assign_continue value = 1 }
		}
	}
	remove_variable = tv_govhouse_auto_assign_continue
	remove_variable = tv_govhouse_auto_assign_safety
	tv_govhouse_refresh_region_list_effect = yes
}

# Executes one auto-assign pass only while the Governor's House automation switch
# is enabled and a living Grand General can legally appoint governors.
# Scope: country.
tv_govhouse_auto_assign_if_enabled_effect = {
	if = {
		limit = {
			has_variable = tv_govhouse_governor_automation_enabled
			has_variable = tv_govhouse_member
			has_variable = tv_conquest_general_char
			var:tv_conquest_general_char ?= {
				is_alive = yes
				is_female = no
				is_artist = no
			}
		}
		tv_governor_auto_assign_effect = yes
	}
}

# Clears a Local Governor who became a consort through marriage and refills the
# vacancy if Governor's House automation is enabled. Scope: character.
tv_govhouse_clear_governor_if_consort_effect = {
	if = {
		limit = {
			has_variable = tv_is_governor
			is_consort = yes
			has_variable = tv_governor_country
			var:tv_governor_country ?= {
				has_variable = tv_govhouse_member
			}
		}
		save_scope_as = tv_consort_governor
		var:tv_governor_country ?= {
			save_scope_as = tv_governing_country
		}
		tv_governor_clear_assignment_effect = yes
		scope:tv_governing_country = {
			remove_list_variable = { name = tv_governor_list target = scope:tv_consort_governor }
			tv_govhouse_refresh_region_list_effect = yes
			tv_govhouse_auto_assign_if_enabled_effect = yes
		}
	}
	else_if = {
		limit = {
			has_variable = tv_is_governor
			is_consort = yes
			owner ?= {
				has_variable = tv_govhouse_member
			}
		}
		save_scope_as = tv_consort_governor
		owner = {
			save_scope_as = tv_governing_country
		}
		tv_governor_clear_assignment_effect = yes
		scope:tv_governing_country = {
			remove_list_variable = { name = tv_governor_list target = scope:tv_consort_governor }
			tv_govhouse_refresh_region_list_effect = yes
			tv_govhouse_auto_assign_if_enabled_effect = yes
		}
	}
}

# Dismisses a governor, removing their location modifiers.
# Scope: country. Prefer scope:target_1 = region (set via GUI button parameter).
# Legacy scope:recipient = governor character is still accepted for old UI/save compatibility.
tv_governor_remove_effect = {
	save_scope_as = tv_governing_country
	if = {
		limit = { exists = scope:recipient }
		scope:recipient = {
			tv_governor_clear_assignment_effect = yes
		}
		remove_list_variable = { name = tv_governor_list target = scope:recipient }
	}
	else_if = {
		limit = { exists = scope:target_1 }
		every_in_list = { variable = tv_governor_list
			limit = {
				OR = {
					var:tv_governed_region ?= scope:target_1
					var:tv_governed_area ?= {
						region = scope:target_1
					}
				}
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_removed_region_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_removed_region_governor }
			}
		}
	}
	tv_govhouse_refresh_region_list_effect = yes
	tv_govhouse_auto_assign_if_enabled_effect = yes
}
