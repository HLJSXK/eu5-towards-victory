# Creates the Governor's House IO for scope country (must be called as owner country).
# Sets the tv_govhouse_member flag, creates the IO, and adds the country as its first member.
tv_govhouse_create_effect = {
	save_scope_as = tv_govhouse_founder
	if = {
		limit = {
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_govhouse
					leader_country ?= scope:tv_govhouse_founder
				}
			}
		}
		set_variable = { name = tv_govhouse_member value = 1 }
		create_international_organization = {
			type = international_organization_type:tv_govhouse
			add_country_to_international_organization = prev
			set_leader_country = prev
		}
		tv_govhouse_refresh_region_list_effect = yes
	}
}

# Resets the IO-scoped Grand General consolidation value for this country's
# Governor's House. Scope: leader country / Governor's House owner.
tv_govhouse_reset_power_consolidation_effect = {
	save_scope_as = tv_govhouse_power_owner
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_govhouse
			leader_country ?= scope:tv_govhouse_power_owner
		}
		set_variable = { name = tv_govhouse_power_consolidation value = 0 }
	}
}

# Lowers this country's Governor's House balance of power by 5 percentage
# points. Scope: leader country / Governor's House owner.
tv_govhouse_reduce_power_balance_effect = {
	save_scope_as = tv_govhouse_power_owner
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_govhouse
			leader_country ?= scope:tv_govhouse_power_owner
		}
		set_variable = {
			name = tv_govhouse_power_balance
			value = {
				value = var:tv_govhouse_power_balance
				subtract = 0.05
				min = 0
			}
		}
	}
}

# Pays the dismissal/replacement price for the current Grand General.
# At 100% power consolidation this costs 20 government power.
# Scope: leader country / Governor's House owner.
tv_govhouse_pay_grand_general_replacement_cost_effect = {
	save_scope_as = tv_govhouse_power_owner
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_govhouse
			leader_country ?= scope:tv_govhouse_power_owner
		}
		scope:tv_govhouse_power_owner = {
			add_government_power = {
				value = prev.var:tv_govhouse_power_consolidation
				multiply = -20
			}
		}
	}
}

# Starts Grand General training. Marks the Grand General as in training,
# starts the month counter, and applies the temporary court cost modifier.
# Called by the tv_train_grand_general generic action.
# Scope: actor country.
tv_training_start_effect = {
	var:tv_conquest_general_char ?= {
		set_variable = { name = tv_gg_training_active value = 1 }
		set_variable = { name = tv_gg_training_months value = 0 }
	}
	set_variable = { name = tv_gg_training_progress value = 0 }
	add_country_modifier = {
		modifier = tv_grand_general_training_modifier
		years = 1
		mode = replace
	}
}

# Clears Grand General training state from the current stored Grand General.
# Scope: owner country.
tv_training_clear_effect = {
	var:tv_conquest_general_char ?= {
		remove_variable = tv_gg_training_active
		remove_variable = tv_gg_training_months
	}
	remove_variable = tv_gg_training_progress
	remove_country_modifier = tv_grand_general_training_modifier
}

# Fires the weighted training outcome for the Grand General.
# Called once tv_gg_training_active has accumulated 12 monthly ticks.
# Scope: actor country. 49 / 49 / 1 / 1 split.
# Routing is resolved here so each outcome event is directly meaningful.
tv_training_resolve_effect = {
	if = {
		limit = {
			var:tv_conquest_general_char ?= {
				is_alive = yes
				has_variable = tv_gg_training_active
			}
		}
		random_list = {
			49 = {
				if = {
					limit = { var:tv_conquest_general_char ?= { mil >= 100 } }
					trigger_event_non_silently = { id = tv_grand_general.2 }
				}
				else = {
					trigger_event_non_silently = { id = tv_grand_general.1 }
				}
			}
			49 = {
				if = {
					limit = {
						var:tv_conquest_general_char ?= {
							has_trait = tv_gg_offense_3
							has_trait = tv_gg_defense_3
							has_trait = tv_gg_tactics_3
						}
					}
					if = {
						limit = { var:tv_conquest_general_char ?= { mil >= 100 } }
						trigger_event_non_silently = { id = tv_grand_general.5 }
					}
					else = {
						trigger_event_non_silently = { id = tv_grand_general.4 }
					}
				}
				else = {
					trigger_event_non_silently = { id = tv_grand_general.3 }
				}
			}
			1 = {
				if = {
					limit = { var:tv_conquest_general_char ?= { has_trait = tv_gg_injured } }
					trigger_event_non_silently = { id = tv_grand_general.7 }
				}
				else = {
					trigger_event_non_silently = { id = tv_grand_general.6 }
				}
			}
			1 = {
				trigger_event_non_silently = { id = tv_grand_general.8 }
			}
		}
	}
	tv_training_clear_effect = yes
}

# ── Governor system (总督 per region) ───────────────────────────────────────
# Data model:
#   tv_govhouse_region_list: country variable list of region scopes, rebuilt monthly
#                            from the country's currently owned locations.
#   tv_govhouse_region_display_list:
#                            country variable list of representative owned location scopes,
#                            one per owned region, used by the Governor's House GUI.
#   tv_govhouse_region_display_region_list:
#                            country variable list of region scopes, internal de-dupe helper
#                            for building tv_govhouse_region_display_list.
#   tv_governor_list:        country variable list of governor character scopes.
# Each representative display location may carry:
#   tv_govhouse_display_governor_char      (character scope value)
#   tv_govhouse_display_max_control_pct    (copied from governor)
#   tv_govhouse_display_build_cost_pct      (copied from governor)
#   tv_govhouse_display_unrest_pct          (copied from governor)
# Each governor character carries:
#   tv_is_governor = 1            (presence flag)
#   tv_governed_region            (region scope value)
#   tv_governed_region_capital    (representative owned location, for legacy display compatibility)
#   tv_gov_max_control_pct        (adm × 0.001 — fractional; GUI |% format → "+10%")
#   tv_gov_build_cost_pct         (dip × 0.002 — fractional; GUI |% format → "-20%")
#   tv_gov_unrest_pct             (mil × 0.002 — fractional; GUI |% format → "-20%")
# Location modifiers applied: tv_governor_max_control, tv_governor_build_cost, tv_governor_unrest
# Scaling: size reads pre-computed pct variable directly (no inline multiply needed)

# Rebuilds the player-region backing list from currently owned locations.
# Scope: country.
tv_govhouse_refresh_region_list_effect = {
	save_scope_as = tv_governing_country
	clear_variable_list = tv_govhouse_region_list
	clear_variable_list = tv_govhouse_region_display_list
	clear_variable_list = tv_govhouse_region_display_region_list
	every_owned_location = {
		remove_variable = tv_govhouse_display_governor_char
		remove_variable = tv_govhouse_display_max_control_pct
		remove_variable = tv_govhouse_display_build_cost_pct
		remove_variable = tv_govhouse_display_unrest_pct
		region = {
			save_scope_as = tv_owned_region
		}
		scope:tv_governing_country = {
			if = {
				limit = {
					exists = scope:tv_owned_region
					OR = {
						NOT = { has_variable_list = tv_govhouse_region_list }
						NOT = {
							is_target_in_variable_list = {
								name = tv_govhouse_region_list
								target = scope:tv_owned_region
							}
						}
					}
				}
				add_to_variable_list = {
					name = tv_govhouse_region_list
					target = scope:tv_owned_region
				}
			}
		}
	}
	every_in_list = { variable = tv_governor_list
		limit = {
			is_alive = yes
			var:tv_governed_region ?= {
				any_location_in_region = {
					owner ?= scope:tv_governing_country
				}
			}
		}
		save_scope_as = tv_display_governor
		var:tv_governed_region ?= {
			save_scope_as = tv_display_region
			random_location_in_region = {
				limit = { owner ?= scope:tv_governing_country }
				save_scope_as = tv_display_location
			}
		}
		if = {
			limit = { exists = scope:tv_display_location }
			set_variable = { name = tv_governed_region_capital value = scope:tv_display_location }
			scope:tv_display_location = {
				set_variable = { name = tv_govhouse_display_governor_char value = scope:tv_display_governor }
				set_variable = { name = tv_govhouse_display_max_control_pct value = scope:tv_display_governor.var:tv_gov_max_control_pct }
				set_variable = { name = tv_govhouse_display_build_cost_pct value = scope:tv_display_governor.var:tv_gov_build_cost_pct }
				set_variable = { name = tv_govhouse_display_unrest_pct value = scope:tv_display_governor.var:tv_gov_unrest_pct }
			}
			scope:tv_governing_country = {
				if = {
					limit = {
						OR = {
							NOT = { has_variable_list = tv_govhouse_region_display_region_list }
							NOT = {
								is_target_in_variable_list = {
									name = tv_govhouse_region_display_region_list
									target = scope:tv_display_region
								}
							}
						}
					}
					add_to_variable_list = { name = tv_govhouse_region_display_region_list target = scope:tv_display_region }
					add_to_variable_list = { name = tv_govhouse_region_display_list target = scope:tv_display_location }
				}
			}
		}
	}
	every_owned_location = {
		save_scope_as = tv_display_location
		region = {
			save_scope_as = tv_display_region
		}
		scope:tv_governing_country = {
			if = {
				limit = {
					exists = scope:tv_display_region
					OR = {
						NOT = { has_variable_list = tv_govhouse_region_display_region_list }
						NOT = {
							is_target_in_variable_list = {
								name = tv_govhouse_region_display_region_list
								target = scope:tv_display_region
							}
						}
					}
				}
				add_to_variable_list = { name = tv_govhouse_region_display_region_list target = scope:tv_display_region }
				add_to_variable_list = { name = tv_govhouse_region_display_list target = scope:tv_display_location }
			}
		}
	}
}

# Clears one governor assignment and removes its active location modifiers.
# Scope: governor character. Caller must provide scope:tv_governing_country.
tv_governor_clear_assignment_effect = {
	var:tv_governed_region ?= {
		every_location_in_region = {
			limit = { owner ?= scope:tv_governing_country }
			remove_location_modifier = tv_governor_max_control
			remove_location_modifier = tv_governor_build_cost
			remove_location_modifier = tv_governor_unrest
		}
	}
	var:tv_governed_area ?= {
		every_location_in_area = {
			limit = { owner ?= scope:tv_governing_country }
			remove_location_modifier = tv_governor_max_control
			remove_location_modifier = tv_governor_build_cost
			remove_location_modifier = tv_governor_unrest
		}
	}
	remove_variable = tv_is_governor
	remove_variable = tv_governed_region
	remove_variable = tv_governed_region_capital
	remove_variable = tv_governed_area
	remove_variable = tv_governed_area_capital
	remove_variable = tv_gov_max_control_pct
	remove_variable = tv_gov_build_cost_pct
	remove_variable = tv_gov_unrest_pct
}

# Applies or refreshes location modifiers and display variables for one governor.
# Scope: country. scope:target = governor character. scope:tv_governing_country = owning country.
# Caller must save_scope_as = tv_governing_country before calling this effect.
# Note: scope:target here is the character (set by select_trigger target_flag = target in
# tv_appoint_governor, or by save_scope_as = target in tv_apply_governor_bonuses_effect).
tv_governor_apply_modifiers_effect = {
	scope:target = {
		# Refresh display variables (shared scale with size — GUI uses |% format)
		set_variable = { name = tv_gov_max_control_pct value = { value = this.adm multiply = 0.001 } }
		set_variable = { name = tv_gov_build_cost_pct value = { value = this.dip multiply = 0.002 } }
		set_variable = { name = tv_gov_unrest_pct value = { value = this.mil multiply = 0.002 } }
		# Migrate pre-region governors from the old area assignment, preserving the appointment.
		if = {
			limit = {
				has_variable = tv_governed_area
				NOT = { has_variable = tv_governed_region }
			}
			var:tv_governed_area ?= {
				random_location_in_area = {
					limit = { owner ?= scope:tv_governing_country }
					save_scope_as = tv_temp_legacy_region_loc
				}
			}
			if = {
				limit = { exists = scope:tv_temp_legacy_region_loc }
				scope:tv_temp_legacy_region_loc = {
					region = {
						save_scope_as = tv_temp_legacy_region
					}
				}
			}
			if = {
				limit = { exists = scope:tv_temp_legacy_region }
				set_variable = { name = tv_governed_region value = scope:tv_temp_legacy_region }
				set_variable = { name = tv_governed_region_capital value = scope:tv_temp_legacy_region_loc }
			}
		}
		# Remove any old area-scoped modifiers before applying the region-wide set.
		var:tv_governed_area ?= {
			every_location_in_area = {
				limit = { owner ?= scope:tv_governing_country }
				remove_location_modifier = tv_governor_max_control
				remove_location_modifier = tv_governor_build_cost
				remove_location_modifier = tv_governor_unrest
			}
		}
		remove_variable = tv_governed_area
		remove_variable = tv_governed_area_capital
		# Apply to all owned locations in the governed region.
		# ?= guards the case where scope:target has no tv_governed_region yet
		# (e.g. EU5 pre-evaluates this effect after select_trigger step 1 picks the character
		# but before step 2 sets the region — fresh characters have no tv_governed_region variable).
		var:tv_governed_region ?= {
			every_location_in_region = {
				limit = { owner ?= scope:tv_governing_country }
				remove_location_modifier = tv_governor_max_control
				remove_location_modifier = tv_governor_build_cost
				remove_location_modifier = tv_governor_unrest
				add_location_modifier = {
					modifier = tv_governor_max_control
					years = -1
					mode = add_and_extend
					size = { value = scope:target.var:tv_gov_max_control_pct }
				}
				add_location_modifier = {
					modifier = tv_governor_build_cost
					years = -1
					mode = add_and_extend
					size = { value = scope:target.var:tv_gov_build_cost_pct }
				}
				add_location_modifier = {
					modifier = tv_governor_unrest
					years = -1
					mode = add_and_extend
					size = { value = scope:target.var:tv_gov_unrest_pct }
				}
			}
		}
	}
}

# Refreshes all governor modifiers; cleans up dead governors.
# Scope: country. Called monthly from the IO monthly_effect.
# root may be the IO when called via monthly_effect — use save_scope_as to pin the country.
tv_apply_governor_bonuses_effect = {
	save_scope_as = tv_governing_country
	tv_govhouse_refresh_region_list_effect = yes
	every_in_list = { variable = tv_governor_list
		if = {
			limit = { is_alive = no }
			# Dead governor — strip their region's modifiers and remove them from the list
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_dead_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_dead_governor }
			}
		}
		else_if = {
			limit = {
				NOT = { has_variable = tv_governed_region }
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_orphaned_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_orphaned_governor }
			}
		}
		else_if = {
			limit = {
				var:tv_governed_region ?= {
					NOT = {
						any_location_in_region = {
							owner ?= scope:tv_governing_country
						}
					}
				}
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_lost_region_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_lost_region_governor }
			}
		}
		else = {
			save_scope_as = target
			scope:tv_governing_country = {
				tv_governor_apply_modifiers_effect = yes
			}
		}
	}
	tv_govhouse_refresh_region_list_effect = yes
}

# Appoints a governor to a region. Displaces any existing governor for that region.
# Scope: country. scope:target = character (from select_trigger target_flag = target).
#                  scope:target_1 = region (from select_trigger target_flag = target_1).
tv_governor_appoint_effect = {
	save_scope_as = tv_governing_country
	# Guard: EU5 pre-evaluates the effect body during select_trigger steps.
	# Both scopes are unset at pre-eval time; skip silently and execute only when both are set.
	if = {
		limit = {
			exists = scope:target
			exists = scope:target_1
			scope:target = {
				is_female = no
				is_artist = no
			}
		}
		# Displace any existing governor for this region first
		every_in_list = { variable = tv_governor_list
			limit = {
				OR = {
					var:tv_governed_region ?= scope:target_1
					var:tv_governed_area ?= {
						region = scope:target_1
					}
				}
			}
			tv_governor_clear_assignment_effect = yes
			save_scope_as = tv_displaced_governor
			scope:tv_governing_country = {
				remove_list_variable = { name = tv_governor_list target = scope:tv_displaced_governor }
			}
		}
		# Set governor variables on the character
		scope:target = {
			set_variable = { name = tv_is_governor value = 1 }
			set_variable = { name = tv_governed_region value = scope:target_1 }
		}
		# Store a representative location from the region so the GUI can display the region name.
		scope:target_1 = {
			random_location_in_region = {
				limit = { owner ?= scope:tv_governing_country }
				save_scope_as = tv_temp_region_loc
			}
		}
		if = {
			limit = { exists = scope:tv_temp_region_loc }
			scope:target = {
				set_variable = { name = tv_governed_region_capital value = scope:tv_temp_region_loc }
			}
		}
		# Register in the country governor list
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = tv_governor_list }
					NOT = {
						is_target_in_variable_list = {
							name = tv_governor_list
							target = scope:target
						}
					}
				}
			}
			add_to_variable_list = { name = tv_governor_list target = scope:target }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = tv_govhouse_region_list }
					NOT = {
						is_target_in_variable_list = {
							name = tv_govhouse_region_list
							target = scope:target_1
						}
					}
				}
			}
			add_to_variable_list = { name = tv_govhouse_region_list target = scope:target_1 }
		}
		# Apply location modifiers and initialise display variables
		tv_governor_apply_modifiers_effect = yes
		tv_govhouse_refresh_region_list_effect = yes
	}
}
