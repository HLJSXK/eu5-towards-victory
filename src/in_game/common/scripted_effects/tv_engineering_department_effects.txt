# Towards Victory - Engineering Department effects
# Country-scoped unless noted.

tv_engineering_department_create_effect = {
	save_scope_as = tv_engineering_department_founder
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_engineering_department_member }
			}
		}
		set_variable = { name = tv_engineering_department_member value = 1 }
	}
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_wonder_stage }
			}
		}
		set_variable = { name = tv_wonder_stage value = 0 }
	}
	if = {
		limit = {
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_engineering_department
					leader_country ?= scope:tv_engineering_department_founder
				}
			}
		}
		create_international_organization = {
			type = international_organization_type:tv_engineering_department
			add_country_to_international_organization = prev
			set_leader_country = prev
		}
	}
}

tv_wonder_set_proposal_on_great_engineer_effect = {
	if = {
		limit = { var:tv_wonder_proposal ?= { this >= 1 } }
		var:tv_great_engineer_char ?= {
			set_variable = { name = tv_wonder_engineer_proposal value = prev.var:tv_wonder_proposal }
		}
	}
}

tv_wonder_clear_proposal_on_great_engineer_effect = {
	var:tv_great_engineer_char ?= {
		remove_variable = tv_wonder_engineer_proposal
	}
}

tv_wonder_restore_proposal_from_great_engineer_effect = {
	var:tv_great_engineer_char ?= {
		save_scope_as = tv_wonder_current_engineer
	}
	if = {
		limit = {
			exists = scope:tv_wonder_current_engineer
			scope:tv_wonder_current_engineer = {
				var:tv_wonder_engineer_proposal ?= { this >= 1 }
			}
		}
		set_variable = {
			name = tv_wonder_proposal
			value = scope:tv_wonder_current_engineer.var:tv_wonder_engineer_proposal
		}
	}
}

tv_wonder_roll_new_proposal_effect = {
	tv_wonder_clear_proposal_on_great_engineer_effect = yes
	remove_variable = tv_wonder_proposal
	if = {
		limit = { tv_wonder_has_any_feasible_proposal_trigger = yes }
		random_list = {
			1 = {
				trigger = { has_variable = tv_wonder_feasible_sacred_mountain }
				set_variable = { name = tv_wonder_proposal value = 1 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_triumphal_axis }
				set_variable = { name = tv_wonder_proposal value = 2 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_great_port }
				set_variable = { name = tv_wonder_proposal value = 3 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_giant_necropolis }
				set_variable = { name = tv_wonder_proposal value = 4 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_great_lighthouse }
				set_variable = { name = tv_wonder_proposal value = 5 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_hydraulic_workshop }
				set_variable = { name = tv_wonder_proposal value = 6 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_mining_city }
				set_variable = { name = tv_wonder_proposal value = 7 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_giant_observatory }
				set_variable = { name = tv_wonder_proposal value = 8 }
			}
			1 = {
				trigger = { has_variable = tv_wonder_feasible_palace_of_nations }
				set_variable = { name = tv_wonder_proposal value = 9 }
			}
		}
		tv_wonder_set_proposal_on_great_engineer_effect = yes
	}
}

		if = {
			limit = { NOT = { has_variable = tv_wonder_nobles_demand } }
			tv_wonder_initialize_debate_demands_effect = yes
		}
	}
}

	if = {

tv_wonder_update_construction_tiers_from_competence_effect = {
	if = {
		limit = { NOT = { has_variable = tv_wonder_scale_competence } }
		set_variable = { name = tv_wonder_scale_competence value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = tv_wonder_logistics_competence } }
		set_variable = { name = tv_wonder_logistics_competence value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = tv_wonder_organization_competence } }
		set_variable = { name = tv_wonder_organization_competence value = 0 }
	}
	clamp_variable = { name = tv_wonder_scale_competence min = 0 max = 100 }
	clamp_variable = { name = tv_wonder_logistics_competence min = 0 max = 100 }
	clamp_variable = { name = tv_wonder_organization_competence min = 0 max = 100 }

	set_variable = { name = tv_wonder_scale_tier value = 1 }
	if = { limit = { var:tv_wonder_scale_competence ?= { this >= 20 } } set_variable = { name = tv_wonder_scale_tier value = 2 } }
	if = { limit = { var:tv_wonder_scale_competence ?= { this >= 40 } } set_variable = { name = tv_wonder_scale_tier value = 3 } }
	if = { limit = { var:tv_wonder_scale_competence ?= { this >= 55 } } set_variable = { name = tv_wonder_scale_tier value = 4 } }
	if = { limit = { var:tv_wonder_scale_competence ?= { this >= 70 } } set_variable = { name = tv_wonder_scale_tier value = 5 } }
	if = { limit = { var:tv_wonder_scale_competence ?= { this >= 85 } } set_variable = { name = tv_wonder_scale_tier value = 6 } }

	set_variable = { name = tv_wonder_logistics_tier value = 1 }
	if = { limit = { var:tv_wonder_logistics_competence ?= { this >= 20 } } set_variable = { name = tv_wonder_logistics_tier value = 2 } }
	if = { limit = { var:tv_wonder_logistics_competence ?= { this >= 40 } } set_variable = { name = tv_wonder_logistics_tier value = 3 } }
	if = { limit = { var:tv_wonder_logistics_competence ?= { this >= 55 } } set_variable = { name = tv_wonder_logistics_tier value = 4 } }
	if = { limit = { var:tv_wonder_logistics_competence ?= { this >= 70 } } set_variable = { name = tv_wonder_logistics_tier value = 5 } }
	if = { limit = { var:tv_wonder_logistics_competence ?= { this >= 85 } } set_variable = { name = tv_wonder_logistics_tier value = 6 } }

	set_variable = { name = tv_wonder_organization_tier value = 1 }
	if = { limit = { var:tv_wonder_organization_competence ?= { this >= 20 } } set_variable = { name = tv_wonder_organization_tier value = 2 } }
