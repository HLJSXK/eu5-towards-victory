	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_engineering_department
		}
		custom_description = {
			text = tv_change_wonder_domestic_support_text
			value = $value$
			change_variable = { name = tv_wonder_domestic_support add = $value$ }
			clamp_variable = { name = tv_wonder_domestic_support min = 0 max = 200 }
		}
	}
}

tv_wonder_clamp_materials_to_country_cap_effect = {
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_engineering_department
		}
		clamp_variable = { name = tv_wonder_materials_stockpile min = 0 max = 300000 }
		if = {
			limit = {
				prev = { has_variable = tv_wonder_materials_cap }
				var:tv_wonder_materials_stockpile > prev.var:tv_wonder_materials_cap
			}
			set_variable = { name = tv_wonder_materials_stockpile value = prev.var:tv_wonder_materials_cap }
		}
	}
}

tv_wonder_clear_feasible_deck_effect = {
	remove_variable = tv_wonder_feasible_sacred_mountain

tv_wonder_rebuild_feasible_deck_effect = {

		set_variable = { name = tv_wonder_feasible_sacred_mountain value = 1 }
	}

		limit = { tv_wonder_can_build_triumphal_axis_trigger = yes }

tv_wonder_set_proposal_on_great_engineer_effect = {
	if = {
		limit = { var:tv_wonder_proposal ?= { this >= 1 } }
		var:tv_great_engineer_char ?= {
			set_variable = { name = tv_wonder_engineer_proposal value = prev.var:tv_wonder_proposal }
		}
	}
}

tv_wonder_clear_proposal_on_great_engineer_effect = {
	var:tv_great_engineer_char ?= {
		remove_variable = tv_wonder_engineer_proposal
	}
}

tv_wonder_restore_proposal_from_great_engineer_effect = {
	var:tv_great_engineer_char ?= {
		save_scope_as = tv_wonder_current_engineer
	}
	if = {
		limit = {
			exists = scope:tv_wonder_current_engineer
			scope:tv_wonder_current_engineer = {
				var:tv_wonder_engineer_proposal ?= { this >= 1 }
			}
		}
		set_variable = {
			name = tv_wonder_proposal
			value = scope:tv_wonder_current_engineer.var:tv_wonder_engineer_proposal
		}
	}
}

tv_wonder_roll_new_proposal_effect = {
	tv_wonder_clear_proposal_on_great_engineer_effect = yes
	remove_variable = tv_wonder_proposal
	if = {
		limit = { tv_wonder_has_any_feasible_proposal_trigger = yes }
		random_list = {
			1 = {
				modifier = { factor = 0 NOT = { has_variable = tv_wonder_feasible_sacred_mountain } }
				set_variable = { name = tv_wonder_proposal value = 1 }
			}
			1 = {
				modifier = { factor = 0 NOT = { has_variable = tv_wonder_feasible_triumphal_axis } }
				set_variable = { name = tv_wonder_proposal value = 2 }
			}
			1 = {
				modifier = { factor = 0 NOT = { has_variable = tv_wonder_feasible_great_port } }
				set_variable = { name = tv_wonder_proposal value = 3 }
			}
		}
		tv_wonder_set_proposal_on_great_engineer_effect = yes
	}
}

tv_wonder_update_logistics_metrics_effect = {
	set_variable = { name = tv_wonder_laborers value = 0 }
	set_variable = { name = tv_wonder_labor_cap value = 0 }
	set_variable = { name = tv_wonder_depot_level value = 0 }
	set_variable = { name = tv_wonder_materials_cap value = 0 }
	set_variable = { name = tv_wonder_dispatch_points value = 0 }
	set_variable = { name = tv_wonder_material_income value = 0 }

	if = {
		limit = { tv_wonder_construction_site_selected_trigger = yes }
		var:tv_wonder_site ?= { save_scope_as = tv_wonder_metric_site }
		set_variable = {
			name = tv_wonder_laborers
			value = {
				value = "scope:tv_wonder_metric_site.location_building_level(building_type:tv_wonder_labor_camp)"
				multiply = 1000
			}
		}
		set_variable = {
			name = tv_wonder_labor_cap
			value = {
				value = var:tv_wonder_scale_tier
				multiply = 5000
			}
		}
		set_variable = {
			name = tv_wonder_depot_level
			value = "scope:tv_wonder_metric_site.location_building_level(building_type:tv_wonder_material_depot)"
		}
		set_variable = {
			name = tv_wonder_materials_cap
			value = {
				value = var:tv_wonder_depot_level
				multiply = 60000
				max = 300000
			}
		}
		set_variable = {
			name = tv_wonder_dispatch_points
			value = total_effective_building_levels:tv_wonder_material_dispatch_point
		}
		set_variable = {
			name = tv_wonder_material_income
			value = {
				value = var:tv_wonder_dispatch_points
				multiply = 10000
			}
		}
		if = {
			limit = { var:tv_wonder_materials_cap > 0 }
			tv_wonder_change_materials_stockpile_effect = { value = var:tv_wonder_material_income }
		}
	}

	tv_wonder_clamp_materials_to_country_cap_effect = yes
}
