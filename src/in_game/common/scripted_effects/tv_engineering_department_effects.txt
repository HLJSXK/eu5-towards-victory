# Country-scoped unless noted.

tv_engineering_department_create_effect = {
	save_scope_as = tv_engineering_department_founder
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_engineering_department_member }
			}
		}
		set_variable = { name = tv_engineering_department_member value = 1 }
	}
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_wonder_stage }
			}
		}
		set_variable = { name = tv_wonder_stage value = 0 }
	}
	if = {
		limit = {
			NOT = {
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_engineering_department
					leader_country ?= scope:tv_engineering_department_founder
				}
			}
		}
		create_international_organization = {
			type = international_organization_type:tv_engineering_department
			add_country_to_international_organization = prev
			set_leader_country = prev
		}
	}
}

tv_wonder_clear_feasible_deck_effect = {
	remove_variable = tv_wonder_feasible_sacred_mountain
	remove_variable = tv_wonder_feasible_triumphal_axis
	remove_variable = tv_wonder_feasible_great_port
}

tv_wonder_rebuild_feasible_deck_effect = {
	tv_wonder_clear_feasible_deck_effect = yes
	if = {
		limit = { tv_wonder_can_build_sacred_mountain_trigger = yes }
		set_variable = { name = tv_wonder_feasible_sacred_mountain value = 1 }
	}
	if = {
		limit = { tv_wonder_can_build_triumphal_axis_trigger = yes }
		set_variable = { name = tv_wonder_feasible_triumphal_axis value = 1 }
	}
	if = {
		limit = { tv_wonder_can_build_great_port_trigger = yes }
		set_variable = { name = tv_wonder_feasible_great_port value = 1 }
	}
}

tv_wonder_set_proposal_on_great_engineer_effect = {
	if = {
		limit = { var:tv_wonder_proposal ?= { this >= 1 } }
		var:tv_great_engineer_char ?= {
			set_variable = { name = tv_wonder_engineer_proposal value = prev.var:tv_wonder_proposal }
		}
	}
}

tv_wonder_clear_proposal_on_great_engineer_effect = {
	var:tv_great_engineer_char ?= {
		remove_variable = tv_wonder_engineer_proposal
	}
}

tv_wonder_restore_proposal_from_great_engineer_effect = {
	var:tv_great_engineer_char ?= {
		save_scope_as = tv_wonder_current_engineer
	}
	if = {
		limit = {
			exists = scope:tv_wonder_current_engineer
			scope:tv_wonder_current_engineer = {
				var:tv_wonder_engineer_proposal ?= { this >= 1 }
			}
		}
		set_variable = {
			name = tv_wonder_proposal
			value = scope:tv_wonder_current_engineer.var:tv_wonder_engineer_proposal
		}
	}
}

tv_wonder_roll_new_proposal_effect = {
	tv_wonder_clear_proposal_on_great_engineer_effect = yes
	remove_variable = tv_wonder_proposal
	if = {
		limit = { tv_wonder_has_any_feasible_proposal_trigger = yes }
		random_list = {
			1 = {
				modifier = { factor = 0 NOT = { has_variable = tv_wonder_feasible_sacred_mountain } }
				set_variable = { name = tv_wonder_proposal value = 1 }
			}
			1 = {
				modifier = { factor = 0 NOT = { has_variable = tv_wonder_feasible_triumphal_axis } }
				set_variable = { name = tv_wonder_proposal value = 2 }
			}
			1 = {
				modifier = { factor = 0 NOT = { has_variable = tv_wonder_feasible_great_port } }
				set_variable = { name = tv_wonder_proposal value = 3 }
			}
		}
		tv_wonder_set_proposal_on_great_engineer_effect = yes
	}
}

tv_wonder_remove_current_proposal_from_deck_effect = {

}

tv_engineering_on_great_engineer_appointed_effect = {
	if = {
		limit = { NOT = { has_variable = tv_wonder_stage } }
		set_variable = { name = tv_wonder_stage value = 0 }
	}
	if = {
		limit = {
			tv_wonder_in_conception_trigger = yes
			tv_wonder_has_great_engineer_trigger = yes
		}
		remove_variable = tv_wonder_proposal
		tv_wonder_restore_proposal_from_great_engineer_effect = yes
		if = {
			limit = { NOT = { has_variable = tv_wonder_proposal } }
			tv_wonder_rebuild_feasible_deck_effect = yes
			tv_wonder_roll_new_proposal_effect = yes
		}
		if = {
			limit = { NOT = { has_variable = tv_wonder_nobles_demand } }
			tv_wonder_initialize_debate_demands_effect = yes
		}
	}
}

tv_engineering_on_great_engineer_removed_effect = {
	if = {
		limit = { tv_wonder_in_conception_trigger = yes }
		remove_variable = tv_wonder_proposal
		tv_wonder_clear_feasible_deck_effect = yes
	}
}

tv_wonder_accept_proposal_effect = {
	if = {
		limit = { var:tv_wonder_proposal ?= 1 }
		custom_tooltip = { text = TV_WONDER_LOCK_SACRED_MOUNTAIN_TT }
	}
	if = {
		limit = { var:tv_wonder_proposal ?= 2 }
		custom_tooltip = { text = TV_WONDER_LOCK_TRIUMPHAL_AXIS_TT }
	}
	if = {
		limit = { var:tv_wonder_proposal ?= 3 }
		custom_tooltip = { text = TV_WONDER_LOCK_GREAT_PORT_TT }
	}
	hidden_effect = {
		if = {
			limit = { var:tv_wonder_proposal ?= { this >= 1 } }
			set_variable = { name = tv_wonder_locked value = var:tv_wonder_proposal }
			remove_variable = tv_wonder_proposal
			set_variable = { name = tv_wonder_stage value = 1 }
			tv_wonder_reset_io_progress_effect = yes
			tv_wonder_initialize_debate_demands_effect = yes
		}
	}
}
