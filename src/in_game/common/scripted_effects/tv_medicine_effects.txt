# Towards Victory — Medicine mechanic scripted effects

# Reset all medicine state. Call from completion or patient death.
tv_med_cleanup_effect = {
	remove_variable = tv_med_target_char
	set_variable = { name = tv_med_progress  value = 0 }
	set_variable = { name = tv_med_disp_adm  value = 0 }
	set_variable = { name = tv_med_disp_dip  value = 0 }
	set_variable = { name = tv_med_disp_mil  value = 0 }
}

# Remove all known bad health traits from the patient character (called in character scope).
tv_med_remove_health_traits_effect = {
	if = { limit = { has_trait = blind }               remove_trait = trait:blind }
	if = { limit = { has_trait = one_eyed }            remove_trait = trait:one_eyed }
	if = { limit = { has_trait = maimed }              remove_trait = trait:maimed }
	if = { limit = { has_trait = disfigured }          remove_trait = trait:disfigured }
	if = { limit = { has_trait = scarred }             remove_trait = trait:scarred }
	if = { limit = { has_trait = hunchback }           remove_trait = trait:hunchback }
	if = { limit = { has_trait = pockmarked_trait }    remove_trait = trait:pockmarked_trait }
	if = { limit = { has_trait = smallpox_trait }      remove_trait = trait:smallpox_trait }
	if = { limit = { has_trait = bubonic_plague_trait } remove_trait = trait:bubonic_plague_trait }
	if = { limit = { has_trait = sickly }              remove_trait = trait:sickly }
}

# Called when tv_med_progress >= 100. Fires the recovery event; trait removal and cleanup
# happen in the event option so the player sees the patient's condition before it is removed.
# Progress is zeroed immediately to prevent re-triggering before the event is clicked.
# Must be called from country scope (leader_country).
tv_med_complete_treatment_effect = {
	set_variable = { name = tv_med_progress  value = 0 }
	trigger_event_non_silently = { id = tv_medicine.3 days = 1 }
}

# Reset meditation state and remove temporary balance modifiers.
tv_meditation_cleanup_effect = {
	var:tv_meditation_target_char ?= {
		if = {
			limit = { has_character_modifier = tv_meditating_character_modifier }
			remove_character_modifier = tv_meditating_character_modifier
		}
	}
	if = {
		limit = { has_country_modifier = tv_meditating_ruler_country_modifier }
		remove_country_modifier = tv_meditating_ruler_country_modifier
	}
	remove_variable = tv_meditation_target_char
	set_variable = { name = tv_meditation_progress  value = 0 }
}

# Meditation progress helper (country scope).
tv_change_meditation_progress_effect = {
	custom_description = {
		text = tv_change_meditation_progress_text
		value = $value$
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_MEDITATION_PROGRESS_TT
					has_variable = tv_meditation_progress
				}
			}
			change_variable = { name = tv_meditation_progress  add = $value$ }
			clamp_variable = { name = tv_meditation_progress min = 0 max = 100 }
			if = {
				limit = { var:tv_meditation_progress >= 100 }
				tv_meditation_complete_effect = yes
			}
		}
	}
}

# Called when meditation progress reaches 100. Trait removal and cleanup happen
# in the event option so the player can still see the target before resolution.
tv_meditation_complete_effect = {
	set_variable = { name = tv_meditation_progress  value = 0 }
	trigger_event_non_silently = { id = tv_medicine.20 days = 1 }
}
