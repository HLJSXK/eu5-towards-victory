# Towards Victory - Academy of Sciences domestic faction meetings
# These IO parliament issues turn Academy Assembly meetings into support tokens.

tv_academy_crown_mandate_issue = {
	type = international_organization
	special_status = tv_academy_president
	potential = { international_organization_type = international_organization_type:tv_academy_of_sciences }
	allow = { NOT = { has_cooldown = tv_academy_crown_mandate_issue } }
	chance = 20
	selectable_for = { is_leader_of_international_organization = scope:recipient }
	on_debate_passed = {
		add_cooldown = { type = tv_academy_crown_mandate_issue years = 3 }
		if = { limit = { NOT = { has_variable = tv_academy_crown_support } } set_variable = { name = tv_academy_crown_support value = 0 } }
		change_variable = { name = tv_academy_crown_support add = 1 }
		if = { limit = { var:tv_academy_crown_support > 3 } set_variable = { name = tv_academy_crown_support value = 3 } }
	}
	on_debate_failed = { add_cooldown = { type = tv_academy_crown_mandate_issue years = 1 } }
	wants_this_parliament_issue_bias = { add = { desc = "VOTE_BASE_VALUE" value = 1000 } }
}
