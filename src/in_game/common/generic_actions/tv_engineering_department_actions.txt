tv_wonder_start_survey = {
	type = owncountry
	sound = UI_action_religion_generic
	ai_tick = monthly
	ai_tick_frequency = 99999
	potential = { scope:actor = { has_variable = tv_engineering_department_member } }
	allow = {
		scope:actor = {
			tv_wonder_can_start_survey_trigger = yes
			tv_wonder_has_valid_site_candidate_trigger = yes
		}
	}
	select_trigger = {
		looking_for_a = location
		source = actor
		target_flag = target
		name = "tv_select_wonder_survey_site"
		none_available_msg_key = "tv_no_wonder_survey_site_available"
		column = { data = name }
		column = { data = population }
		column = { data = rank }
		visible = {
			has_owner = yes
			owner ?= scope:actor
			OR = {
				AND = {
					scope:actor = { var:tv_wonder_locked ?= 1 }
					OR = {
						topography = mountains
						topography = plateau
						topography = hills
					}
				}
				AND = {
					scope:actor = { var:tv_wonder_locked ?= 2 }
					OR = {
						location_rank ?= location_rank:city
						location_rank ?= location_rank:megalopolis
					}
				}
				AND = {
					scope:actor = { var:tv_wonder_locked ?= 3 }
					is_port = yes
				}
					AND = {
						scope:actor = { var:tv_wonder_locked ?= 4 }
						location_rank ?= location_rank:rural_settlement
					}
				AND = {
					scope:actor = { var:tv_wonder_locked ?= 5 }
					is_port = yes
				}
					AND = {
						scope:actor = { var:tv_wonder_locked ?= 6 }
						OR = {
							has_river = yes
							is_adjacent_to_lake = yes
						}
					}
					AND = {
						scope:actor = { var:tv_wonder_locked ?= 7 }
						OR = {
							raw_material = goods:iron
							raw_material = goods:copper
							raw_material = goods:tin
							raw_material = goods:lead
							raw_material = goods:silver
							raw_material = goods:goods_gold
						}
					}
					AND = {
						scope:actor = { var:tv_wonder_locked ?= 8 }
						OR = {
							topography = mountains
							topography = plateau
							topography = hills
						}
					}
					AND = {
						scope:actor = { var:tv_wonder_locked ?= 9 }
						is_capital = yes
					}
				}
			}
		}
	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = { tv_wonder_start_survey_effect = yes }
		}
	}
	ai_will_do = { add = -100 }
}

	type = owncountry

	ai_tick = monthly
	ai_tick_frequency = 99999
	potential = { scope:actor = { has_variable = tv_engineering_department_member } }
	allow = { scope:actor = { tv_wonder_can_request_emissary_trigger = yes } }
	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_neighbor_country = {
					limit = {
						country_type = location
						is_rebel_country = no
						capital = { is_discovered_by = scope:actor }
						OR = {
							NOT = { scope:actor = { has_variable_list = tv_wonder_emissary_countries } }
							NOT = {
								scope:actor = {
									is_target_in_variable_list = {
										name = tv_wonder_emissary_countries
										target = prev
									}
								}
							}
						}
					}
					add_to_list = source
				}
			}
		}
		target_flag = target
		allow_self = no
		name = "tv_select_wonder_emissary_country"
		none_available_msg_key = "tv_no_wonder_emissary_country_available"
		column = { data = name }
		column = { data = population }
		column = { data = army_size }
		visible = {
			country_type = location
			this != scope:actor
			is_neighbor_of = scope:actor
			is_rebel_country = no
			capital = { is_discovered_by = scope:actor }
			OR = {
				NOT = { scope:actor = { has_variable_list = tv_wonder_emissary_countries } }
				NOT = {
					scope:actor = {
						is_target_in_variable_list = {
							name = tv_wonder_emissary_countries
							target = root
						}
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = { exists = scope:target }
