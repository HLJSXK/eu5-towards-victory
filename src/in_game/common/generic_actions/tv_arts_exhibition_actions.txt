# Towards Victory — Arts Exhibition generic actions
# Player-facing project actions. International tours are proposed autonomously
# by artists through the quarterly country pulse.

# ── Host Domestic Exhibition ──────────────────────────────────────────────────
# Starts a persistent local exhibition. Requires at least 5 owned artworks,
# costs scaled_gold = 3.5, and asks the player to choose the exhibition location.
tv_host_domestic_exhibition = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 60

	price = price:tv_host_domestic_exhibition_price

	potential = {
		scope:actor = {
			custom_tooltip = {
				text = TV_HAS_ARTS_EXHIBITION_MEMBER_TT
				has_variable = tv_arts_exhibition_member
			}
		}
	}

	allow = {
		custom_tooltip = {
			text = tv_host_domestic_exhibition_requires_art_tt
			scope:actor = {
				any_work_of_art_in_country = { count >= 5 }
			}
		}
		custom_tooltip = {
			text = TV_HAS_LOCAL_EXHIBITION_ACTIVE_NOT_SET_TT
			scope:actor = {
				NOT = { has_variable = tv_local_exhibition_active }
			}
		}
		scope:actor = {
			tv_chief_artist_available_for_domestic_exhibition_trigger = yes
		}
	}

	select_trigger = {
		looking_for_a = location
		source = actor
		target_flag = target
		name = "tv_select_local_exhibition_location"
		none_available_msg_key = "tv_no_local_exhibition_location_available"
		column = { data = name }
		column = { data = population }
		column = { data = rank }
		visible = {
			has_owner = yes
			owner ?= scope:actor
			NOT = { has_building = building_type:tv_local_arts_exhibition }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				tv_start_local_exhibition_effect = yes
			}
		}
	}

	ai_will_do = {
		add = -50
	}
}

tv_local_exhibition_expand = {
	type = owncountry
	sound = UI_action_religion_generic
	ai_tick = monthly
	ai_tick_frequency = 12
	price = price:tv_local_exhibition_expand_price

	potential = {
		scope:actor = {
			custom_tooltip = {
				text = TV_HAS_ARTS_EXHIBITION_MEMBER_TT
				has_variable = tv_arts_exhibition_member
			}
		}
	}

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = TV_HAS_LOCAL_EXHIBITION_ACTIVE_TT
				has_variable = tv_local_exhibition_active
			}
			custom_tooltip = {
				text = tv_local_exhibition_add_artworks_allow_tt
				trigger_if = {
					limit = {
						has_variable = tv_local_exhibition_active
						has_variable = tv_local_exhibition_artworks_committed
						has_variable = tv_local_exhibition_add_art_threshold
					}
					any_work_of_art_in_country = {
						count >= {
							value = {
								value = var:tv_local_exhibition_artworks_committed
								add = var:tv_local_exhibition_add_art_threshold
							}
						}
					}
				}
				trigger_else = { always = no }
			}
			custom_tooltip = {
				text = TV_HAS_LOCAL_EXHIBITION_EXPAND_TARGET_TT
				trigger_if = {
					limit = { has_variable = tv_local_exhibition_active }
					OR = {
						any_in_list = {
							variable = tv_local_exhibition_locations
							any_neighbor_location = {
								has_owner = yes
								owner ?= scope:actor
								NOT = { has_building = building_type:tv_local_arts_exhibition }
							}
						}
						var:tv_local_exhibition_location ?= {
							any_neighbor_location = {
								has_owner = yes
								owner ?= scope:actor
								NOT = { has_building = building_type:tv_local_arts_exhibition }
							}
						}
						var:tv_local_exhibition_secondary_location ?= {
							any_neighbor_location = {
								has_owner = yes
								owner ?= scope:actor
								NOT = { has_building = building_type:tv_local_arts_exhibition }
							}
						}
					}
				}
				trigger_else = { always = no }
			}
		}
	}

	select_trigger = {
		looking_for_a = location
		source = actor
		target_flag = target
		name = "tv_select_local_exhibition_secondary_location"
		none_available_msg_key = "tv_no_local_exhibition_secondary_location_available"
		column = { data = name }
		column = { data = population }
		column = { data = rank }
		visible = {
			has_owner = yes
			owner ?= scope:actor
			NOT = { has_building = building_type:tv_local_arts_exhibition }
			scope:actor = {
				OR = {
					any_in_list = {
						variable = tv_local_exhibition_locations
						any_neighbor_location = {
							this = root
						}
					}
					var:tv_local_exhibition_location ?= {
						any_neighbor_location = {
							this = root
						}
					}
					var:tv_local_exhibition_secondary_location ?= {
						any_neighbor_location = {
							this = root
						}
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				tv_add_local_exhibition_artworks_effect = yes
			}
		}
	}

	ai_will_do = { add = 5 }
}
