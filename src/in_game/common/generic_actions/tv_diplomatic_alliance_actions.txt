# Towards Victory - Diplomatic Alliance member management actions

tv_expel_diplomatic_alliance_member = {
	type = owncountry
	sound = UI_action_diplomacy_generic
	ai_tick = monthly
	ai_tick_frequency = 99999
	potential = {
		scope:actor = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:actor
			}
		}
	}
	allow = {
		scope:actor = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:actor
			}
		}
	}
	select_trigger = {
		looking_for_a = country
		source = actor
		target_flag = target
		name = "tv_select_diplomatic_alliance_member"
		none_available_msg_key = "tv_no_diplomatic_alliance_member_available"
		column = {
			data = name
		}
		visible = {
			NOT = { this = scope:actor }
			is_subject = no
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:actor
			}
		}
		enabled = {
			NOT = { this = scope:actor }
			is_subject = no
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:actor
			}
		}
		show_why_not_enabled = yes
	}
	effect = {
		if = {
			limit = {
				exists = scope:target
				scope:target = { is_subject = no }
			}
			scope:actor = {
				every_international_organizations_member_of = {
					limit = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
						leader_country ?= scope:actor
					}
					remove_country_from_international_organization = scope:target
					tv_diplomatic_alliance_update_member_count_effect = yes
				}
			}
		}
	}
	ai_will_do = { add = -100 }
}

tv_alliance_sublime_empire_subjugate_member = {
	type = owncountry
	sound = UI_action_diplomacy_generic
	ai_tick = monthly
	ai_tick_frequency = 99999
	potential = {
		scope:actor = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_diplomatic_alliance
				leader_country ?= scope:actor
				international_organization_has_policy = policy:tv_sublime_empire_policy
			}
		}
	}
	allow = {
		scope:actor = {
			custom_tooltip = {
				text = TV_ALLIANCE_SUBJUGATION_COHESION_COST_TT
				any_international_organizations_member_of = {
					international_organization_type = international_organization_type:tv_diplomatic_alliance
					leader_country ?= scope:actor
					international_organization_has_policy = policy:tv_sublime_empire_policy
					trigger_if = {
						limit = { exists = scope:target }
						var:tv_alliance_cohesion ?= {
							this >= {
								value = "scope:target.relative_strength(scope:actor)"
								multiply = 50
								add = 10
							}
						}
					}
					trigger_else = {
						var:tv_alliance_cohesion ?= { this >= 10 }
					}
				}
			}
		}
		trigger_if = {
			limit = { exists = scope:target }
			custom_tooltip = {
				text = TV_ALLIANCE_SUBJUGATION_TARGET_MEMBER_TT
				scope:target = {
					NOT = { this = scope:actor }
					is_subject = no
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
						leader_country ?= scope:actor
						international_organization_has_policy = policy:tv_sublime_empire_policy
					}
				}
			}
			custom_tooltip = {
				text = TV_ALLIANCE_SUBJUGATION_MEMBER_TENURE_TT
				scope:actor = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
						leader_country ?= scope:actor
						international_organization_has_policy = policy:tv_sublime_empire_policy
						scope:target = {
							years_in_international_organization = {
								international_organization = prev
								value >= 5
							}
						}
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				every_international_organizations_member_of = {
					limit = {
						international_organization_type = international_organization_type:tv_diplomatic_alliance
						leader_country ?= scope:actor
						international_organization_has_policy = policy:tv_sublime_empire_policy
						scope:target = {
							NOT = { this = scope:actor }
							is_subject = no
							any_international_organizations_member_of = {
								international_organization_type = international_organization_type:tv_diplomatic_alliance
								leader_country ?= scope:actor
							}
							years_in_international_organization = {
								international_organization = prev
								value >= 5
							}
						}
					}
					set_local_variable = {
						name = tv_alliance_subjugation_cost
						value = {
							value = "scope:target.relative_strength(scope:actor)"
							multiply = 50
							add = 10
						}
					}
					if = {
						limit = {
							var:tv_alliance_cohesion ?= { this >= local_var:tv_alliance_subjugation_cost }
						}
						change_variable = { name = tv_alliance_cohesion subtract = local_var:tv_alliance_subjugation_cost }
						clamp_variable = { name = tv_alliance_cohesion min = 0 max = 100 }
						if = {
							limit = {
								NOT = { scope:actor = { has_advance = samanta_advance } }
							}
							scope:target = {
								make_subject_of = {
									target = scope:actor
									type = subject_type:vassal
								}
							}
						}
						else = {
							scope:target = {
								make_subject_of = {
									target = scope:actor
									type = subject_type:samanta
								}
							}
						}
					}
				}
			}
		}
	}
	ai_will_do = { add = -100 }
}
