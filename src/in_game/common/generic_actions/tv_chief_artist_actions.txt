# Towards Victory -- Chief Artist generic action
# Appoints a court artist as Chief Artist for the Arts Exhibition.
# The Chief Artist passively increases International Influence each month
# based on their Art Skill (artist_skill x 0.002 per month).

tv_appoint_chief_artist = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 99999

	potential = {
		scope:actor = {
			has_variable = tv_arts_exhibition_member
		}
	}

	allow = {
		always = yes
	}

	# Select an artist from own court
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = target
		name = "tv_select_chief_artist"
		none_available_msg_key = "tv_no_chief_artist_available"
		column = {
			data = character_info
		}
		visible = {
			is_alive = yes
			is_artist = yes
			is_ruler = no
			is_heir = no
			is_consort = no
			NOT = { has_variable = tv_on_exhibition }
			NOT = { has_variable = tv_is_chief_artist }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }

			# Clear the previous chief artist if one is assigned
			scope:actor = {
				if = {
					limit = { has_variable = tv_chief_artist }
					var:tv_chief_artist ?= {
						remove_variable = tv_is_chief_artist
						remove_variable = tv_chief_artist_country
					}
					remove_variable = tv_chief_artist
					remove_variable = tv_chief_artist_monthly_bonus
				}
			}

			# Mark the new chief artist and record their appointing country
			scope:target = {
				set_variable = { name = tv_is_chief_artist value = 1 }
				set_variable = { name = tv_chief_artist_country value = scope:actor }
			}

			# Store reference on the country; initial bonus computed via chained access
			scope:actor = {
				set_variable = { name = tv_chief_artist value = scope:target }
				set_variable = {
					name = tv_chief_artist_monthly_bonus
					value = {
						value = scope:target.artist_skill
						multiply = 0.002
					}
				}
			}
		}
	}

	ai_will_do = {
		add = -100
	}
}
