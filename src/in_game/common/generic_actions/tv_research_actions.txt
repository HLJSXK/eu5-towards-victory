# ── Request Research Target ────────────────────────────────────────────────────
# Opens the semi-random target selection event.
# Disabled when a cooldown is active or a target is already selected.
tv_request_research_target = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 24

	potential = {
		scope:actor = {
			custom_tooltip = {
				text = TV_HAS_ACADEMY_IO_MEMBER_TT
				has_variable = tv_academy_io_member
			}
		}
	}

	allow = {
		scope:actor = {
			tv_research_can_request_target_trigger = yes
		}
	}

	effect = {
		scope:actor = {
			tv_research_propose_target_effect = yes
		}
	}

	ai_will_do = {
		add = 10
	}
}

# ── Abandon Research Target ────────────────────────────────────────────────────
# Fires confirmation event that clears all progress and sets 12-month cooldown.
tv_abandon_research_target = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = {
			custom_tooltip = {
				text = TV_HAS_ACADEMY_IO_MEMBER_TT
				has_variable = tv_academy_io_member
			}
			tv_research_has_active_target_trigger = yes
		}
	}

	allow = {
		scope:actor = {
			tv_research_has_active_target_trigger = yes
		}
	}

	effect = {
		scope:actor = {
			trigger_event_non_silently = { id = tv_research.11 }
		}
	}

	ai_will_do = {
		add = -100
	}
}

# ── Start Concentrated Research (D) ───────────────────────────────────────────
# Available only when A, B, and C are all complete and D has not yet started.
# Fires a confirmation event; the player may then confirm or cancel.
tv_start_concentrated_research = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 6

	potential = {
		custom_tooltip = {
			text = TV_HAS_ACADEMY_IO_MEMBER_TT
			scope:actor = { has_variable = tv_academy_io_member }
		}
	}

	allow = {
		custom_tooltip = {
			text = tv_start_d_requires_phase1_tt
			scope:actor = { var:tv_research_phase ?= 1 }
		}
		custom_tooltip = {
			text = tv_start_d_requires_a_tt
			scope:actor = { var:tv_rm_subprocess_a ?= { this >= 2 } }
		}
		custom_tooltip = {
			text = tv_start_d_requires_b_tt
			scope:actor = { var:tv_rm_subprocess_b ?= { this >= 2 } }
		}
		custom_tooltip = {
			text = tv_start_d_requires_c_tt
			scope:actor = { var:tv_rm_subprocess_c ?= { this >= 2 } }
		}
	}

	effect = {
		scope:actor = {
			trigger_event_non_silently = { id = tv_research.20 }
		}
	}

	ai_will_do = {
		add = 50
	}
}

# ── Treat Character (Medicine Tab) ────────────────────────────────────────────
# Selects a character with a health condition to begin treatment.
# Disabled while another character is already being treated.
tv_treat_character = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 99999

	potential = {
		scope:actor = {
			custom_tooltip = {
				text = TV_HAS_ACADEMY_IO_MEMBER_TT
				has_variable = tv_academy_io_member
			}
		}
	}

	allow = {
		custom_tooltip = {
			text = tv_treat_character_no_active_patient_tt
			scope:actor = {
				NOT = { has_variable = tv_med_target_char }
			}
		}
		custom_tooltip = {
			text = tv_treat_character_requires_scientist_tt
			scope:actor = {
				has_variable = tv_academy_leader_char
			}
			scope:actor = {
				var:tv_academy_leader_char ?= { is_alive = yes }
			}
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = target
		name = "tv_select_med_target"
		none_available_msg_key = "tv_no_med_target_available"
		column = { data = character_info }
		visible = {
			is_alive = yes
			tv_has_health_condition_trigger = yes
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				set_variable = { name = tv_med_target_char  value = scope:target }
				set_variable = { name = tv_med_progress     value = 0 }
			}
		}
	}

	ai_will_do = {
		add = -100
	}
}
