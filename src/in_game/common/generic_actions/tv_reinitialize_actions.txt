# Towards Victory - Reinitialization Generic Actions
# Used by the Victory Situation panel as a manual recovery button.
# type = owncountry; scope:actor = the country performing the action.

tv_reinitialize_mod = {
	type = owncountry

	sound = UI_action_religion_generic

	ai_tick = monthly
	ai_tick_frequency = 99999

	potential = {
		always = yes
	}

	allow = {
		always = yes
	}

	effect = {
		scope:actor = {
			hidden_effect = {
				tv_diplomatic_alliance_reinitialize_membership_effect = yes
			}

			# Create missing IOs for paths where M1 was already granted but IO was never created.
			# has_variable and var: are in separate limit blocks (nested if) to avoid the
			# var_read_without_outer_scope_guard anti-pattern.

			# Conquest → Governor's House
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_CONQUEST_MILESTONE_TT
						has_variable = tv_conquest_milestone
					}
				}
				if = {
					limit = {
						is_subject = no
						var:tv_conquest_milestone >= 1
					}
					tv_govhouse_create_effect = yes
				}
			}

			# Diplomatic → Diplomatic Alliance
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_DIPLOMATIC_MILESTONE_TT
						has_variable = tv_diplomatic_milestone
					}
				}
				if = {
					limit = {
						var:tv_diplomatic_milestone >= 1
						NOT = {
							any_international_organizations_member_of = {
								international_organization_type = international_organization_type:tv_diplomatic_alliance
							}
						}
					}
					trigger_event_non_silently = { id = tv_diplo_alliance.2 }
				}
			}

			# Cultural → Arts Exhibition
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_CULTURAL_MILESTONE_TT
						has_variable = tv_cultural_milestone
					}
				}
				if = {
					limit = {
						var:tv_cultural_milestone >= 1
						NOT = {
							any_international_organizations_member_of = {
								international_organization_type = international_organization_type:tv_arts_exhibition
							}
						}
					}
					set_variable = { name = tv_arts_exhibition_member value = 1 }
					create_international_organization = {
						type = international_organization_type:tv_arts_exhibition
						add_country_to_international_organization = prev
						set_leader_country = prev
					}
				}
			}

			# Prosperity -> Engineering Department
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_PROSPERITY_MILESTONE_TT
						has_variable = tv_prosperity_milestone
					}
				}
				if = {
					limit = {
						var:tv_prosperity_milestone >= 1
					}
					tv_engineering_department_create_effect = yes
				}
			}

			# Trade -> Trade League
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_TRADE_MILESTONE_TT
						has_variable = tv_trade_milestone
					}
				}
				if = {
					limit = {
						var:tv_trade_milestone >= 1
					}
					tv_trade_league_create_effect = yes
				}
			}

			# Science → Academy of Sciences
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_SCIENCE_MILESTONE_TT
						has_variable = tv_science_milestone
					}
				}
				if = {
					limit = {
						var:tv_science_milestone >= 1
						NOT = {
							any_international_organizations_member_of = {
								international_organization_type = international_organization_type:tv_academy_of_sciences
							}
						}
					}
					if = {
						limit = {
							custom_tooltip = {
								text = TV_HAS_VARIABLE_NOT_SET_TT
								NOT = { has_variable = tv_academy_io_member }
							}
						}
						set_variable = { name = tv_academy_io_member value = 1 }
						set_variable = { name = tv_research_phase value = 0 }
						set_variable = { name = tv_research_interest value = 0 }
						set_variable = { name = tv_research_b_done value = 0 }
						set_variable = { name = tv_rm_subprocess_a value = 0 }
						set_variable = { name = tv_rm_subprocess_b value = 0 }
						set_variable = { name = tv_rm_subprocess_c value = 0 }
						set_variable = { name = tv_rm_subprocess_d_ready value = 0 }
						set_variable = { name = tv_med_progress value = 0 }
						set_variable = { name = tv_med_disp_adm value = 0 }
						set_variable = { name = tv_med_disp_dip value = 0 }
						set_variable = { name = tv_med_disp_mil value = 0 }
					}
					create_international_organization = {
						type = international_organization_type:tv_academy_of_sciences
						hidden_effect = {
							add_country_to_international_organization = prev
							set_leader_country = prev
						}
					}
				}
			}

			# Reassign leader for existing owned Governor's Houses only.
			if = {
				limit = {
					is_subject = no
					has_variable = tv_govhouse_member
				}
				every_international_organizations_member_of = {
					limit = {
						international_organization_type = international_organization_type:tv_govhouse
						OR = {
							leader_country ?= scope:actor
							NOT = { exists = leader_country }
						}
					}
					set_leader_country = scope:actor
					tv_govhouse_sync_supreme_overlord_status_effect = yes
				}
				tv_govhouse_sync_subject_chain_to_owned_house_effect = yes
			}
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_arts_exhibition
				}
				set_leader_country = scope:actor
			}
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_engineering_department
				}
				set_leader_country = scope:actor
			}
			tv_wonder_reinitialize_building_state_effect = yes
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_trade_league
				}
				set_leader_country = scope:actor
			}
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_academy_of_sciences
				}
				set_leader_country = scope:actor
			}
		}
	}

	ai_will_do = {
		add = -100
	}
}
