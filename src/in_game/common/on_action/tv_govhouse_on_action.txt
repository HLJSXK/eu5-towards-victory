# Towards Victory — Governor's House on_action hooks

# Yearly tick: check if Grand General is in training; resolve outcome if active.
tv_govhouse_yearly_pulse = {
	effect = {
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_GOVHOUSE_MEMBER_TT
					has_variable = tv_govhouse_member
				}
				custom_tooltip = {
					text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
					has_variable = tv_conquest_general_char
				}
				var:tv_conquest_general_char ?= {
					is_alive = yes
					custom_tooltip = {
						text = TV_HAS_GG_TRAINING_ACTIVE_TT
						has_variable = tv_gg_training_active
					}
				}
			}
			tv_training_resolve_effect = yes
		}
	}
}

# Death cleanup: if the Grand General dies, clear country variables and notify.
tv_govhouse_char_death = {
	effect = {
		scope:target = {
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_GG_TRAINING_ACTIVE_TT
						has_variable = tv_gg_training_active
					}
				}
				remove_variable = tv_gg_training_active
			}
		}
		# If the deceased was this country's Grand General, clear the reference
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
					has_variable = tv_conquest_general_char
				}
				var:tv_conquest_general_char ?= scope:target
			}
			tv_govhouse_reset_power_consolidation_effect = yes
			remove_variable = tv_conquest_general_char
			# TODO: fire a notification event to inform the player their Grand General has died
		}
	}
}
