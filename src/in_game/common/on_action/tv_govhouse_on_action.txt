# Towards Victory — Governor's House on_action hooks

# Monthly tick: advance Grand General training and resolve it after 12 months.
tv_govhouse_training_monthly_pulse = {
	effect = {
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_GOVHOUSE_MEMBER_TT
					has_variable = tv_govhouse_member
				}
				custom_tooltip = {
					text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
					has_variable = tv_conquest_general_char
				}
				var:tv_conquest_general_char ?= {
					is_alive = yes
					custom_tooltip = {
						text = TV_HAS_GG_TRAINING_ACTIVE_TT
						has_variable = tv_gg_training_active
					}
				}
			}
			var:tv_conquest_general_char ?= {
				change_variable = { name = tv_gg_training_months value = 1 }
			}
			if = {
				limit = {
					var:tv_conquest_general_char ?= {
						var:tv_gg_training_months >= 12
					}
				}
				tv_training_resolve_effect = yes
			}
		}
	}
}

# Death cleanup: if the Grand General dies, clear country variables and notify.
tv_govhouse_char_death = {
	effect = {
		scope:target = {
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_GG_TRAINING_ACTIVE_TT
						has_variable = tv_gg_training_active
					}
				}
				remove_variable = tv_gg_training_active
				remove_variable = tv_gg_training_months
				root = { remove_country_modifier = tv_grand_general_training_modifier }
			}
		}
		# If the deceased was this country's Grand General, clear the reference
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_CONQUEST_GENERAL_CHAR_TT
					has_variable = tv_conquest_general_char
				}
				var:tv_conquest_general_char ?= scope:target
			}
			scope:target = { save_scope_as = tv_dead_grand_general }
			if = {
				limit = {
					scope:target = {
						NOT = { has_variable = tv_gg_training_death_notified }
					}
				}
				trigger_event_non_silently = { id = tv_govhouse.2 }
			}
			tv_govhouse_reset_power_consolidation_effect = yes
			tv_training_clear_effect = yes
			remove_variable = tv_conquest_general_char
		}
	}
}

# Death cleanup: if a regional Governor dies, clear their assignment and notify.
tv_govhouse_governor_death = {
	trigger = {
		scope:target = { has_variable = tv_is_governor }
	}
	effect = {
		save_scope_as = tv_governing_country
		scope:target = {
			save_scope_as = tv_dead_governor
		}
		trigger_event_non_silently = { id = tv_govhouse.1 }
		scope:target = {
			tv_governor_clear_assignment_effect = yes
		}
		remove_list_variable = { name = tv_governor_list target = scope:tv_dead_governor }
		tv_govhouse_refresh_region_list_effect = yes
	}
}
