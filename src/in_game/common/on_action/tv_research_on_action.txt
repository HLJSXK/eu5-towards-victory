# Towards Victory — Research Mechanism Monthly On-Action Hooks
# C and D accumulation are handled by IO variables (monthly_change) and monthly_effect.
# This file handles: B passive check, A random events, and D random events.

# monthly_country_pulse logic is owned by towards_victory_leaderboard.txt to avoid
# EU5 singleton on_action overrides for duplicate effect blocks.

# ── tv_research_a_pulse ────────────────────────────────────────────────────────
# Sub-action for Sub-process A interest collection.
# Trigger gates the whole block; one random_list selects at most one event each month.
tv_research_a_pulse = {
	trigger = {
		custom_tooltip = {
			text = TV_HAS_ACADEMY_IO_MEMBER_TT
			has_variable = tv_academy_io_member
		}
		custom_tooltip = {
			text = TV_HAS_RESEARCH_PHASE_TT
			has_variable = tv_research_phase
		}
		var:tv_research_phase = 1
		NOT = { var:tv_rm_subprocess_a >= 2 }
	}
	effect = {
		random_list = {
			90 = { }
			5 = { trigger_event_non_silently = { id = tv_research.1 } }
			1 = { trigger_event_non_silently = { id = tv_research.2 } }
			1 = { trigger_event_non_silently = { id = tv_research.3 } }
			1 = { trigger_event_non_silently = { id = tv_research.4 } }
			1 = { trigger_event_non_silently = { id = tv_research.5 } }
			1 = { trigger_event_non_silently = { id = tv_research.6 } }
		}
	}
}

# ── tv_research_d_pulse ────────────────────────────────────────────────────────
# Sub-action for Sub-process D random events.
# Each currently eligible event has a 1% monthly weight. The tier-dependent
# chance_of_no_event keeps the total list at 100 while still selecting at most
# one event each month.

	trigger = {
		custom_tooltip = {
			text = TV_HAS_ACADEMY_IO_MEMBER_TT

		var:tv_academy_leader_char ?= { is_alive = yes }

		chance_to_happen = 100

			value = 100

				limit = { tv_chief_scientist_effective_adm_less_than_20_trigger = yes }
				add = -11

			if = {
				limit = {
					tv_chief_scientist_effective_adm_at_least_20_trigger = yes
					tv_chief_scientist_effective_adm_less_than_50_trigger = yes
					any_cabinet_character = { count > 0 }
				}

			}
			if = {
				limit = {
					tv_chief_scientist_effective_adm_at_least_20_trigger = yes
					tv_chief_scientist_effective_adm_less_than_50_trigger = yes
					NOT = { any_cabinet_character = { count > 0 } }

			}
			if = {
				limit = {
					tv_chief_scientist_effective_adm_at_least_50_trigger = yes
					tv_chief_scientist_effective_adm_less_than_80_trigger = yes
					any_cabinet_character = { count > 0 }
				}
				add = -17
			}
			if = {
				limit = {
					tv_chief_scientist_effective_adm_at_least_50_trigger = yes
					tv_chief_scientist_effective_adm_less_than_80_trigger = yes
					NOT = { any_cabinet_character = { count > 0 } }
