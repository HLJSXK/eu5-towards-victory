# Towards Victory — Research Mechanism Monthly On-Action Hooks
# C and D accumulation are handled by IO variables (monthly_change) and monthly_effect.
# This file handles: selection cooldown countdown, B passive check, A random events.

monthly_country_pulse = {
	effect = {
		if = {
			limit = { has_variable = tv_academy_io_member }

			# ── Selection cooldown countdown ───────────────────────────────────
			if = {
				limit = {
					has_variable = tv_research_selection_cd
					var:tv_research_selection_cd >= 1
				}
				change_variable = { name = tv_research_selection_cd add = -1 }
				if = {
					limit = { var:tv_research_selection_cd <= 0 }
					remove_variable = tv_research_selection_cd
				}
			}

			# ── Sub-process B: passive prerequisite check (phase 1 only) ──────
			# Phase 4 stub: tv_research_check_b_trigger = always, so B auto-completes
			# on the first monthly tick after a research target is confirmed.
			if = {
				limit = {
					has_variable = tv_research_phase
					var:tv_research_phase = 1
					NOT = { var:tv_rm_subprocess_b >= 2 }
					tv_research_check_b_trigger = yes
				}
				tv_research_complete_b_effect = yes
			}
		}
	}

	# ── Sub-process A: monthly interest event (random) ─────────────────────────
	# Weight ~25/125 ≈ 20% base monthly chance; event trigger enforces preconditions.
	random_events = {
		100 = 0
		25 = tv_research.1
	}
}
