# Towards Victory — Medicine mechanic monthly on-action sub-actions
# Registered in towards_victory_leaderboard.txt via on_actions = { ... }
# Each has a trigger block that gates execution automatically.

# ── Discovery event pulse ──────────────────────────────────────────────────────
# ~10% monthly chance (approximates dip*0.2% at median dip=50).
# Fires tv_medicine.1 — player must choose a treatment approach.
tv_med_discovery_pulse = {
	trigger = {
		custom_tooltip = {
			text = TV_HAS_ACADEMY_IO_MEMBER_TT
			has_variable = tv_academy_io_member
		}
		custom_tooltip = {
			text = TV_HAS_MED_TARGET_CHAR_TT
			has_variable = tv_med_target_char
		}
		var:tv_med_target_char ?= { is_alive = yes }
		custom_tooltip = {
			text = TV_HAS_ACADEMY_LEADER_CHAR_TT
			has_variable = tv_academy_leader_char
		}
		var:tv_academy_leader_char ?= { is_alive = yes }
	}
	random_events = {
		90 = 0
		10 = tv_medicine.1
	}
}
