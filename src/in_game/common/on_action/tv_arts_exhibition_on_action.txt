# Towards Victory — Arts Exhibition on_action hooks

# When a touring artist dies: notify home country and clean up exhibition state.
# root = country the artist belonged to at death, scope:target = dead character.
tv_arts_exhibition_char_death = {
	effect = {
		scope:target = {
			# Unanswered proposal cleanup
			if = {
				limit = { has_variable = tv_artist_tour_proposal_pending }
				tv_clear_artist_tour_proposal_effect = yes
			}
			# Touring artist cleanup
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_ON_EXHIBITION_TT
						has_variable = tv_on_exhibition
					}
				}
				save_scope_as = dead_touring_artist
				var:tv_exhibition_home_country ?= {
					trigger_event_non_silently = { id = tv_arts_exhibition.6 }
				}
				tv_cleanup_artist_tour_effect = yes
			}
			# Guest artist cleanup for local exhibitions
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_LOCAL_EXHIBITION_GUEST_HOST_TT
						has_variable = tv_local_exhibition_guest_host
					}
				}
				var:tv_local_exhibition_guest_host ?= {
					if = {
						limit = {
							custom_tooltip = {
								text = TV_HAS_LOCAL_EXHIBITION_GUEST_ARTIST_TT
								has_variable = tv_local_exhibition_guest_artist
							}
							var:tv_local_exhibition_guest_artist = scope:target
						}
						remove_variable = tv_local_exhibition_guest_artist
						remove_variable = tv_local_exhibition_guest_country
					}
				}
				remove_character_modifier = busy_modifier
				remove_variable = tv_local_exhibition_guest_home
				remove_variable = tv_local_exhibition_guest_host
			}
			# Arts Exchange participant cleanup
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_ARTS_EXCHANGE_PARTICIPANT_TT
						has_variable = tv_arts_exchange_participant
					}
				}
				save_scope_as = dead_exchange_artist
				var:tv_arts_exchange_host_country ?= {
					remove_list_variable = {
						name = tv_arts_exchange_participants
						target = scope:dead_exchange_artist
					}
					remove_list_variable = {
						name = tv_arts_exchange_foreign_participants
						target = scope:dead_exchange_artist
					}
				}
				remove_variable = tv_arts_exchange_participant
				remove_variable = tv_arts_exchange_foreign_participant
				remove_variable = tv_arts_exchange_host_country
				remove_variable = tv_arts_exchange_auto_invitee
				remove_character_modifier = busy_modifier
			}
		}
		if = {
			limit = {
			has_variable = tv_local_exhibition_host_artist
				var:tv_local_exhibition_host_artist = scope:target
			}
			remove_variable = tv_local_exhibition_host_artist
		}
		# If the dead character was the appointed Chief Artist, notify the player.
		# tv_arts_exhibition_leader_char is cleared in the event option.
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_ARTS_EXHIBITION_LEADER_CHAR_TT
					has_variable = tv_arts_exhibition_leader_char
				}
				var:tv_arts_exhibition_leader_char = scope:target
			}
			trigger_event_non_silently = { id = tv_arts_exhibition.7 }
		}
	}
}

# Quarterly artist-initiated tour proposals. EU5 has no native quarterly
# country pulse, so this callback is registered under monthly_country_pulse and
# gated to January, April, July, and October.
tv_artist_tour_quarterly_pulse = {
	trigger = {
		OR = {
			current_month = 1
			current_month = 4
			current_month = 7
			current_month = 10
		}
		any_international_organizations_member_of = {
			international_organization_type = international_organization_type:tv_arts_exhibition
		}
		any_neighbor_country = {
			any_owned_location = { always = yes }
		}
	}
	effect = {
		save_scope_as = artist_tour_home
		every_character = {
			limit = {
				is_alive = yes
				is_artist = yes
				NOT = { has_character_modifier = busy_modifier }
				NOT = { has_variable = tv_artist_tour_proposal_pending }
				any_work_of_art_by_creator = {
					owner = scope:artist_tour_home
				}
			}
			random_list = {
				99 = {}
				1 = {
					save_scope_as = proposed_artist
					scope:artist_tour_home = {
						random_neighbor_country = {
							limit = { any_owned_location = { always = yes } }
							save_scope_as = proposed_tour_target
						}
					}
					scope:proposed_tour_target = {
						random_owned_location = {
							save_scope_as = proposed_tour_location
						}
					}
					if = {
						limit = {
							exists = scope:proposed_tour_target
							exists = scope:proposed_tour_location
						}
						set_variable = { name = tv_artist_tour_proposal_pending value = 1 }
						clear_variable_list = tv_pending_artist_tour_artworks
						every_work_of_art_by_creator = {
							limit = { owner = scope:artist_tour_home }
							save_scope_as = pending_tour_artwork
							scope:proposed_artist = {
								add_to_variable_list = {
									name = tv_pending_artist_tour_artworks
									target = scope:pending_tour_artwork
								}
							}
						}
						scope:artist_tour_home = {
							trigger_event_non_silently = { id = tv_arts_exhibition.10 days = 1 }
						}
					}
				}
			}
		}
	}
}

# Yearly tick: advance elapsed counter for touring artists currently in this country's court.
# When an artist's elapsed time reaches their duration, return them home via the scripted effect.
tv_arts_exhibition_yearly_pulse = {
	effect = {
		every_character = {
			limit = {
				is_alive = yes
				is_artist = yes
				custom_tooltip = {
					text = TV_HAS_ON_EXHIBITION_TT
					has_variable = tv_on_exhibition
				}
			}
			change_variable = { name = tv_exhibition_elapsed add = 1 }
			if = {
				limit = { var:tv_exhibition_elapsed >= var:tv_exhibition_duration }
				tv_end_artist_exhibition_effect = yes
			}
		}
	}
}

# Monthly tick: advance active Domestic Exhibitions in country scope.
tv_local_exhibition_monthly_pulse = {
	trigger = {
		has_variable = tv_local_exhibition_active
	}
	effect = {
		if = {
			limit = { has_variable = tv_local_exhibition_active }
			change_variable = { name = tv_local_exhibition_duration add = 1 }
			tv_local_exhibition_ensure_location_list_effect = yes
			tv_update_local_exhibition_rating_effect = yes
			if = {
				limit = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:tv_arts_exhibition
						var:tv_local_exhibition_enthusiasm <= 0
					}
				}
				tv_end_local_exhibition_effect = yes
			}
			else = {
				tv_roll_local_exhibition_monthly_events_effect = yes
			}
		}
	}
}

# Monthly tick: advance active Arts Exchanges.
tv_arts_exchange_monthly_pulse = {
	effect = {
		tv_arts_exchange_monthly_effect = yes
	}
}
