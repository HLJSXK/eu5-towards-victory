# Towards Victory — Arts Exhibition on_action hooks

# When a touring artist dies: notify home country and clean up exhibition state.
# root = country the artist belonged to at death, scope:target = dead character.
tv_arts_exhibition_char_death = {
	effect = {
		scope:target = {
			# Touring artist cleanup
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_ON_EXHIBITION_TT
						has_variable = tv_on_exhibition
					}
				}
				save_scope_as = dead_touring_artist
				# Clear marker on the target country before removing the variable
				if = {
					limit = {
						custom_tooltip = {
							text = TV_HAS_EXHIBITION_TARGET_COUNTRY_TT
							has_variable = tv_exhibition_target_country
						}
					}
					var:tv_exhibition_target_country ?= {
						remove_variable = tv_has_visiting_artist
					}
				}
				var:tv_exhibition_home_country ?= {
					trigger_event_non_silently = { id = tv_arts_exhibition.6 }
					# Decrement counter and update capacity penalty
					if = {
						limit = {
							custom_tooltip = {
								text = TV_HAS_DISPATCHED_ARTISTS_COUNT_TT
								has_variable = tv_dispatched_artists_count
							}
							var:tv_dispatched_artists_count >= 1
						}
						change_variable = { name = tv_dispatched_artists_count add = -1 }
					}
					tv_update_tour_capacity_effect = yes
					remove_list_variable = {
						name = tv_dispatched_artists_list
						target = scope:dead_touring_artist
					}
				}
				remove_variable = tv_on_exhibition
				remove_variable = tv_exhibition_home_country
				remove_variable = tv_exhibition_target_country
				remove_variable = tv_exhibition_location
				remove_variable = tv_exhibition_elapsed
				remove_variable = tv_exhibition_duration
				remove_variable = tv_tour_event_1_fired
				remove_variable = tv_tour_event_2_fired
				remove_variable = tv_tour_event_3_fired
				remove_variable = tv_tour_event_4_fired
				remove_variable = tv_tour_event_5_fired
			}
			# Guest artist cleanup for local exhibitions
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_LOCAL_EXHIBITION_GUEST_HOST_TT
						has_variable = tv_local_exhibition_guest_host
					}
				}
				var:tv_local_exhibition_guest_host ?= {
					if = {
						limit = {
							custom_tooltip = {
								text = TV_HAS_LOCAL_EXHIBITION_GUEST_ARTIST_TT
								has_variable = tv_local_exhibition_guest_artist
							}
							var:tv_local_exhibition_guest_artist = scope:target
						}
						remove_variable = tv_local_exhibition_guest_artist
						remove_variable = tv_local_exhibition_guest_country
					}
				}
				remove_variable = tv_local_exhibition_guest_home
				remove_variable = tv_local_exhibition_guest_host
			}
			# Arts Exchange participant cleanup
			if = {
				limit = {
					custom_tooltip = {
						text = TV_HAS_ARTS_EXCHANGE_PARTICIPANT_TT
						has_variable = tv_arts_exchange_participant
					}
				}
				save_scope_as = dead_exchange_artist
				var:tv_arts_exchange_host_country ?= {
					remove_list_variable = {
						name = tv_arts_exchange_participants
						target = scope:dead_exchange_artist
					}
					remove_list_variable = {
						name = tv_arts_exchange_foreign_participants
						target = scope:dead_exchange_artist
					}
				}
				remove_variable = tv_arts_exchange_participant
				remove_variable = tv_arts_exchange_foreign_participant
				remove_variable = tv_arts_exchange_host_country
				remove_variable = tv_arts_exchange_auto_invitee
			}
		}
		# If the dead character was the appointed Chief Artist, notify the player.
		# tv_arts_exhibition_leader_char is cleared in the event option.
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_ARTS_EXHIBITION_LEADER_CHAR_TT
					has_variable = tv_arts_exhibition_leader_char
				}
				var:tv_arts_exhibition_leader_char = scope:target
			}
			trigger_event_non_silently = { id = tv_arts_exhibition.7 }
		}
	}
}

# Yearly tick: advance elapsed counter for touring artists currently in this country's court.
# When an artist's elapsed time reaches their duration, return them home via the scripted effect.
tv_arts_exhibition_yearly_pulse = {
	effect = {
		every_character = {
			limit = {
				is_alive = yes
				is_artist = yes
				custom_tooltip = {
					text = TV_HAS_ON_EXHIBITION_TT
					has_variable = tv_on_exhibition
				}
			}
			change_variable = { name = tv_exhibition_elapsed add = 1 }
			if = {
				limit = { var:tv_exhibition_elapsed >= var:tv_exhibition_duration }
				tv_end_artist_exhibition_effect = yes
			}
		}
	}
}

# Monthly tick: advance active Arts Exchanges.
tv_arts_exchange_monthly_pulse = {
	effect = {
		tv_arts_exchange_monthly_effect = yes
	}
}
