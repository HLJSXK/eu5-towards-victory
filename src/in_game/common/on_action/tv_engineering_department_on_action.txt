# Towards Victory - Engineering Department on_action hooks

on_game_start = {
	on_actions = {
		tv_wonder_initialize_index_on_game_start
		tv_wonder_initialize_location_display_state_on_game_start
	}
}

on_game_load = {
	on_actions = {
		tv_wonder_initialize_index_on_game_load
		tv_wonder_initialize_location_display_state_on_game_load
	}
}

tv_wonder_initialize_index_on_game_start = {
	effect = {
		hidden_effect = {
			tv_wonder_index_rebuild_global_maps_effect = yes
			if = {
				limit = { has_global_variable_map = tv_wonder_id_to_is_unique }
				every_country = {
					tv_wonder_index_refresh_country_cache_effect = yes
				}
			}
		}
	}
}

}

# Death cleanup: if the Great Engineer dies, clear country variable and notify.
# root = country the character belonged to, scope:target = dead character.
tv_engineering_department_char_death = {
	effect = {
		if = {
			limit = {
				custom_tooltip = {
					text = TV_HAS_GREAT_ENGINEER_CHAR_TT
					has_variable = tv_great_engineer_char
				}
				var:tv_great_engineer_char ?= scope:target
			}
			trigger_event_non_silently = { id = tv_engineering_department.10 }
		}
	}
}

tv_engineering_department_concept_monthly_pulse = {
	trigger = {
		has_variable = tv_engineering_department_member
		has_variable = tv_wonder_concept_in_progress
		NOT = { has_variable = tv_wonder_concept_completion_pending }
		tv_wonder_in_conception_trigger = yes
		tv_wonder_has_great_engineer_trigger = yes
	}
	effect = {
		hidden_effect = {
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_engineering_department
					var:tv_wonder_concept_progress >= 100
					prev = { NOT = { has_variable = tv_wonder_concept_completion_pending } }
				}
				prev = {
					set_variable = { name = tv_wonder_concept_completion_pending value = 1 }
					trigger_event_non_silently = { id = tv_engineering_department.30 }
				}
			}
		}
	}
}
