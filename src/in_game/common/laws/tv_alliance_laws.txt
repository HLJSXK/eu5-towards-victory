# LAW 1: Court Protocol & Legitimacy (宫廷礼仪与法理认同)

tv_court_protocol = {
	type = international_organization
	law_category = diplomacy
	potential = {
		international_organization_type = international_organization_type:tv_diplomatic_alliance
	}
	allow = {
		always = yes
	}
	locked = {
	}
	has_levels = yes
	custom_tags = { forbids_no_policy }

	# Level 1 — Ad-hoc Envoys (initial, no effect)
	tv_adhoc_envoys_policy = {
		level = 1
		wants_propose_policy = {
			subtract = {
				desc = "TV_ALREADY_HAVE_BETTER"
				value = 100
			}
		}
		wants_this_policy_bias = {
			add = {
				desc = "TV_DEFAULT_POLICY"
				value = 10
			}
		}
		wants_keep_policy = {
			add = {
				desc = "TV_STATUS_QUO"
				value = 20
			}
		}
	}

	# Level 2 — Resident Ambassadors (cohesion ≥ 25)
	tv_resident_ambassadors_policy = {
		level = 2
		allow = {
			international_organization_has_policy = policy:tv_adhoc_envoys_policy
			scope:recipient = {
				has_variable = tv_alliance_cohesion
				var:tv_alliance_cohesion >= 25
			}
		}
		on_activate = {
			scope:recipient = {
				change_variable = { name = tv_alliance_cohesion add = -25 }
				change_variable = { name = tv_alliance_tier add = 1 }
				# Sync tier to leader country
				leader_country ?= {
					set_variable = { name = tv_alliance_tier value = scope:recipient.var:tv_alliance_tier }
				}
			}
		}
		country_modifier = {
			diplomatic_reputation = 1
		}
		wants_propose_policy = {
			add = {
				desc = "TV_COHESION_REQUIREMENT"
				value = 20
			}
			if = {
				limit = {
					scope:recipient = {
						OR = {
							NOT = { has_variable = tv_alliance_cohesion }
							var:tv_alliance_cohesion < 25
						}
					}
				}
				subtract = {
					desc = "TV_INSUFFICIENT_COHESION"
					value = 1000
				}
			}
		}
		wants_this_policy_bias = {
			add = {
				desc = "TV_UPGRADE_DESIRE"

			}
		}
		wants_keep_policy = {
			add = {
				desc = "TV_STATUS_QUO"

			}
		}
	}
