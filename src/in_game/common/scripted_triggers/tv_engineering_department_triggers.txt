tv_wonder_location_is_city_trigger = {
	trigger_if = {
		limit = { is_land = yes }
		location_rank = location_rank:city
	}
	trigger_else = { always = no }
}

tv_wonder_location_is_town_trigger = {
	trigger_if = {
		limit = { is_land = yes }
		location_rank = location_rank:town
	}
	trigger_else = { always = no }
}

tv_wonder_location_is_city_or_town_trigger = {
	OR = {
		tv_wonder_location_is_city_trigger = yes
		tv_wonder_location_is_town_trigger = yes

}
