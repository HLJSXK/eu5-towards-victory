# Towards Victory — Research Mechanism Subprocess Triggers
# All triggers are country-scoped unless noted.

# ── Sub-process A completion check ────────────────────────────────────────────
# A is complete when interest reaches 5 via monthly random events.
tv_research_check_a_trigger = {
	var:tv_research_interest ?= { this >= 5 }
}

# ── Sub-process B completion check ────────────────────────────────────────────
# Institution-root targets require at least one owned location that satisfies the
# matching institution's plausible birthplace conditions, with geographic and
# existing-institution gates removed. Non-institution-root targets keep the old
# automatic completion.
tv_research_check_b_trigger = {
	# 1: Printing Press -> printing_press institution
	trigger_if = {
		limit = { var:tv_research_target ?= 1 }
		any_owned_location = {
			owner ?= {
				produced_in_country:paper > 5
			}
			num_pop_type:burghers > 1
			has_building = building_type:university
			owner = {
				societal_value:traditionalist_vs_innovative > 20
			}
		}
	}
	# 7: Scientific Revolution -> scientific_revolution institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 7 }
		any_owned_location = {
			has_building = building_type:university
			average_location_literacy > 30
			owner = {
				societal_value:traditionalist_vs_innovative >= 20
			}
		}
	}
	# 8: Manufactories -> manufactories institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 8 }
		any_owned_location = {
			num_pop_type:burghers > 20
			total_building_levels > 80
			owner = {
				societal_value:capital_economy_vs_traditional_economy < -30
			}
		}
	}
	# 9: Industrialization -> industrialization institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 9 }
		any_owned_location = {
			num_pop_type:burghers > 20
			total_building_levels > 120
			owner = {
				societal_value:capital_economy_vs_traditional_economy < -30
			}
		}
	}
	# 11: Renaissance -> renaissance institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 11 }
		any_owned_location = {
			has_owner = yes
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
				has_building_with_at_least_one_level = university
			}
			custom_tooltip = {
				text = dominant_culture_is_owners_culture
				dominant_culture = owner.culture
			}
		}
	}
	# 12: Banking -> banking institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 12 }
		any_owned_location = {
			has_owner = yes
			num_pop_type:burghers > 1.0
			custom_tooltip = {
				text = dominant_culture_is_owners_culture
				dominant_culture = owner.culture
			}
			owner = {
				societal_value:capital_economy_vs_traditional_economy < -50
			}
		}
	}
	# 13: Professional Armies -> professional_armies institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 13 }
		any_owned_location = {
			has_owner = yes
			custom_tooltip = {
				text = dominant_culture_is_owners_culture
				dominant_culture = owner.culture
			}
			modifier:local_manpower > 0
			owner = {
				societal_value:quality_vs_quantity < -10
			}
		}
	}
	# 14: New World -> new_world institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 14 }
		any_owned_location = {
			has_owner = yes
			is_port = yes
			num_pop_type:burghers >= 2.0
			dominant_culture = owner.culture
		}
	}
	# 15: Pike and Shot -> pike_and_shot institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 15 }
		any_owned_location = {
			owner ?= {
				army_tradition > 20
				has_advance = gunpowder_advance
				has_advance = professional_armies_advance
			}
			modifier:local_manpower > 0
			owner = {
				societal_value:land_vs_naval < -20
			}
		}
	}
	# 16: Confessionalism -> confessionalism institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 16 }
		any_owned_location = {
			has_owner = yes
			has_building_with_at_least_one_level = printing_press_shop
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
			dominant_religion = owner.religion
			dominant_religion = religion:catholic
			owner = {
				societal_value:spiritualist_vs_humanist < -25
			}
		}
	}
	# 17: Global Trade -> global_trade institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 17 }
		any_owned_location = {
			is_market_center = yes
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
			market = {
				total_goods_value_traded > 500
			}
		}
	}
	# 18: Artillery Institution -> artillery_institution institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 18 }
		any_owned_location = {
			has_building_with_at_least_one_level = gun_smith
			has_building_with_at_least_one_level = tools_guild
			owner = {
				societal_value:offensive_vs_defensive < -20
			}
		}
	}
	# 19: Military Revolution -> military_revolution institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 19 }
		any_owned_location = {
			is_capital = yes
			population > 30
			owner = {
				yearly_manpower > 2
				societal_value:offensive_vs_defensive < -20
			}
		}
	}
	# 20: Enlightenment -> enlightenment institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 20 }
		any_owned_location = {
			has_building_with_at_least_one_level = university
			average_location_literacy > 30
			owner = {
				societal_value:traditionalist_vs_innovative > 20
			}
		}
	}
	# 21: Levee en Masse -> levee_en_masse institution
	trigger_else_if = {
		limit = { var:tv_research_target ?= 21 }
		any_owned_location = {
			is_capital = yes
			population > 200
			owner = {
				societal_value:quality_vs_quantity > 0
			}
		}
	}
	trigger_else = {
		always = yes
	}
}

# ── Sub-process D start eligibility ───────────────────────────────────────────
# D is available only after A, B, and C are all complete, and the cycle is in phase 1.
tv_research_can_start_d_trigger = {
	var:tv_research_phase ?= 1
	var:tv_rm_subprocess_a ?= { this >= 2 }
	var:tv_rm_subprocess_b ?= { this >= 2 }
	var:tv_rm_subprocess_c ?= { this >= 2 }
}

# ── Chief Scientist effective ADM tiers ──────────────────────────────────────
# Country-scoped. Effective ADM = real ADM + Great Scientist trait bonus while
# the character is appointed as the Academy Chief Scientist.
tv_chief_scientist_effective_adm_at_least_20_trigger = {
	has_variable = tv_academy_leader_char
	var:tv_academy_leader_char ?= {
		is_alive = yes
		OR = {
			adm >= 20
			has_trait = tv_great_scientist_1
			has_trait = tv_great_scientist_2
			has_trait = tv_great_scientist_3
		}
	}
}

tv_chief_scientist_effective_adm_at_least_50_trigger = {
	has_variable = tv_academy_leader_char
	var:tv_academy_leader_char ?= {
		is_alive = yes
		OR = {
			adm >= 50
			AND = {
				adm >= 30
				has_trait = tv_great_scientist_1
			}
			AND = {
				adm >= 10
				has_trait = tv_great_scientist_2
			}
			has_trait = tv_great_scientist_3
		}
	}
}

tv_chief_scientist_effective_adm_at_least_80_trigger = {
	has_variable = tv_academy_leader_char
	var:tv_academy_leader_char ?= {
		is_alive = yes
		OR = {
			adm >= 80
			AND = {
				adm >= 60
				has_trait = tv_great_scientist_1
			}
			AND = {
				adm >= 40
				has_trait = tv_great_scientist_2
			}
			AND = {
				adm >= 20
				has_trait = tv_great_scientist_3
			}
		}
	}
}

tv_chief_scientist_effective_adm_less_than_20_trigger = {
	NOT = { tv_chief_scientist_effective_adm_at_least_20_trigger = yes }
}

tv_chief_scientist_effective_adm_less_than_50_trigger = {
	NOT = { tv_chief_scientist_effective_adm_at_least_50_trigger = yes }
}

tv_chief_scientist_effective_adm_less_than_80_trigger = {
	NOT = { tv_chief_scientist_effective_adm_at_least_80_trigger = yes }
}

# ── Target request eligibility ─────────────────────────────────────────────────
# Player may request a research target when: member of Academy IO, no target selected,
# and at least one Frontier advance is unresearched.

	custom_tooltip = {
		text = TV_HAS_ACADEMY_IO_MEMBER_TT

		text = TV_HAS_VARIABLE_NOT_SET_TT

	}
