# Towards Victory — Research Mechanism Subprocess Triggers
# All triggers are country-scoped unless noted.

# ── Sub-process A completion check ────────────────────────────────────────────
# A is complete when interest reaches 5 via monthly random events.
tv_research_check_a_trigger = {
	var:tv_research_interest ?= { this >= 5 }
}

# ── Sub-process B completion check ────────────────────────────────────────────
# Phase 4 stub: always = yes while phase=1 (B auto-completes within the first monthly tick).
# Phase 5 will replace this with per-advance prerequisite checks (required_advance + required_building).
tv_research_check_b_trigger = {
	# Phase 4 stub — replace in Phase 5 with:
	#   has_researched_advance = advance:<required_advance>
	#   has_building_in_capital = building:<required_building>
	always = yes
}

# ── Sub-process C completion check ────────────────────────────────────────────
# C is complete when tv_rm_subprocess_c display flag = 2.
# The actual accumulation is tracked as the IO variable tv_research_c_progress (0-100);
# the IO monthly_effect sets tv_rm_subprocess_c = 2 when c_progress >= 100.
tv_research_check_c_trigger = {
	var:tv_rm_subprocess_c ?= { this >= 2 }
}

# ── Sub-process D start eligibility ───────────────────────────────────────────
# D is available only after A, B, and C are all complete, and the cycle is in phase 1.
tv_research_can_start_d_trigger = {
	var:tv_research_phase ?= 1
	var:tv_rm_subprocess_a ?= { this >= 2 }
	var:tv_rm_subprocess_b ?= { this >= 2 }
	var:tv_rm_subprocess_c ?= { this >= 2 }
}

# ── Chief Scientist effective ADM tiers ──────────────────────────────────────
# Country-scoped. Effective ADM = real ADM + Great Scientist trait bonus while
# the character is appointed as the Academy Chief Scientist.
tv_chief_scientist_effective_adm_at_least_20_trigger = {
	has_variable = tv_academy_leader_char
	var:tv_academy_leader_char ?= {
		is_alive = yes
		OR = {
			adm >= 20
			has_trait = tv_great_scientist_1
			has_trait = tv_great_scientist_2
			has_trait = tv_great_scientist_3
		}
	}
}

tv_chief_scientist_effective_adm_at_least_50_trigger = {
	has_variable = tv_academy_leader_char
	var:tv_academy_leader_char ?= {
		is_alive = yes
		OR = {
			adm >= 50
			AND = {
				adm >= 30
				has_trait = tv_great_scientist_1
			}
			AND = {
				adm >= 10
				has_trait = tv_great_scientist_2
			}
			has_trait = tv_great_scientist_3
		}
	}
}

tv_chief_scientist_effective_adm_at_least_80_trigger = {
	has_variable = tv_academy_leader_char
	var:tv_academy_leader_char ?= {
		is_alive = yes
		OR = {
			adm >= 80
			AND = {
				adm >= 60
				has_trait = tv_great_scientist_1
			}
			AND = {
				adm >= 40
				has_trait = tv_great_scientist_2
			}
			AND = {
				adm >= 20
				has_trait = tv_great_scientist_3
			}
		}
	}
}

tv_chief_scientist_effective_adm_less_than_20_trigger = {
	NOT = { tv_chief_scientist_effective_adm_at_least_20_trigger = yes }
}

tv_chief_scientist_effective_adm_less_than_50_trigger = {
	NOT = { tv_chief_scientist_effective_adm_at_least_50_trigger = yes }
}

tv_chief_scientist_effective_adm_less_than_80_trigger = {
	NOT = { tv_chief_scientist_effective_adm_at_least_80_trigger = yes }
}

# ── Target request eligibility ─────────────────────────────────────────────────
# Player may request a research target when: member of Academy IO, no target selected,
# selection cooldown is not active, and at least one Frontier advance is unresearched.
tv_research_can_request_target_trigger = {
	custom_tooltip = {
		text = TV_HAS_ACADEMY_IO_MEMBER_TT
		has_variable = tv_academy_io_member
	}
	custom_tooltip = {
		text = TV_HAS_VARIABLE_NOT_SET_TT
		NOT = { has_variable = tv_research_target }
	}
	custom_tooltip = {
		text = TV_HAS_VARIABLE_NOT_SET_TT
		NOT = { has_variable = tv_research_selection_cd }
	}
	OR = {
		tv_research_has_unresearched_age_1_frontier_trigger = yes
		tv_research_has_unresearched_age_2_frontier_trigger = yes
		tv_research_has_unresearched_age_3_frontier_trigger = yes
		tv_research_has_unresearched_age_4_frontier_trigger = yes
		tv_research_has_unresearched_age_5_frontier_trigger = yes
		tv_research_has_unresearched_age_6_frontier_trigger = yes
	}
}

# ── Chief Scientist busy-state helpers ───────────────────────────────────────
# Country-scoped. Used to make the Chief Scientist's long-running tasks mutually
# exclusive and to show the player which current task is blocking a new one.
tv_chief_scientist_busy_with_concentrated_research_trigger = {
	tv_research_d_in_progress_trigger = yes
}

tv_chief_scientist_busy_with_law_change_trigger = {
	any_international_organizations_member_of = {
		international_organization_type = international_organization_type:tv_academy_of_sciences
		resolution_is_active = resolution:policy_vote
	}
}
