# Towards Victory - Trade League scripted triggers
# All triggers are country-scoped unless noted.

# Grand Merchant busy-state helpers. Trade Chain is currently represented only
# as a busy-state flag; Commercial Intelligence, monopoly work, and law votes
# use the same surface so policy changes cannot overlap with other Grand
# Merchant work.
tv_grand_merchant_busy_with_law_change_trigger = {
	any_international_organizations_member_of = {
		international_organization_type = international_organization_type:tv_trade_league
		resolution_is_active = resolution:policy_vote
	}
}

tv_grand_merchant_busy_with_trade_chain_trigger = {
	has_variable = tv_trade_chain_active
}

tv_grand_merchant_busy_with_commercial_intelligence_trigger = {
	has_variable = tv_trade_commercial_intelligence_active
}

tv_grand_merchant_available_for_law_change_trigger = {
	trigger_if = {
		limit = { tv_grand_merchant_busy_with_trade_chain_trigger = yes }
		custom_description = {
			text = tv_grand_merchant_busy_with_trade_chain_tt
			always = no
		}
	}
	trigger_else_if = {
		limit = { tv_grand_merchant_busy_with_commercial_intelligence_trigger = yes }
		custom_description = {
			text = tv_grand_merchant_busy_with_commercial_intelligence_tt
			always = no
		}
	}
	trigger_else_if = {
		limit = { tv_grand_merchant_busy_with_law_change_trigger = yes }
		custom_description = {
			text = tv_grand_merchant_busy_with_law_change_tt
			always = no
		}
	}
	trigger_else = {
		always = yes
	}
}
