		has_trait = maimed
		has_trait = disfigured
		has_trait = scarred
		has_trait = hunchback
		has_trait = pockmarked_trait
		has_trait = smallpox_trait
		has_trait = bubonic_plague_trait
		has_trait = sickly
	}
}
