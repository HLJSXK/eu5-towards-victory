# Towards Victory — Academy of Sciences building chain
# Five buildings gate the five Scientific Victory milestones.
# All five are visible in every capital from game start.
# Construction requires reaching the corresponding Science Score threshold
# (auto-parsed score variables need custom_tooltip — see allow blocks below).
# Verification — Step 3, Reference: capital_buildings.txt:17-56, Quote: "location_potential = { is_capital = yes }"
# Verification — Step 3, Reference: capital_buildings.txt:15, Quote: "build_time = large_capital_build_time"
# Verification — Step 3, Reference: culture_buildings.txt:1-48, Quote: "category = cultural_category", "construction_demand = university_construction"
# Verification — Step 3, Reference: unique_buildings.txt:516-524, Quote: "location.owner = { add_opinion = { ... } }" (on_built owner access)
# Verification — Step 3, Reference: estate_buildings.txt (from explorer), Quote: "scope:actor = { ... }" in allow block

tv_academy_of_sciences_1 = {
	is_foreign = no
	pop_type = clergy
	max_levels = 1

	category = cultural_category
	price = tv_academy_1_price

	town = yes
	city = yes
	megalopolis = yes
	rural_settlement = no

	important_for_AI = yes
	automation_build_allowed = yes

	location_potential = {
		is_capital = yes
	}

	allow = {
		custom_tooltip = {
			text = TV_ACADEMY_SCORE_1_TT
			scope:actor = {
				has_variable = tv_science_score
				var:tv_science_score >= 100
			}
		}
	}

	build_time = large_capital_build_time
	construction_demand = tv_academy_1_construction

	possible_production_methods = {
		tv_academy_maintenance
	}

	on_built = {
		hidden_effect = {
			location.owner = {
				set_variable = { name = tv_academy_of_sciences_level value = 1 }
				tv_check_science_milestones_effect = yes
				trigger_event_non_silently = { id = tv_academy.1 }
			}
		}
	}
}
