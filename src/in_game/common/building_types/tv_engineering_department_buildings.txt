tv_wonder_labor_camp = {
	is_foreign = no
	pop_type = laborers
	max_levels = {
		add = 5
		if = {
			limit = { scope:owner = { has_variable = tv_wonder_scale_tier } }
			add = {
				value = scope:owner.var:tv_wonder_scale_tier
				subtract = 1
				multiply = 5
			}
		}
		max = 30
	}
	employment_size = 1

	category = infrastructure_category
	price = tv_wonder_labor_camp_price

	town = yes
	city = yes
	megalopolis = yes
	rural_settlement = yes

	important_for_AI = no
	automation_build_allowed = no

	country_potential = {
		has_variable = tv_engineering_department_member
		tv_wonder_debate_complete_trigger = yes
		NOT = { has_variable = tv_wonder_construction_complete }
	}

	location_potential = {
		owner ?= {
			var:tv_wonder_site ?= prev
		}
	}

	allow = {
		owner = scope:actor
	}

	build_time = village_build_time
	construction_demand = tv_wonder_labor_camp_construction

	modifier = {
		local_migration_attraction = 2
	}

	possible_production_methods = {
		tv_wonder_labor_camp_maintenance
	}
}
