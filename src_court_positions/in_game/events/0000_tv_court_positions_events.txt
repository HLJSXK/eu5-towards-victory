# A disease trait is checked by the Court Positions monthly callback. The
# country keeps the selected patient in a scope variable for the delayed event.
tv_court_physician.10 = {
	type = country_event
	title = tv_court_physician.10.t
	desc = tv_court_physician.10.desc
	outcome = neutral
	trigger = {
		has_variable = tv_court_physician_pending_patient
		var:tv_court_physician_pending_patient ?= {
			is_alive = yes
			OR = {
				has_trait = sickly
				has_trait = smallpox_trait
				has_trait = bubonic_plague_trait
			}
		}
	}

	option = {
		name = tv_court_physician.10.a
		trigger = { NOT = { has_variable = tv_court_physician_character } }
		change_gold_effect = { scale = -1 }
		trigger_event_non_silently = { id = tv_court_physician.11 days = 30 }
	}
	option = {
		name = tv_court_physician.10.b
		trigger = { NOT = { has_variable = tv_court_physician_character } }
		remove_variable = tv_court_physician_pending_patient
	}
	option = {
		name = tv_court_physician.10.c
		trigger = { has_variable = tv_court_physician_character }
		remove_variable = tv_court_physician_pending_patient
	}
	option = {
		name = tv_court_physician.10.d
		trigger = { has_variable = tv_court_physician_character }
			set_variable = { name = tv_court_physician_treatment_mode value = 2 }
			set_variable = { name = tv_court_physician_treatment_result value = 0 }
			# Administrative skill gives up to 50% success and up to 25% great success.
			random = {
				chance = 0
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 20 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 40 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 60 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 80 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 100 } }
				set_variable = { name = tv_court_physician_treatment_result value = 1 }
			}
			if = {
				limit = { var:tv_court_physician_treatment_result = 1 }
				random = {
					chance = 50
					set_variable = { name = tv_court_physician_treatment_result value = 2 }
				}
			}
		trigger_event_non_silently = { id = tv_court_physician.13 days = 30 }
	}
	option = {
		name = tv_court_physician.10.e
		trigger = { has_variable = tv_court_physician_character }
			set_variable = { name = tv_court_physician_treatment_mode value = 3 }
			set_variable = { name = tv_court_physician_treatment_result value = 0 }
			random = {
				chance = 0
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 20 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 40 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 60 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 80 } }
				modifier = { add = 10 var:tv_court_physician_character ?= { adm >= 100 } }
				set_variable = { name = tv_court_physician_treatment_result value = 1 }
			}
		trigger_event_non_silently = { id = tv_court_physician.13 days = 30 }
	}
}

		if = {
			limit = { var:tv_court_physician_treatment_result = 0 var:tv_court_physician_treatment_mode = 2 }
			var:tv_court_physician_pending_patient ?= { add_character_modifier = { modifier = tv_court_physician_treatment_bad years = 5 mode = add_and_extend } }
		}
		remove_variable = tv_court_physician_pending_patient
		remove_variable = tv_court_physician_treatment_mode

	}

namespace = tv_court_tutor

# Action 1: educate children — ability improvement.
tv_court_tutor.1 = {
	type = country_event
	title = tv_court_tutor.1.t
	desc = tv_court_tutor.1.d
	outcome = positive
	trigger = {
		has_variable = tv_court_tutor_character
		var:tv_court_tutor_character ?= { is_alive = yes }
		var:tv_court_tutor_action = 1
		NOT = { has_variable = tv_court_tutor_event_pending }
		any_character = {
			is_alive = yes
			age_in_years < 15
			owner = root
		}
	}
	immediate = {
		set_variable = { name = tv_court_tutor_event_pending value = 1 }
		random_character = {
			limit = { is_alive = yes age_in_years < 15 owner = root }
			save_scope_as = tv_court_tutor_target_child
		}
		random_list = {
			1 = { set_variable = { name = tv_court_tutor_ability_field value = 1 } }
			1 = { set_variable = { name = tv_court_tutor_ability_field value = 2 } }
			1 = { set_variable = { name = tv_court_tutor_ability_field value = 3 } }
		}
	}
	option = {
		name = tv_court_tutor.1.a
		trigger = {
			exists = scope:tv_court_tutor_target_child
			scope:tv_court_tutor_target_child = { is_alive = yes age_in_years < 15 owner = root }
		}
		change_gold_effect = { scale = -1 }
		if = {
			limit = { var:tv_court_tutor_ability_field = 1 }
			scope:tv_court_tutor_target_child = { add_adm = { value = 5 max = 100 } }
		}
		if = {
			limit = { var:tv_court_tutor_ability_field = 2 }
			scope:tv_court_tutor_target_child = { add_dip = { value = 5 max = 100 } }
		}
		if = {
			limit = { var:tv_court_tutor_ability_field = 3 }
			scope:tv_court_tutor_target_child = { add_mil = { value = 5 max = 100 } }
		}
		hidden_effect = {
			remove_variable = tv_court_tutor_ability_field
			remove_variable = tv_court_tutor_event_pending
		}
	}
	option = {
		name = tv_court_tutor.1.b
		hidden_effect = {
			remove_variable = tv_court_tutor_ability_field
			remove_variable = tv_court_tutor_event_pending
		}
	}
}

				has_trait = child_idiot
				has_trait = child_slow
				has_trait = child_intelligent
				has_trait = child_gregarious
				has_trait = child_rowdy
				has_trait = child_ambitious
				has_trait = child_gallant
				has_trait = child_shrewd
				has_trait = child_gifted
			}
		}
	}
	immediate = {
		set_variable = { name = tv_court_tutor_event_pending value = 1 }

					has_trait = child_gregarious

		}
		scope:tv_court_tutor_target_child = {
			remove_variable = tv_court_tutor_trait_upgrade
			if = {
				limit = { has_trait = child_idiot }
				set_variable = { name = tv_court_tutor_trait_upgrade value = 1 }
			}

				limit = { has_trait = child_slow }

					1 = { set_variable = { name = tv_court_tutor_trait_upgrade value = 2 } }

				scope:tv_court_tutor_target_child = {
					is_alive = yes
					age_in_years < 15

			}

# Action 2: educate the court — weighted adult-courtier ability improvement.
tv_court_tutor.3 = {
	type = country_event
	title = tv_court_tutor.3.t
	desc = tv_court_tutor.3.d
	outcome = positive
	trigger = {
		has_variable = tv_court_tutor_character
		var:tv_court_tutor_character ?= { is_alive = yes }
		var:tv_court_tutor_action = 2
		NOT = { has_variable = tv_court_tutor_event_pending }
		any_character = { is_alive = yes is_adult = yes owner = root }
	}
	immediate = {
		set_variable = { name = tv_court_tutor_event_pending value = 1 }
		random_list = {
			50 = {
				ruler ?= {
					if = { limit = { is_alive = yes is_adult = yes owner = root } save_scope_as = tv_court_tutor_target_courtier }
				}
			}
			25 = {
				heir ?= {
					if = { limit = { is_alive = yes is_adult = yes owner = root } save_scope_as = tv_court_tutor_target_courtier }
				}
			}
			10 = {
				ruler ?= {
					random_child = {
						limit = { is_alive = yes is_adult = yes owner = root is_heir = no }
						save_scope_as = tv_court_tutor_target_courtier
					}
				}
			}
			10 = {
				random_character = {
					limit = { is_alive = yes is_adult = yes owner = root in_cabinet = yes is_ruler = no is_heir = no }
					save_scope_as = tv_court_tutor_target_courtier
				}
			}
			5 = {
				random_character = {
					limit = {
						is_alive = yes
						is_adult = yes
						owner = root
						is_ruler = no
						is_heir = no
						in_cabinet = no
						NOT = { is_child_of = root.ruler }
					}
					save_scope_as = tv_court_tutor_target_courtier
				}
			}
		}
		if = {
