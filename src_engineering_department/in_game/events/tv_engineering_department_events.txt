# One-day delayed conception skip. The phase snapshot is captured when
# conception starts, so changing CMM afterwards cannot affect this project.
tv_engineering_department.31 = {
	type = country_event
	title = empty_text
	desc = empty_text
	hidden = yes
	outcome = neutral

	immediate = {
		if = {
			limit = {
				has_variable = tv_wonder_cmm_skip_concept
				has_variable = tv_wonder_concept_in_progress
				var:tv_wonder_stage ?= 0
			}
			every_international_organizations_member_of = {
				limit = {
					international_organization_type = international_organization_type:tv_engineering_department
				}
				set_variable = { name = tv_wonder_concept_progress value = 100 }
			}
			tv_wonder_complete_concept_effect = yes
		}
	}
}

tv_engineering_department.201 = {
	type = country_event
	title = tv_engineering_department.201.t
	desc = tv_engineering_department.201.d
	image = "gfx/interface/icons/towards_victory/wonders/tv_wonder_complete_cropped.dds"
	outcome = positive

	immediate = {
		if = {
			limit = { tv_wonder_construction_site_selected_trigger = yes }
			var:tv_wonder_site ?= { save_scope_as = tv_wonder_event_location }
		}
	}

	option = {
		name = tv_engineering_department.201.a
		if = {
			limit = {
				tv_wonder_construction_site_selected_trigger = yes
				has_variable = tv_wonder_previous_level
				var:tv_wonder_level > var:tv_wonder_previous_level
			}
			tv_wonder_apply_helper_building_to_final_site_effect = yes
			tv_wonder_materialize_module_buildings_from_unit_vars_effect = yes
			tv_change_wonder_domestic_support_effect = { value = 20 }
			tv_wonder_after_unit_completed_event_effect = yes
			hidden_effect = {
				if = {
					limit = {
						tv_wonder_construction_site_selected_trigger = yes
						NOT = { has_variable = tv_wonder_construction_complete }
						var:tv_wonder_level ?= { this >= 1 }
						OR = {
							AND = { var:tv_wonder_scale_tier ?= 1 var:tv_wonder_level ?= { this >= 1 } }
							AND = { var:tv_wonder_scale_tier ?= 2 var:tv_wonder_level ?= { this >= 2 } }
							AND = { var:tv_wonder_scale_tier ?= 3 var:tv_wonder_level ?= { this >= 3 } }
							AND = { var:tv_wonder_scale_tier ?= 4 var:tv_wonder_level ?= { this >= 4 } }
							AND = { var:tv_wonder_scale_tier ?= 5 var:tv_wonder_level ?= { this >= 5 } }
							AND = { var:tv_wonder_scale_tier ?= 6 var:tv_wonder_level ?= { this >= 6 } }
						}
					}
					tv_wonder_finish_construction_effect = yes
				}
			}
		}
		else = {
			hidden_effect = {
				remove_variable = tv_wonder_previous_level
			}
		}
	}
}

tv_engineering_department.300 = {
	type = country_event
	title = tv_engineering_department.300.t

		first_valid = {
			triggered_desc = {
				trigger = { var:tv_wonder_survey_competence_average >= 75 }

			}
			triggered_desc = {
				trigger = { var:tv_wonder_survey_competence_average >= 50 }
				desc = tv_engineering_department.300.d_mid
			}
			desc = tv_engineering_department.300.d_low
		}
	}
	image = "gfx/interface/icons/towards_victory/wonders/tv_wonder_construction_cropped.dds"
	outcome = positive

	immediate = {
		if = {
			limit = { tv_wonder_survey_site_selected_trigger = yes }
			var:tv_wonder_survey_site ?= { save_scope_as = tv_wonder_event_location }
		}
		set_variable = { name = tv_wonder_survey_competence_average value = var:tv_wonder_scale_competence }
		change_variable = { name = tv_wonder_survey_competence_average add = var:tv_wonder_logistics_competence }

		change_variable = { name = tv_wonder_survey_competence_average divide = 3 }
