@tv_wonder_zoom_step_near = 2

@tv_wonder_gradient_alpha_inside = 1
@tv_wonder_gradient_alpha_outside = 1
@tv_wonder_gradient_width = 0.25
@tv_wonder_gradient_color_mult = 0.9
@tv_wonder_edge_width = 0
@tv_wonder_edge_sharpness = 0.01
@tv_wonder_edge_alpha = 0
@tv_wonder_edge_color_mult = 0
@tv_wonder_before_lighting_blend = 0.5
@tv_wonder_after_lighting_blend = 0.5

tv_wonder = {
	map_color = {
		if = {
			limit = { is_ownable = no }
			value = rgb { 128 128 128 }
		}
		else_if = {
			limit = {
				has_variable = tv_wonder_map_unique_level
				var:tv_wonder_map_unique_level ?= { this > 0 }
			}
			lerp = {
				min_color = rgb { 176 226 141 }
				max_color = rgb { 31 151 80 }
				factor = {
					value = var:tv_wonder_map_unique_level
					divide = 6
				}
			}
		}
		else_if = {
			limit = {
				has_variable = tv_wonder_map_generic_level
				var:tv_wonder_map_generic_level ?= { this > 0 }
			}
			lerp = {
				min_color = rgb { 139 190 255 }
				max_color = rgb { 35 103 222 }
				factor = {
					value = var:tv_wonder_map_generic_level
					divide = 6
				}
			}
		}
		else_if = {
			limit = { has_variable = tv_wonder_map_has_potential_unique }
			value = rgb { 255 181 181 }
		}
		else = {
			value = rgb { 249 214 120 }
		}
	}

	legend_key = { desc = "MAPMODE_TV_WONDER_LEGEND_UNIQUE" color = rgb { 31 151 80 } }
	legend_key = { desc = "MAPMODE_TV_WONDER_LEGEND_GENERIC" color = rgb { 35 103 222 } }
	legend_key = { desc = "MAPMODE_TV_WONDER_LEGEND_POTENTIAL" color = rgb { 255 181 181 } }
	legend_key = { desc = "MAPMODE_TV_WONDER_LEGEND_NONE" color = rgb { 249 214 120 } }

	tooltip_key = {
		if = {
			limit = { is_ownable = no }
			value = MAPMODE_TV_WONDER_TT_WATER
		}
		else_if = {
			limit = {
				has_variable = tv_wonder_map_unique_level
				var:tv_wonder_map_unique_level ?= { this > 0 }
			}
			value = MAPMODE_TV_WONDER_TT_UNIQUE
		}
		else_if = {
			limit = {
				has_variable = tv_wonder_map_generic_level
				var:tv_wonder_map_generic_level ?= { this > 0 }
			}
			value = MAPMODE_TV_WONDER_TT_GENERIC
		}
		else_if = {
			limit = { has_variable = tv_wonder_map_has_potential_unique }
			value = MAPMODE_TV_WONDER_TT_POTENTIAL
		}
		else = {
			value = MAPMODE_TV_WONDER_TT_NONE
		}
	}

	small_map_names = location
	medium_map_names = location
	large_map_names = location

	small_tooltip_context = location
	medium_tooltip_context = location
	large_tooltip_context = location

	fill_in_impassable = yes
	enable_snow = no

	flatmap_behaviour = always
	use_fow = no

	category = economy
	index = 3

	map_markers = {
		all = no
	}

	gradient_parameters = {
		zoom_step = @tv_wonder_zoom_step_near
		gradient_alpha_inside = @tv_wonder_gradient_alpha_inside
		gradient_alpha_outside = @tv_wonder_gradient_alpha_outside
		gradient_width = @tv_wonder_gradient_width
		gradient_color_mult = @tv_wonder_gradient_color_mult
		edge_width = @tv_wonder_edge_width
		edge_sharpness = @tv_wonder_edge_sharpness
		edge_alpha = @tv_wonder_edge_alpha
		edge_color_mult = @tv_wonder_edge_color_mult
		before_lighting_blend = @tv_wonder_before_lighting_blend
		after_lighting_blend = @tv_wonder_after_lighting_blend
	}

	refresh_colors_on_selection_change = yes
	color_refresh_counters = { Day }
	color_and_names_refresh_counters = { Day }
}
