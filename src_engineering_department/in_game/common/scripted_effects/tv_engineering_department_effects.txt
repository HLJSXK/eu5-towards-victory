tv_engineering_department_create_effect = {
	save_scope_as = tv_engineering_department_founder
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_engineering_department_member }
			}
		}
		set_variable = { name = tv_engineering_department_member value = 1 }
	}
	if = {
		limit = {
			custom_tooltip = {
				text = TV_HAS_VARIABLE_NOT_SET_TT
				NOT = { has_variable = tv_wonder_stage }
			}
		}
		set_variable = { name = tv_wonder_stage value = 0 }
	}
	if = {
		limit = {
			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:tv_engineering_department
			}
		}
		# Membership identifies this founder-only IO even if a damaged save has
		# lost or misplaced its leader. Repair that IO instead of creating a
		# second instance whose country effects would run in parallel.
		every_international_organizations_member_of = {
			limit = {
				international_organization_type = international_organization_type:tv_engineering_department
			}
			if = {
				limit = {
					NOT = { leader_country ?= scope:tv_engineering_department_founder }
				}
				set_leader_country = scope:tv_engineering_department_founder
			}
		}
	}
	else = {
		create_international_organization = {
			type = international_organization_type:tv_engineering_department
			add_country_to_international_organization = prev
			set_leader_country = prev
		}
	}
	tv_wonder_initialize_building_state_light_effect = yes
}

tv_wonder_update_construction_progress_display_effect = {
	tv_wonder_set_required_progress_effect = yes
	tv_wonder_update_logistics_metrics_effect = yes
	set_variable = { name = tv_wonder_current_unit_progress value = 0 }
	every_international_organizations_member_of = {
		limit = {
			international_organization_type = international_organization_type:tv_engineering_department
		}
		prev = {
			set_variable = { name = tv_wonder_current_unit_progress value = prev.var:tv_wonder_construction_progress }
		}
	}
	if = {
		limit = {
			has_variable = tv_wonder_unit_required_progress
			var:tv_wonder_current_unit_progress > var:tv_wonder_unit_required_progress
		}
		set_variable = { name = tv_wonder_current_unit_progress value = var:tv_wonder_unit_required_progress }
	}
	clamp_variable = { name = tv_wonder_current_unit_progress min = 0 max = 300000 }
	set_variable = { name = tv_wonder_construction_progress_pct value = 0 }
	set_variable = { name = tv_wonder_remaining_progress value = 0 }
	set_variable = { name = tv_wonder_remaining_months value = 0 }
	set_variable = { name = tv_wonder_remaining_years value = 0 }
	set_variable = { name = tv_wonder_remaining_months_mod value = 0 }
	if = {
		limit = {
			has_variable = tv_wonder_unit_required_progress
			var:tv_wonder_unit_required_progress > 0
		}
		set_variable = { name = tv_wonder_construction_progress_pct value = var:tv_wonder_current_unit_progress }
		change_variable = { name = tv_wonder_construction_progress_pct multiply = 100 }
		change_variable = { name = tv_wonder_construction_progress_pct divide = var:tv_wonder_unit_required_progress }
		clamp_variable = { name = tv_wonder_construction_progress_pct min = 0 max = 100 }
		set_variable = { name = tv_wonder_remaining_progress value = var:tv_wonder_unit_required_progress }
		change_variable = { name = tv_wonder_remaining_progress subtract = var:tv_wonder_current_unit_progress }
		clamp_variable = { name = tv_wonder_remaining_progress min = 0 max = 300000 }
	}
	if = {
		limit = {
			var:tv_wonder_remaining_progress ?= { this > 0 }
			var:tv_wonder_monthly_construction_progress ?= { this > 0 }
		}
		set_variable = {
			name = tv_wonder_remaining_months
			value = {
				value = var:tv_wonder_remaining_progress
				divide = var:tv_wonder_monthly_construction_progress
				ceiling = yes
			}
		}
		if = {
			limit = { var:tv_wonder_remaining_months < 1 }
			set_variable = { name = tv_wonder_remaining_months value = 1 }
		}
		set_variable = {
			name = tv_wonder_remaining_years
			value = {
				value = var:tv_wonder_remaining_months
				divide = 12
				floor = yes
			}
		}
		set_variable = { name = tv_wonder_remaining_months_mod value = var:tv_wonder_remaining_months }
		set_variable = { name = tv_wonder_remaining_year_months_temp value = var:tv_wonder_remaining_years }
		change_variable = { name = tv_wonder_remaining_year_months_temp multiply = 12 }
		change_variable = { name = tv_wonder_remaining_months_mod subtract = var:tv_wonder_remaining_year_months_temp }
		clamp_variable = { name = tv_wonder_remaining_months_mod min = 0 max = 11 }
		remove_variable = tv_wonder_remaining_year_months_temp
	}
}

tv_wonder_set_current_project_priority_candidate_mode_effect = {
	remove_variable = tv_wonder_priority_candidate_current_mode
	if = {
		limit = {
			has_variable = tv_wonder_locked
			tv_wonder_construction_site_selected_trigger = yes
		}
		if = {
			limit = {
				NOT = { has_variable = tv_wonder_finalized }
				OR = {
					AND = {
						var:tv_wonder_project_mode ?= 2
						OR = {
							var:tv_wonder_level > var:tv_wonder_project_base_level
							var:tv_wonder_foundation_units > var:tv_wonder_level
							var:tv_wonder_body_units > var:tv_wonder_level
							var:tv_wonder_function_units > var:tv_wonder_level
							var:tv_wonder_decoration_units > var:tv_wonder_level
						}
					}
					AND = {
						NOT = { var:tv_wonder_project_mode ?= 2 }
						OR = {
							var:tv_wonder_level ?= { this >= 1 }
							var:tv_wonder_foundation_units ?= { this >= 1 }
							var:tv_wonder_body_units ?= { this >= 1 }
							var:tv_wonder_function_units ?= { this >= 1 }
							var:tv_wonder_decoration_units ?= { this >= 1 }
						}
					}
				}
			}
			set_variable = { name = tv_wonder_priority_candidate_current_mode value = 1 }
		}
		else_if = {
			limit = {
				OR = {
					has_variable = tv_wonder_finalized
					var:tv_wonder_project_mode ?= 2
				}
				var:tv_wonder_level ?= { this >= 1 }
				var:tv_wonder_level ?= { this < 6 }
				var:tv_wonder_scale_tier > var:tv_wonder_level
			}
			set_variable = { name = tv_wonder_priority_candidate_current_mode value = 2 }
		}
	}
}
