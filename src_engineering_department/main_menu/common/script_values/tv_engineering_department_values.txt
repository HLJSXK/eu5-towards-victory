# Engineering Department - reusable construction progress values

# Native shortage-adjusted work efficiency of the active Great Project worksite.
tv_wonder_labor_camp_input_goods_access = {
	value = var:tv_wonder_site.building_efficiency:tv_wonder_labor_camp
	min = 0
	max = 1
}

# Progress lost after logistics because the worksite cannot obtain all inputs.
tv_wonder_input_shortage_construction_progress = {
	value = var:tv_wonder_monthly_construction_progress
	multiply = {
		value = 1
		subtract = tv_wonder_labor_camp_input_goods_access
	}
	min = 0
}

# Absolute gold prices for direct construction. CMM values are stored in thousands.
tv_wonder_direct_build_small_price_value = {
	value = 10000
	if = {
		limit = { has_global_variable = tv_engineering_department_direct_build_small_price }
		multiply = 0
		add = {
			value = global_var:tv_engineering_department_direct_build_small_price
			multiply = 1000
		}
	}
	min = 0
	max = 50000
}
