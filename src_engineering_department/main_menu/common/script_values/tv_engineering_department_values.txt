# Engineering Department - reusable construction progress values

# Native shortage-adjusted work efficiency of the active Great Project worksite.
# building_efficiency is readable as a quoted function-call-style value link,
# e.g. "location.building_efficiency(scope:saved_building_type)". The logistics
# refresher caches the active site's labor-camp efficiency as a country variable
# so this monthly-change value can reuse it safely.
# Progress lost after logistics because the worksite cannot obtain all inputs.
tv_wonder_input_shortage_construction_progress = {
	value = var:tv_wonder_monthly_construction_progress
	multiply = {
		value = 1
		subtract = var:tv_wonder_labor_camp_efficiency
	}
	min = 0
}
